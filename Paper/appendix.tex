\section{Proofs for Section~\ref{sec: preliminaries}}
\label{app: elliptic}

Appendices~\ref{sec: dissipative} to \ref{sec: elliptic conclusion} prove Theorem~\ref{thm: u decay}: the exponential decay of the Kolmogorov solution $u$ of a Lipschitz test function and of its derivatives of orders up to $4$, with the singularity $t^{-\frac{j-1}{2}}$ of the derivative of order $j$ at $t = 0$. The estimates of orders up to $3$ are those of \citep{ye2026mse}, so essentially only the derivative of order $4$ needs to be estimated.\footnote{The paper \citep{ye2026mse} is an earlier work of the first author, Xuda Ye, and is under review at the IMA Journal of Numerical Analysis. Beyond it, the proofs of this paper use no self-cited results that are not yet peer-reviewed.} The bound \eqref{eq: poisson bound} on $e^{h\mathcal L} \phi_h$ then follows by summation over the time steps. Appendix~\ref{sec: dissipative} records the dissipativity of $U$ and the moments of $\pi$. Appendix~\ref{sec: variations} writes out the variation processes and their pathwise bounds. Appendix~\ref{sec: semigroup} states the estimates on the semigroup $e^{t\mathcal L}$ that the proof uses and proves the short-time bound \eqref{eq: short time}. For a function $f$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$, Appendix~\ref{sec: fourth} bounds $\nabla^4 (e^{t\mathcal L} f)$ by differentiating the derivative formula \eqref{eq: BEL} $3$ times. Appendix~\ref{sec: elliptic conclusion} combines this bound with the decay \eqref{eq: grad decay} and the short-time bound \eqref{eq: short time} into Lemma~\ref{lem: smoothing}, which also covers the modified dynamics \eqref{eq: modified langevin}, and deduces Theorem~\ref{thm: u decay}. Appendix~\ref{sec: riesz well-posedness} contains the proofs for the Riesz representors of Section~\ref{subsec: riesz}.

The proof of Theorem~\ref{thm: u decay} uses the derivatives of $U$ up to order $5$. Throughout this appendix, the constant $C$ depends only on $m$, $M$, $R$ and $d$, unless otherwise specified.

\subsection{Dissipativity of the potential}
\label{sec: dissipative}

The next lemma records the dissipativity of $U$ used in the moment estimates, and the Gaussian tail of $\pi$ that follows from it.

\begin{lemma}
\label{lem: dissipative}
    Under Assumption~\ref{ass: U}, for all $x \in \mathbb R^d$,
    \begin{equation}
        \langle x, \nabla U(x) \rangle \Ge \frac{m}{4} |x|^2 - C , \qquad U(x) - U(0) \Ge \frac{m}{8} |x|^2 - C ,
        \label{eq: dissipative bounds}
    \end{equation}
    and consequently $\pi(x) \Le C e^{-\frac{m}{8}|x|^2}$ and $\int_{\mathbb R^d} |x|^q\, \pi(\D x) \Le C_q$ for every $q \Ge 1$.
\end{lemma}

\begin{proof}
    By \citep[Lemma~4]{ye2026mse}, with $R_1 := 4R(1+M/m)$,
    \begin{equation}
        \langle \nabla U(x) - \nabla U(y), x-y \rangle \Ge -M |x-y|^2 , \quad
        \langle \nabla U(x) - \nabla U(y), x-y \rangle \Ge \frac{m}{2} |x-y|^2 \ \text{ for } |x-y| \Ge R_1 .
        \label{eq: dissipative}
    \end{equation}
    Applying \eqref{eq: dissipative} to the pair $(rx,0)$ with $r \in [0,1]$ and using $|\nabla U(0)| \Le M$ give
    \begin{equation}
        \langle \nabla U(rx), x \rangle \Ge \frac m2\, r\, |x|^2 - C |x| , \qquad r \in [0,1] .
        \label{eq: radial dissipative}
    \end{equation}
    Taking $r = 1$ in \eqref{eq: radial dissipative} gives the bound on $\langle x, \nabla U(x) \rangle$ in \eqref{eq: dissipative bounds}, and integrating \eqref{eq: radial dissipative} over $r \in [0,1]$ gives the bound on $U(x) - U(0)$. Since $U(y) - U(0) \Le M|y| + \frac{M}{2}|y|^2$ bounds the normalizing constant of $\pi$ from below, the bound on $U(x) - U(0)$ gives $\pi(x) \Le C e^{-\frac m8 |x|^2}$, and hence $\int_{\mathbb R^d} |x|^q\, \pi(\D x) \Le C_q$ for every $q \Ge 1$.
\end{proof}

\subsection{Variation processes}
\label{sec: variations}

Let $X_t$ be the solution of \eqref{eq: langevin} started at $X_0 = x$, and write $\partial_w$ for the derivative with respect to the initial state $x$ along a vector $w$. For a function $f$ on $\mathbb R^d$, the derivatives of $e^{t\mathcal L} f$ in the initial state are expressed through the variation processes. The variation processes of orders $1$ to $4$ are
\begin{equation}
    J_t := \partial_x X_t , \qquad
    K_t[w] := \partial_w J_t , \qquad
    \Theta_t[w,w'] := \partial_{w'} K_t[w] , \qquad
    \Xi_t[w,w',w''] := \partial_{w''} \Theta_t[w,w'] ,
    \label{eq: variation processes}
\end{equation}
where the variations of orders $2$ to $4$ are taken along vectors $w, w', w'' \in \mathbb R^d$, so that, like $J_t$, they take values in $\mathbb R^{d \times d}$. From the definition \eqref{eq: variation processes},
\begin{equation*}
\begin{gathered}
    J_t w = \partial_w X_t , \qquad K_t[w] w' = \partial_w \partial_{w'} X_t , \\
    \Theta_t[w,w'] w'' = \partial_w \partial_{w'} \partial_{w''} X_t , \qquad \Xi_t[w,w',w''] w''' = \partial_w \partial_{w'} \partial_{w''} \partial_{w'''} X_t ,
\end{gathered}
\end{equation*}
hence $K_t[w] w'$, $\Theta_t[w,w'] w''$ and $\Xi_t[w,w',w''] w'''$ are symmetric in all their directions. Since the noise of \eqref{eq: langevin} is additive, differentiating the dynamics in its initial state gives ordinary differential equations along the path, with the initial values $J_0 = I_d$ and $K_0 = \Theta_0 = \Xi_0 = 0$:
\begin{align}
    \frac{\D J_t}{\D t} &= -\nabla^2 U(X_t)\, J_t , \label{eq: first variation}\\
    \frac{\D K_t[w]}{\D t} &= -\nabla^2 U(X_t)\, K_t[w] - \nabla^3 U(X_t)[J_t w]\, J_t , \label{eq: second variation}\\
    \frac{\D \Theta_t[w,w']}{\D t} &= -\nabla^2 U(X_t)\, \Theta_t[w,w'] - \nabla^3 U(X_t)[J_t w]\, K_t[w'] - \nabla^3 U(X_t)[J_t w']\, K_t[w] \notag\\
    &\qquad - \nabla^3 U(X_t)\bigl[ K_t[w] w' \bigr]\, J_t - \nabla^4 U(X_t)[J_t w, J_t w']\, J_t . \label{eq: third variation}
\end{align}
These are the equations of the variation processes $J_t$, $K_t[w]$ and $\Theta_t[w,w']$ bounded in \citep[Lemma~6]{ye2026mse}. Differentiating \eqref{eq: third variation} once more, along $w''$, gives the equation of the variation $\Xi_t$ of order $4$,
\begin{align}
    & \frac{\D \Xi_t[w,w',w'']}{\D t} = -\nabla^2 U(X_t)\, \Xi_t[w,w',w'']
    - \sum_{(2,1)} \nabla^3 U(X_t)[J_t w'']\, \Theta_t[w,w']
    - \sum_{(2,1)} \nabla^3 U(X_t)\bigl[ K_t[w] w' \bigr]\, K_t[w''] \notag \\
    &\qquad - \nabla^3 U(X_t)\bigl[ \Theta_t[w,w'] w'' \bigr]\, J_t
    - \sum_{(2,1)} \nabla^4 U(X_t)[J_t w, J_t w']\, K_t[w''] \notag \\
    &\qquad - \sum_{(2,1)} \nabla^4 U(X_t)\bigl[ K_t[w] w', J_t w'' \bigr]\, J_t
    - \nabla^5 U(X_t)[J_t w, J_t w', J_t w'']\, J_t ,
    \label{eq: variations}
\end{align}
where, for an expression $F$ of the $3$ directions in which the underlined pair forms a block, the sum runs over the partitions of $\{w,w',w''\}$ into a pair and a singleton,
\begin{equation}
    \sum_{(2,1)} F(\underline{w,w'}, w'') := F(\underline{w,w'}, w'') + F(\underline{w,w''}, w') + F(\underline{w',w''}, w).
    \label{eq: sum21}
\end{equation}
In each sum $\sum_{(2,1)}$ of \eqref{eq: variations}, the summand written out has the pair $\{w,w'\}$ and the singleton $w''$.

The bounds of the next lemma are Gr\"onwall estimates, which grow exponentially in $t$. The factor $t$ in the bounds on $K_t$, $\Theta_t$ and $\Xi_t$ is what the proof of Lemma~\ref{lem: fourth from lipschitz} uses at short times.

\begin{lemma}
\label{lem: variation bounds}
    Under Assumption~\ref{ass: U}, pathwise, for all $t \Ge 0$ and unit vectors $w, w', w''$,
    \begin{equation}
        \norm{J_t} \Le e^{M t} , \qquad
        \norm{K_t[w]} + \norm{\Theta_t[w,w']} + \norm{\Xi_t[w,w',w'']} \Le C\, t\, e^{C t} ,
        \label{eq: variation bounds}
    \end{equation}
    with $C$ depending only on $M$.
\end{lemma}

\begin{proof}
    The bounds on $J_t$, $K_t$ and $\Theta_t$ follow from \citep[Lemma~6]{ye2026mse}. With $\norm{\nabla^i U} \Le M$ for $i = 3, 4, 5$, they bound every term on the right side of \eqref{eq: variations} other than $-\nabla^2 U(X_t)\, \Xi_t[w,w',w'']$ by $C e^{Ct}$. Since $\Xi_0 = 0$ and $\norm{\nabla^2 U} \Le M$, Gr\"onwall's inequality gives $\norm{\Xi_t[w,w',w'']} \Le C\, t\, e^{Ct}$.
\end{proof}

\subsection{Semigroup estimates}
\label{sec: semigroup}

The estimates of this subsection hold for every $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$; the rate $\lambda > 0$ depends only on $m$, $M$, $R$ and $d$. Lemmas~\ref{lem: semigroup}, \ref{lem: BEL} and \ref{lem: variance bound} are proved in \citep[Appendix~A]{ye2026mse} under the bounds of Assumption~\ref{ass: U} of orders up to $4$, and are restated without proof, since the proof of Lemma~\ref{lem: fourth from lipschitz} uses them explicitly. Lemma~\ref{lem: short time} extends the short-time bounds that follow from \citep[Lemma~6]{ye2026mse} to the derivative of order $4$ and is proved here.

\begin{lemma}
\label{lem: semigroup}
    Under Assumption~\ref{ass: U}, for all $t \Ge 0$,
    \begin{equation}
        \bigl\| \nabla (e^{t\mathcal L} f) \bigr\|_\infty \Le C e^{-\lambda t} \norm{\nabla f}_\infty ;
        \label{eq: grad decay}
    \end{equation}
    and for $0 < t \Le 1$,
    \begin{equation}
        \bigl\| \nabla^2 (e^{t\mathcal L} f) \bigr\|_\infty \Le C\, t^{-\frac12}\, \norm{\nabla f}_\infty , \qquad
        \bigl\| \nabla^3 (e^{t\mathcal L} f) \bigr\|_\infty \Le C\, t^{-1}\, \norm{\nabla f}_\infty .
        \label{eq: smoothing 23}
    \end{equation}
\end{lemma}

The decay \eqref{eq: grad decay} follows from \citep[Lemma~5]{ye2026mse}, obtained from the reflection coupling of the overdamped Langevin dynamics \citep{eberle2016reflection}. The smoothing bounds \eqref{eq: smoothing 23} are \citep[Lemmas~8 and~9]{ye2026mse}, obtained by differentiating the derivative formula \eqref{eq: BEL} below up to $2$ times in the initial state.

The derivative formula \citep{elworthy1994formulae}, in the form of \citep[Lemma~7]{ye2026mse}, expresses the derivative of $e^{2\tau\mathcal L} f$ along a vector $v$ through a stochastic integral of the variation process $J_s$ of order $1$.

\begin{lemma}
\label{lem: BEL}
    Under Assumption~\ref{ass: U}, for $\tau > 0$, $\psi := e^{\tau\mathcal L} f$ and $v \in \mathbb R^d$,
    \begin{equation}
        v \cdot \nabla (e^{2\tau\mathcal L} f)(x) = \frac{1}{\sqrt2\,\tau}\, \mathbb E^x \biggl[ \psi(X_\tau) \int_0^\tau (J_s v) \cdot \D B_s \biggr] , \qquad \forall x \in \mathbb R^d .
        \label{eq: BEL}
    \end{equation}
\end{lemma}

The next lemma, from the proof of \citep[Lemma~8]{ye2026mse}, bounds the variance of $\psi(X_\tau)$.

\begin{lemma}
\label{lem: variance bound}
    Under Assumption~\ref{ass: U}, for $\tau > 0$, $\psi := e^{\tau\mathcal L} f$ and $X_0 = x \in \mathbb R^d$,
    \begin{equation}
        \Var\bigl( \psi(X_\tau) \bigr) \Le 2 d\, \tau\, e^{4M\tau}\, \norm{\nabla f}_\infty^2 .
        \label{eq: variance bound}
    \end{equation}
\end{lemma}
The randomness of $\psi(X_\tau)$ comes from the Brownian motion $(B_t)_{t \Ge 0}$ on $[0,\tau]$, which drives $X_t$ from $X_0 = x$.

The last estimate of this subsection bounds the derivatives of $e^{t\mathcal L} f$ by those of $f$, with a constant that grows exponentially in $t$.

\begin{lemma}
\label{lem: short time}
    Under Assumption~\ref{ass: U}, there is $C$, depending only on $M$, such that for every $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$ and $t \Ge 0$,
    \begin{equation}
        \bigl\| \nabla^j (e^{t\mathcal L} f) \bigr\|_\infty \Le C e^{C t} \sum_{i=1}^{j} \norm{\nabla^i f}_\infty , \qquad j = 1,2,3,4 .
        \label{eq: short time}
    \end{equation}
\end{lemma}

\begin{proof}
    For $j \Le 3$, \eqref{eq: short time} follows from the bounds of \citep[Lemma~6]{ye2026mse}, by the argument given for $j = 4$ below. For $j = 4$, fix $x \in \mathbb R^d$ and unit vectors $w, w', w'', w'''$. By the chain rule and \eqref{eq: variation processes}, $\partial_w \partial_{w'} \partial_{w''} \partial_{w'''}\, f(X_t)$ is a sum of $15$ terms, each of one of the forms
    \begin{equation*}
    \begin{gathered}
        \nabla^4 f(X_t)[J_t w, J_t w', J_t w'', J_t w'''] , \qquad
        \nabla^3 f(X_t)\bigl[ K_t[w] w', J_t w'', J_t w''' \bigr] , \\
        \nabla^2 f(X_t)\bigl[ \Theta_t[w,w'] w'', J_t w''' \bigr] , \qquad
        \nabla^2 f(X_t)\bigl[ K_t[w] w', K_t[w''] w''' \bigr] , \qquad
        \nabla f(X_t) \cdot \Xi_t[w,w',w''] w''' ,
    \end{gathered}
    \end{equation*}
    up to a permutation of $w$, $w'$, $w''$ and $w'''$. By \eqref{eq: variation bounds}, a term with the factor $\nabla^i f(X_t)$ is bounded pathwise by $C e^{Ct} \norm{\nabla^i f}_\infty$. Hence $\nabla^4 (e^{t\mathcal L} f)(x)[w,w',w'',w''']$ is the expectation $\mathbb E^x$ of this sum, and taking the supremum over $x$ and over the unit vectors gives \eqref{eq: short time} for $j = 4$.
\end{proof}

\subsection{The derivative of order $4$}
\label{sec: fourth}

This subsection extends the smoothing bounds \eqref{eq: smoothing 23} to the derivative of order $4$.

\begin{lemma}
\label{lem: fourth from lipschitz}
    Under Assumption~\ref{ass: U}, for every $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$ and $0 < t \Le 1$,
    \begin{equation}
        \bigl\| \nabla^4 (e^{t\mathcal L} f)(x) \bigr\| \Le C\, t^{-\frac32}\, \norm{\nabla f}_\infty , \qquad \forall x \in \mathbb R^d .
        \label{eq: fourth from lipschitz}
    \end{equation}
\end{lemma}

\begin{proof}
    Set $\tau = t/2 \Le \frac12$ and $\psi = e^{\tau\mathcal L} f$, so that $e^{2\tau\mathcal L} f = e^{t\mathcal L} f$, and fix $x \in \mathbb R^d$ and unit vectors $v, w, w', w''$. By \eqref{eq: short time}, $\max_{1 \Le j \Le 4} \norm{\nabla^j \psi}_\infty < \infty$. Differentiate \eqref{eq: BEL} in Lemma~\ref{lem: BEL} along $w$, $w'$ and $w''$. By \eqref{eq: variation processes}, a derivative that falls on the stochastic integral raises the order of the variation process in its integrand, from $J_s v$ to $K_s[w] v$, to $\Theta_s[w,w'] v$ and to $\Xi_s[w,w',w''] v$. Distributing the $3$ derivatives between $\psi(X_\tau)$ and the stochastic integral gives
    \begin{align}
        & \sqrt2\, \tau\, \nabla^4 (e^{t\mathcal L} f)(x)[v,w,w',w'']  = \underbrace{\mathbb E^x \biggl[ \partial_w \partial_{w'} \partial_{w''} \psi(X_\tau) \int_0^\tau (J_s v) \cdot \D B_s \biggr]}_{=: \mathcal T_1} \notag \\
        & \quad + \underbrace{\sum_{(2,1)} \mathbb E^x \biggl[ \partial_w \partial_{w'} \psi(X_\tau) \int_0^\tau \bigl( K_s[w''] v \bigr) \cdot \D B_s \biggr]}_{=: \mathcal T_2} + \underbrace{\sum_{(2,1)} \mathbb E^x \biggl[ \partial_{w''} \psi(X_\tau) \int_0^\tau \bigl( \Theta_s[w,w'] v \bigr) \cdot \D B_s \biggr]}_{=: \mathcal T_3} \notag \\
        & \quad + \underbrace{\mathbb E^x \biggl[ \Bigl( \psi(X_\tau) - \mathbb E^x \bigl[ \psi(X_\tau) \bigr] \Bigr) \int_0^\tau \bigl( \Xi_s[w,w',w''] v \bigr) \cdot \D B_s \biggr]}_{=: \mathcal T_4} ,
        \label{eq: BEL third}
    \end{align}
    where $\sum_{(2,1)}$ is the sum over the partitions of $\{w,w',w''\}$ defined in \eqref{eq: sum21}.

    In $\mathcal T_4$, $\psi(X_\tau)$ is centered at its mean $\mathbb E^x \bigl[ \psi(X_\tau) \bigr] = e^{t\mathcal L} f(x)$. The centering does not change the value of $\mathcal T_4$, since the stochastic integral has mean $0$, and it makes \eqref{eq: variance bound} applicable.

    The derivatives of $\psi(X_\tau)$ in \eqref{eq: BEL third} follow from the chain rule and \eqref{eq: variation processes}:
    \begin{equation*}
    \begin{gathered}
        \partial_{w''} \psi(X_\tau) = \nabla \psi(X_\tau) \cdot J_\tau w'' , \qquad
        \partial_w \partial_{w'} \psi(X_\tau) = \nabla^2 \psi(X_\tau)[J_\tau w, J_\tau w'] + \nabla \psi(X_\tau) \cdot K_\tau[w] w' , \\
        \partial_w \partial_{w'} \partial_{w''} \psi(X_\tau) = \nabla^3 \psi(X_\tau)[J_\tau w, J_\tau w', J_\tau w''] + \sum_{(2,1)} \nabla^2 \psi(X_\tau)\bigl[ K_\tau[w] w', J_\tau w'' \bigr] + \nabla \psi(X_\tau) \cdot \Theta_\tau[w,w'] w'' .
    \end{gathered}
    \end{equation*}
    Since $\tau \Le \frac12$, \eqref{eq: short time} and \eqref{eq: smoothing 23} at time $\tau$ give
    \begin{equation*}
        \norm{\nabla\psi}_\infty \Le C\norm{\nabla f}_\infty , \qquad
        \norm{\nabla^2\psi}_\infty \Le C\tau^{-\frac12}\norm{\nabla f}_\infty , \qquad
        \norm{\nabla^3 \psi}_\infty \Le C \tau^{-1}\norm{\nabla f}_\infty ,
    \end{equation*}
    and \eqref{eq: variation bounds} gives $\norm{J_s} \Le C$ and $\norm{K_s[w]}, \norm{\Theta_s[w,w']}, \norm{\Xi_s[w,w',w'']} \Le C s$ for $s \Le \tau$. The bounds on $\nabla^j \psi$ and on the variation processes then give
    \begin{equation}
        \bigl| \partial_{w''} \psi(X_\tau) \bigr| \Le C\norm{\nabla f}_\infty , \quad
        \bigl| \partial_w \partial_{w'} \psi(X_\tau) \bigr| \Le C\tau^{-\frac12}\norm{\nabla f}_\infty , \quad
        \bigl| \partial_w \partial_{w'} \partial_{w''} \psi(X_\tau) \bigr| \Le C\tau^{-1}\norm{\nabla f}_\infty ,
        \label{eq: psi derivative bounds}
    \end{equation}
    and \eqref{eq: variance bound} bounds the centered factor of $\mathcal T_4$:
    \begin{equation}
        \Var\bigl( \psi(X_\tau) \bigr)^{\frac12} = \biggl( \mathbb E^x \Bigl| \psi(X_\tau) - \mathbb E^x \bigl[ \psi(X_\tau) \bigr] \Bigr|^2 \biggr)^{\frac12} \Le C \tau^{\frac12} \norm{\nabla f}_\infty .
        \label{eq: centered factor}
    \end{equation}
    By the It\^o isometry and \eqref{eq: variation bounds},
    \begin{equation}
    \begin{gathered}
        \mathbb E^x \biggl| \int_0^\tau (J_s v) \cdot \D B_s \biggr|^2 \Le C\tau , \qquad
        \mathbb E^x \biggl| \int_0^\tau \bigl( K_s[w''] v \bigr) \cdot \D B_s \biggr|^2 \Le C\tau^3 , \\
        \mathbb E^x \biggl| \int_0^\tau \bigl( \Theta_s[w,w'] v \bigr) \cdot \D B_s \biggr|^2 \Le C\tau^3 , \qquad
        \mathbb E^x \biggl| \int_0^\tau \bigl( \Xi_s[w,w',w''] v \bigr) \cdot \D B_s \biggr|^2 \Le C\tau^3 ,
    \end{gathered}
    \label{eq: ito bounds}
    \end{equation}
    and these bounds hold for every term of the sums $\sum_{(2,1)}$ in \eqref{eq: BEL third}, since $w$, $w'$ and $w''$ are arbitrary unit vectors. The Cauchy--Schwarz inequality, with \eqref{eq: psi derivative bounds}, \eqref{eq: ito bounds} and \eqref{eq: centered factor}, gives
    \begin{equation*}
        |\mathcal T_1| \Le C \tau^{-\frac12} \norm{\nabla f}_\infty , \qquad
        |\mathcal T_2| \Le C \tau\, \norm{\nabla f}_\infty , \qquad
        |\mathcal T_3| \Le C \tau^{\frac32} \norm{\nabla f}_\infty , \qquad
        |\mathcal T_4| \Le C \tau^2 \norm{\nabla f}_\infty .
    \end{equation*}
    Since $\tau \Le \frac12$, dividing \eqref{eq: BEL third} by $\sqrt2\, \tau$ and taking the supremum over the unit vectors gives
    \begin{equation*}
        \bigl\| \nabla^4 (e^{t\mathcal L} f)(x) \bigr\| \Le C \tau^{-\frac32} \norm{\nabla f}_\infty ,
    \end{equation*}
    which is \eqref{eq: fourth from lipschitz} with $\tau = t/2$.
\end{proof}

\subsection{Conclusion of the proof}
\label{sec: elliptic conclusion}

The semigroup estimates combine into one bound for the exact and the modified dynamics. The proof technique is essentially that of \citep[Lemma~10]{ye2026mse}, which splits the times into $0 < t \Le 1$ and $t > 1$.

\begin{lemma}
\label{lem: smoothing}
    Under Assumption~\ref{ass: U}, there is $\lambda > 0$, depending only on $m$, $M$, $R$ and $d$, such that for every $h \in (0,h_0]$, $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$ and $t > 0$,
    \begin{equation}
        \norm{\nabla^j (e^{t\mathcal L} f)}_\infty + \norm{\nabla^j (e^{t\tilde{\mathcal L}_h} f)}_\infty \Le C \min(1,t)^{-\frac{j-1}{2}}\, e^{-\lambda t}\, \norm{\nabla f}_\infty , \qquad j = 1, 2, 3, 4 ,
        \label{eq: smoothing}
    \end{equation}
    and for $t \Ge 0$,
    \begin{equation}
        \norm{\nabla^j (e^{t\mathcal L} f)}_\infty + \norm{\nabla^j (e^{t\tilde{\mathcal L}_h} f)}_\infty \Le C e^{-\lambda t} \sum_{i=1}^{j} \norm{\nabla^i f}_\infty , \qquad j = 1, 2, 3, 4 .
        \label{eq: regular decay}
    \end{equation}
\end{lemma}

\begin{proof}
    For $j = 1$, the bound \eqref{eq: smoothing} for $e^{t\mathcal L}$ is the decay \eqref{eq: grad decay}. For $j = 2, 3, 4$ and $0 < t \Le 1$, \eqref{eq: smoothing} for $e^{t\mathcal L}$ follows from \eqref{eq: smoothing 23}, \eqref{eq: fourth from lipschitz} and $e^{-\lambda t} \Ge e^{-\lambda}$. For $j = 2, 3, 4$ and $t > 1$, the semigroup property gives $e^{t\mathcal L} f = e^{\mathcal L} \bigl( e^{(t-1)\mathcal L} f \bigr)$, and $e^{(t-1)\mathcal L} f$ has bounded derivatives of orders $1$ to $4$ by \eqref{eq: short time}. Hence applying \eqref{eq: smoothing 23} and \eqref{eq: fourth from lipschitz} at time $1$ to $e^{(t-1)\mathcal L} f$ and \eqref{eq: grad decay} at time $t - 1$ to $f$ gives
    \begin{equation*}
        \norm{\nabla^j (e^{t\mathcal L} f)}_\infty \Le C \bigl\| \nabla (e^{(t-1)\mathcal L} f) \bigr\|_\infty \Le C e^{-\lambda (t-1)} \norm{\nabla f}_\infty ,
    \end{equation*}
    which is \eqref{eq: smoothing} for $e^{t\mathcal L}$. The bound \eqref{eq: regular decay} for $e^{t\mathcal L}$ follows from \eqref{eq: short time} for $t \Le 1$ and from \eqref{eq: smoothing} for $t > 1$.

    Lemmas~\ref{lem: semigroup}, \ref{lem: short time} and \ref{lem: fourth from lipschitz} use only the bounds of Assumption~\ref{ass: U} of orders up to $5$ and the convexity outside the ball of radius $R$. By Lemma~\ref{lem: modified class}, for every $h \in (0, h_0]$, $\tilde U_h$ satisfies these bounds and the convexity condition with $2M$ and $\frac m2$ in place of $M$ and $m$. These lemmas therefore hold for $e^{t\tilde{\mathcal L}_h}$ with a rate and a constant that depend only on $m$, $M$, $R$ and $d$. The argument for $e^{t\mathcal L}$ then gives \eqref{eq: smoothing} and \eqref{eq: regular decay} for $e^{t\tilde{\mathcal L}_h}$, and $\lambda$ is the smaller of the rates for $U$ and for $\tilde U_h$.
\end{proof}

The next lemma sums the bounds \eqref{eq: smoothing} at the times $t = ih$ over $i \Ge 1$.

\begin{lemma}
\label{lem: time sums}
    Let $\lambda > 0$. For every $h \in (0,\frac12]$ with $\lambda h \Le 1$,
    \begin{equation}
        h \sum_{i=1}^{\infty} \min( 1, ih )^{-\frac{j-1}{2}} e^{-\lambda i h} \Le
        \begin{cases}
            C , & j = 1, 2 , \\
            C \log\frac1h , & j = 3 , \\
            C h^{-\frac12} , & j = 4 ,
        \end{cases}
        \label{eq: time sums}
    \end{equation}
    with $C$ depending only on $\lambda$.
\end{lemma}

\begin{proof}
    Split the sum at $ih = 1$. For the terms with $ih > 1$, $\min(1, ih) = 1$, and since $\lambda h \Le 1$,
    \begin{equation*}
        h \sum_{i > 1/h} e^{-\lambda i h} \Le \frac{h}{1 - e^{-\lambda h}} \Le \frac{2}{\lambda} .
    \end{equation*}
    For the terms with $ih \Le 1$, bound $e^{-\lambda ih}$ by $1$:
    \begin{equation*}
        h \sum_{i=1}^{\lfloor 1/h \rfloor} (ih)^{-\frac{j-1}{2}} = h^{\frac{3-j}{2}} \sum_{i=1}^{\lfloor 1/h \rfloor} i^{-\frac{j-1}{2}} \Le
        \begin{cases}
            1 , & j = 1 , \\
            2 , & j = 2 , \\
            1 + \log\frac1h , & j = 3 , \\
            3 h^{-\frac12} , & j = 4 .
        \end{cases}
    \end{equation*}
    Adding the bound $2/\lambda$ and using $1 \Le 2 \log\frac1h$ for $h \Le \frac12$ give \eqref{eq: time sums}.
\end{proof}

Let $h_0 \Le \min\{ \lambda^{-1}, \frac12 \}$, with $\lambda$ the rate of Lemma~\ref{lem: smoothing}. For every $h \in (0,h_0]$ and $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$, the bounds \eqref{eq: smoothing} at the times $t = ih$ and \eqref{eq: time sums} give
\begin{equation}
    h \sum_{i=1}^{\infty} \Bigl( \norm{\nabla^j ( e^{ih\mathcal L} f )}_\infty + \norm{\nabla^j ( e^{ih\tilde{\mathcal L}_h} f )}_\infty \Bigr) \Le
    \begin{cases}
        C \norm{\nabla f}_\infty , & j = 1, 2 , \\
        C \norm{\nabla f}_\infty \log\frac1h , & j = 3 , \\
        C \norm{\nabla f}_\infty h^{-\frac12} , & j = 4 .
    \end{cases}
    \label{eq: summed smoothing}
\end{equation}

For $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$, the Kolmogorov equation $\partial_r ( e^{r\mathcal L} f ) = \mathcal L ( e^{r\mathcal L} f )$ gives
\begin{equation}
    e^{t\mathcal L} f - e^{s\mathcal L} f = \int_s^t \mathcal L \bigl( e^{r\mathcal L} f \bigr)\, \D r , \qquad 0 \Le s \Le t ,
    \label{eq: backward equation}
\end{equation}
and the same identity holds for $\tilde{\mathcal L}_h$.

\begin{proof}[Proof of Theorem~\ref{thm: u decay}]
    We first take $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$. Since $u(\cdot,t) = e^{t\mathcal L} f - \pi(f)$, \eqref{eq: smoothing} is the bound on $\nabla^j u(\cdot,t)$ in \eqref{eq: u bound}. Then the moments of $\pi$ in Lemma~\ref{lem: dissipative} give
    \begin{equation}
        |u(x,t)| = \biggl| \int_{\mathbb R^d} \bigl( (e^{t\mathcal L} f)(x) - (e^{t\mathcal L} f)(y) \bigr)\, \pi(\D y) \biggr| \Le C \norm{\nabla f}_\infty e^{-\lambda t} \int_{\mathbb R^d} |x - y|\, \pi(\D y) \Le C \norm{\nabla f}_\infty (1 + |x|)\, e^{-\lambda t} ,
        \label{eq: relaxation}
    \end{equation}
    which is the bound on $|u(x,t)|$ in \eqref{eq: u bound}. For a Lipschitz $f$, a mollification of $f$ gives $u(\cdot,t) \in C^3(\mathbb R^d)$ and \eqref{eq: u bound}, where the bound on $\nabla^4 u(\cdot,t)$ is a bound on the Lipschitz constant of $\nabla^3 u(\cdot,t)$.

    For the discrete Poisson solution $\phi_h$ in \eqref{eq: discrete poisson solution}, the bound on $|u(x,t)|$ in \eqref{eq: u bound} gives the convergence of the series, and $e^{h\mathcal L} \phi_h = h \sum_{k=1}^{\infty} u(\cdot, kh)$, with $\nabla^j u(\cdot, kh) = \nabla^j ( e^{kh\mathcal L} f )$. The bound \eqref{eq: summed smoothing}, which the mollification extends to a Lipschitz $f$, then gives \eqref{eq: poisson bound}. If $f$ has bounded derivatives of orders $1$ to $3$, \eqref{eq: regular decay} at $j = 3$ and the times $t = kh$ gives $\norm{\nabla^3 ( e^{h\mathcal L} \phi_h )}_\infty \Le C \max_{1 \Le j \Le 3} \norm{\nabla^j f}_\infty$, the bound of Remark~\ref{rem: regular}.

    Finally, for the continuous Poisson solution $\phi_0$ in \eqref{eq: continuous poisson solution}, \eqref{eq: u bound} gives the convergence of the integral with its derivatives of orders $1$ and $2$, and \eqref{eq: continuous poisson bound} follows from
    \begin{equation*}
        \norm{\nabla \phi_0}_\infty + \norm{\nabla^2 \phi_0}_\infty \Le C \norm{\nabla f}_\infty \int_0^\infty \min(1,t)^{-\frac12} e^{-\lambda t}\, \D t \Le C \norm{\nabla f}_\infty .
    \end{equation*}
    By \eqref{eq: backward equation}, $u(\cdot,T) - u(\cdot,s) = \int_s^T \mathcal L u(\cdot,t)\, \D t$ for $0 < s < T$, and the limits $s \to 0$ and $T \to \infty$ give the continuous Poisson equation \eqref{eq: continuous poisson}.
\end{proof}

\subsection{Proofs for the Riesz representors}
\label{sec: riesz well-posedness}

Lemma~\ref{lem: poincare} follows from \citep{bakry2008simple}. By \eqref{eq: dissipative bounds}, $U$ is bounded from below, and $\langle x, \nabla U(x) \rangle \Ge \frac m4 |x|^2 - C \Ge |x|$ outside a ball whose radius depends only on $m$, $M$, $R$ and $d$. Hence $U$ satisfies the drift condition of \citep[Corollary~1.6]{bakry2008simple}, and \citep[Theorem~1.4]{bakry2008simple} gives \eqref{eq: poincare} for smooth $f$. In the proof of \citep[Corollary~1.6]{bakry2008simple}, the Lyapunov function is an exponential of $|x|$ outside a larger ball, whose radius depends only on the radius of the first ball and on $d$. The constants of the inequality \citep[(1.3)]{bakry2008simple} for the Lyapunov function depend on $U$ only through $\nabla U$ on the larger ball. By \citep[Theorem~1.4]{bakry2008simple}, $C_P$ depends only on these constants and on the Poincar\'e constant of $\pi$ restricted to the larger ball, which \citep[Section~1]{bakry2008simple} bounds through the oscillation of $U$ on the ball. Since $|\nabla U(x)| \Le M(1 + |x|)$, $C_P$ depends only on $m$, $M$, $R$ and $d$. By a mollification, \eqref{eq: poincare} extends to every $f \in C^1(\mathbb R^d)$ with a bounded gradient.

The next lemma is the identity of Bochner under $\pi$, used in the proof of Theorem~\ref{thm: riesz}.

\begin{lemma}
\label{lem: bochner}
    Under Assumption~\ref{ass: U}, for every $\omega \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j \omega}_\infty < \infty$,
    \begin{equation}
        \pi\bigl( (\mathcal L \omega)^2 \bigr) = \pi\bigl( \nabla^2 \omega : \nabla^2 \omega \bigr) + \pi\bigl( \nabla \omega \cdot \nabla^2 U\, \nabla \omega \bigr) .
        \label{eq: bochner}
    \end{equation}
\end{lemma}

\begin{proof}
    Differentiating \eqref{eq: generator} gives $\partial_i \mathcal L \omega = \mathcal L \partial_i \omega - \sum_{j=1}^d \partial_i \partial_j U\, \partial_j \omega$ for each component $i$. The functions $\omega$, $\mathcal L \omega$ and $\partial_i \omega$ and their derivatives of orders $1$ and $2$ grow at most linearly, so \eqref{eq: generator ibp} with $f_1 = \mathcal L \omega$, $f_2 = \omega$, and then with $f_1 = f_2 = \partial_i \omega$, gives
    \begin{equation*}
    \begin{aligned}
        \pi\bigl( (\mathcal L \omega)^2 \bigr) & = -\sum_{i=1}^d \pi\bigl( \partial_i \omega\, \partial_i \mathcal L \omega \bigr) = -\sum_{i=1}^d \pi\bigl( \partial_i \omega\, \mathcal L \partial_i \omega \bigr) + \pi\bigl( \nabla \omega \cdot \nabla^2 U\, \nabla \omega \bigr) \\
        & = \sum_{i=1}^d \pi\bigl( |\nabla \partial_i \omega|^2 \bigr) + \pi\bigl( \nabla \omega \cdot \nabla^2 U\, \nabla \omega \bigr) ,
    \end{aligned}
    \end{equation*}
    which is \eqref{eq: bochner}.
\end{proof}

Let $L^2_0(\pi) := \bigl\{ \psi \in L^2(\pi) : \pi(\psi) = 0 \bigr\}$, a Hilbert space with the inner product of $L^2(\pi)$.

\begin{proof}[Proof of Theorem~\ref{thm: riesz}]
    Let $\omega \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j \omega}_\infty < \infty$. Since $\omega$ grows at most linearly, \eqref{eq: generator ibp}, the Cauchy--Schwarz inequality and \eqref{eq: poincare} give
    \begin{equation*}
        \pi\bigl( |\nabla \omega|^2 \bigr) = \pi\bigl( (\omega - \pi(\omega)) (-\mathcal L \omega) \bigr) \Le C_P^{\frac12}\, \pi\bigl( |\nabla \omega|^2 \bigr)^{\frac12}\, \pi\bigl( (\mathcal L \omega)^2 \bigr)^{\frac12} ,
    \end{equation*}
    so $\pi(|\nabla \omega|^2) \Le C_P\, \pi((\mathcal L \omega)^2)$, and \eqref{eq: bochner} with $\nabla^2 U \succeq -M I_d$ gives
    \begin{equation}
        \pi\bigl( \nabla^2 \omega : \nabla^2 \omega \bigr) \Le \pi\bigl( (\mathcal L \omega)^2 \bigr) + M\, \pi\bigl( |\nabla \omega|^2 \bigr) \Le (1 + M C_P)\, \pi\bigl( (\mathcal L \omega)^2 \bigr) .
        \label{eq: hessian l2}
    \end{equation}
    Since $\mathcal A(x) : \mathcal A(x) \Le d\, \norm{\mathcal A(x)}^2$, the Cauchy--Schwarz inequality and \eqref{eq: hessian l2} give
    \begin{equation}
        \bigl| \pi\bigl( \nabla^2 \omega : \mathcal A \bigr) \bigr| \Le \sqrt{d\, (1 + M C_P)}\; \norm{\mathcal A}_\infty\; \pi\bigl( (\mathcal L \omega)^2 \bigr)^{\frac12} .
        \label{eq: riesz functional bound}
    \end{equation}

    The functions $-\mathcal L \omega$ lie in $L^2_0(\pi)$ by their linear growth and \eqref{eq: generator ibp} with $f_1 = 1$. For $f \in C_c^\infty(\mathbb R^d)$, the continuous Poisson solution $\phi_0$ in \eqref{eq: continuous poisson solution} has bounded derivatives of orders $1$ to $4$ by \eqref{eq: regular decay} and satisfies \eqref{eq: continuous poisson}. Since the functions $f - \pi(f)$ are dense in $L^2_0(\pi)$, so are the functions $-\mathcal L \omega$. By \eqref{eq: riesz functional bound}, $-\mathcal L \omega \mapsto \pi(\nabla^2 \omega : \mathcal A)$ is a well-defined linear functional on the functions $-\mathcal L \omega$, bounded in the norm of $L^2_0(\pi)$. It extends to $L^2_0(\pi)$, and the Riesz representation theorem gives the unique $\psi_{\mathcal A} \in L^2_0(\pi)$ that satisfies \eqref{eq: riesz}. Since the functional is linear in $\mathcal A$, the map $\mathcal A \mapsto \psi_{\mathcal A}$ is linear, and \eqref{eq: riesz functional bound} gives \eqref{eq: riesz bound}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem: representor gradient}]
    Since $\mathcal A$ is bounded and Lipschitz, the vector function $\mathcal A\, \nabla U - \operatorname{div} \mathcal A$ grows at most linearly and belongs to $L^2(\pi)$ by Lemma~\ref{lem: dissipative}. Let $\mathcal H$ be the set of the functions $f \in L^2_0(\pi)$ that have a weak gradient $\nabla f \in L^2(\pi)$. By \eqref{eq: poincare}, $\mathcal H$ is a Hilbert space with the inner product $\pi(\nabla f_1 \cdot \nabla f_2)$. On $\mathcal H$, the linear functional $f \mapsto \pi\bigl( \nabla f \cdot ( \mathcal A\, \nabla U - \operatorname{div} \mathcal A ) \bigr)$ is bounded, and the Riesz representation theorem gives $\psi \in \mathcal H$ with
    \begin{equation}
        \pi\bigl( \nabla \psi \cdot \nabla f \bigr) = \pi\bigl( \nabla f \cdot ( \mathcal A\, \nabla U - \operatorname{div} \mathcal A ) \bigr) , \quad f \in \mathcal H , \qquad \pi\bigl( |\nabla \psi|^2 \bigr) \Le \pi\bigl( |\mathcal A\, \nabla U - \operatorname{div} \mathcal A|^2 \bigr) .
        \label{eq: weak representor}
    \end{equation}

    Let $\omega \in C^2(\mathbb R^d)$ with $\norm{\nabla \omega}_\infty + \norm{\nabla^2 \omega}_\infty < \infty$, so that $\omega - \pi(\omega) \in \mathcal H$ and $\mathcal L \omega \in L^2(\pi)$. Since $\nabla \pi = -\pi\, \nabla U$ and $\pi$ has the Gaussian tail of Lemma~\ref{lem: dissipative}, an integration by parts gives
    \begin{equation}
        \pi\bigl( \nabla^2 \omega : \mathcal A \bigr) = -\sum_{i,j=1}^{d} \int_{\mathbb R^d} \partial_j \omega\, \partial_i \bigl( \mathcal A_{ij}\, \pi \bigr)\, \D x = \pi\bigl( \nabla \omega \cdot ( \mathcal A\, \nabla U - \operatorname{div} \mathcal A ) \bigr) ,
        \label{eq: matrix ibp}
    \end{equation}
    and, for $f \in \mathcal H$,
    \begin{equation}
        \pi\bigl( \nabla f \cdot \nabla \omega \bigr) = \pi\bigl( f\, (-\mathcal L \omega) \bigr) .
        \label{eq: weak ibp}
    \end{equation}
    By \eqref{eq: weak ibp} with $f = \psi$, \eqref{eq: weak representor} with $f = \omega - \pi(\omega)$ and \eqref{eq: matrix ibp},
    \begin{equation*}
        \pi\bigl( \psi\, (-\mathcal L \omega) \bigr) = \pi\bigl( \nabla \psi \cdot \nabla \omega \bigr) = \pi\bigl( \nabla^2 \omega : \mathcal A \bigr) .
    \end{equation*}
    For $\omega \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j \omega}_\infty < \infty$, this is \eqref{eq: riesz}, so $\psi = \psi_{\mathcal A}$ by the uniqueness in Theorem~\ref{thm: riesz}, which proves \eqref{eq: representor gradient}.

    Finally, let $f$ be Lipschitz. By \eqref{eq: continuous poisson bound}, $\phi_0 \in C^2(\mathbb R^d)$ has bounded derivatives of orders $1$ and $2$, so \eqref{eq: representor gradient} with $\omega = \phi_0$, \eqref{eq: continuous poisson} and $\pi(\psi_{\mathcal A}) = 0$ give \eqref{eq: representor poisson} through $\pi(\nabla^2 \phi_0 : \mathcal A) = \pi\bigl( \psi_{\mathcal A}\, ( f - \pi(f) ) \bigr) = \pi(f \psi_{\mathcal A})$.
\end{proof}

\section{Proofs for Section~\ref{sec: modified}}
\label{app: modified}

Throughout this appendix, the constant $C$ depends only on $m$, $M$, $R$ and $d$.

\subsection{Proof of Lemma~\ref{lem: modified class}: modified potential estimates}
\label{sec: modified class proof}

\begin{proof}[Proof of Lemma~\ref{lem: modified class}]
    The proof bounds the derivatives of $V$ and of $V\theta_h$, which gives \eqref{eq: modified class}, and then chooses $\delta$ so that $\tilde U_h$ satisfies \eqref{eq: modified class bounds} and the convexity condition. By the Leibniz rule, $\nabla^i |\nabla U|^2$ is a sum of contractions of $\nabla^{l+1} U$ with $\nabla^{i-l+1} U$, $0 \Le l \Le i$. For $1 \Le i \Le 5$, each of them contains the factor $\nabla U$ at most once, with $|\nabla U(x)| \Le C(1+|x|)$, while $\norm{\nabla^i \Delta U(x)} \Le dM$ for $0 \Le i \Le 5$. Hence $V = |\nabla U|^2 - \Delta U$ of \eqref{eq: V} satisfies
    \begin{equation}
        |V(x)| \Le C \bigl( 1 + |x| \bigr)^2 , \qquad \norm{\nabla^i V(x)} \Le C \bigl( 1 + |x| \bigr) , \quad 1 \Le i \Le 5 , \qquad \forall x \in \mathbb R^d ,
        \label{eq: V growth}
    \end{equation}
    where $C$ depends only on $M$ and $d$.

    Let $\delta \in (0,1]$, fixed by \eqref{eq: delta bound} at the end of the proof, and let $h \Le \delta$ throughout. By \eqref{eq: modified potential}, $\tilde U_h - U = \frac h4\, V \theta_h$ with $\theta_h(x) = \theta(hx/\delta)$. The function $\theta_h$ is supported in $\{|x| \Le 2\delta/h\}$, where
    \begin{equation}
        \frac h\delta \bigl( 1 + |x| \bigr) \Le \frac h\delta + 2 \Le 3 ,
        \label{eq: cutoff support}
    \end{equation}
    and $\norm{\nabla^l \theta_h}_\infty \Le C (h/\delta)^l$ for $0 \Le l \Le 5$, with $C$ depending on the fixed function $\theta$ only. By the Leibniz rule, $\nabla^i (V\theta_h)$ is a sum of products of $\nabla^{i-l} V$ and $\nabla^l \theta_h$, $0 \Le l \Le i$. On the support of $\theta_h$ and for $1 \Le i \Le 5$, \eqref{eq: V growth} and $h \Le \delta$ bound the products with $l \Le i - 1$ by $C(1+|x|)$, and \eqref{eq: cutoff support} bounds the product with $l = i$ by $C (h/\delta)^i (1+|x|)^2 \Le C(1+|x|)$. Hence
    \begin{equation}
        \bigl\| \nabla^i (V\theta_h)(x) \bigr\| \Le C \bigl( 1 + |x| \bigr) , \qquad 1 \Le i \Le 5 , \qquad \forall x\in \mathbb R^d: |x| \Le \frac{2\delta}{h} .
        \label{eq: V cutoff growth}
    \end{equation}
    Since $\nabla^i (V\theta_h)$ vanishes for $|x| > 2\delta/h$, \eqref{eq: V cutoff growth} gives \eqref{eq: modified class}, and, by \eqref{eq: cutoff support},
    \begin{equation}
        \norm{\nabla^i \tilde U_h(x) - \nabla^i U(x)} = \frac h4\, \bigl\| \nabla^i (V\theta_h)(x) \bigr\| \Le C \delta , \qquad 1 \Le i \Le 5 , \qquad \forall x \in \mathbb R^d ,
        \label{eq: modified derivative difference}
    \end{equation}
    with $C$ independent of $\delta$ and $h$. Choose $\delta \in (0,1]$ so small that this constant satisfies
    \begin{equation}
    C\delta \Le \min\Bigl\{ \frac m2 ,\, M \Bigr\} ,
    \label{eq: delta bound}
    \end{equation}
    and let $h_0 \Le \delta$. Then \eqref{eq: modified derivative difference} and Assumption~\ref{ass: U} give \eqref{eq: modified class bounds} and the convexity condition of Lemma~\ref{lem: modified class}. Finally, $\tilde U_h \in C^5(\mathbb R^d)$, since $\theta_h \in C^\infty(\mathbb R^d)$ and $U \in C^7(\mathbb R^d)$ gives $V \in C^5(\mathbb R^d)$.
\end{proof}

\subsection{Proof of Lemma~\ref{lem: shifted bias}: modified bias estimate}
\label{sec: shifted bias proof}

\begin{proof}[Proof of Lemma~\ref{lem: shifted bias}]
    By the symmetry of $\mathcal N(0, \frac h2 I_d)$,
    \begin{equation*}
        \tilde\pi_h(\tilde f_h) = \int_{\mathbb R^d} f(x)\, \biggl( \tilde\pi_h * \mathcal N\Bigl( 0, \frac h2 I_d \Bigr) \biggr)(x) \,\D x .
    \end{equation*}
    Since $\tilde\pi_h * \mathcal N(0, \frac h2 I_d)$ and $\pi$ are probability densities, \eqref{eq: shifted bias} follows from the bound
    \begin{equation}
        \int_{\mathbb R^d} \bigl( 1 + |x| \bigr)\, \Biggl| \biggl( \tilde\pi_h * \mathcal N\Bigl( 0, \frac h2 I_d \Bigr) \biggr)(x) - \pi(x) \Biggr| \,\D x \Le C h^2 ,
        \label{eq: density comparison}
    \end{equation}
    which the rest of the proof establishes.

    Using $\tilde\pi_h \propto e^{-\tilde U_h} = e^{-U} e^{-(\tilde U_h - U)}$, where $\tilde U_h - U = \frac h4\, V\theta_h$ by \eqref{eq: modified potential}, we have
    \begin{equation}
        \tilde\pi_h(x) = \frac{\pi(x)\, e^{-(\tilde U_h - U)(x)}}{\pi\bigl( e^{-(\tilde U_h - U)} \bigr)} .
        \label{eq: modified target ratio}
    \end{equation}
    By \eqref{eq: V growth}, $|\tilde U_h - U| \Le C h (1 + |x|^2)$, while $\pi(x) \Le C e^{-\frac{m}{8}|x|^2}$ by Lemma~\ref{lem: dissipative}. Shrinking $h_0$ so that $C h_0 \Le \frac{m}{16}$, with $C$ the constant of the bound on $|\tilde U_h - U|$, we have for $h \Le h_0$,
    \begin{equation}
        e^{|\tilde U_h - U|} \Le C e^{\frac{m}{16}|x|^2} , \qquad \pi\bigl( (1+|x|)^5\, e^{|\tilde U_h - U|} \bigr) \Le C .
        \label{eq: weighted moments}
    \end{equation}
    Let $\chi \sim \mathcal N(0,I_d)$, so that
    \begin{equation*}
        \biggl( \tilde\pi_h * \mathcal N\Bigl( 0, \frac h2 I_d \Bigr) \biggr)(x) = \mathbb E\biggl[ \tilde\pi_h\Bigl( x - \sqrt{\frac h2}\, \chi \Bigr) \biggr] .
    \end{equation*}
    Since $\tilde U_h \in C^5$ by Lemma~\ref{lem: modified class}, $\tilde\pi_h \in C^5$. Expanding $\tilde\pi_h$ about $x$ to the order $4$ along $-\sqrt{h/2}\, \chi$,
    \begin{equation*}
        \tilde\pi_h\Bigl( x - \sqrt{\frac h2}\, \chi \Bigr) = \tilde\pi_h(x) - \sqrt{\frac h2}\, \nabla \tilde\pi_h(x) \cdot \chi + \frac h4\, \nabla^2 \tilde\pi_h(x)[\chi,\chi] - \frac16 \Bigl( \frac h2 \Bigr)^{\frac32} \nabla^3\tilde\pi_h(x)\bigl[ \chi^{\otimes3} \bigr] + \varrho_4^{\chi}(x) ,
    \end{equation*}
    where the remainder satisfies
    \begin{equation}
        |\varrho_4^{\chi}(x)| \Le \frac{h^2}{24}\, |\chi|^4 \int_0^1 (1-s)^3\, \biggl\| \nabla^4 \tilde\pi_h\Bigl( x - s\sqrt{\frac h2}\, \chi \Bigr) \biggr\| \,\D s .
        \label{eq: fourth remainder}
    \end{equation}
    Under the expectation the terms of odd order in $\chi$ vanish and the term of order $2$ has mean $\frac h4\, \Delta \tilde\pi_h(x)$, hence
    \begin{equation}
        \biggl( \tilde\pi_h * \mathcal N\Bigl( 0, \frac h2 I_d \Bigr) \biggr)(x) = \tilde\pi_h(x) + \frac h4\, \Delta \tilde\pi_h(x) + \mathbb E\bigl[ \varrho_4^{\chi}(x) \bigr] .
        \label{eq: shift of density}
    \end{equation}
    Differentiating $\tilde\pi_h \propto e^{-\tilde U_h}$ to the order $2$,
    \begin{equation}
        \Delta \tilde\pi_h = \tilde\pi_h\, \bigl( |\nabla \tilde U_h|^2 - \Delta \tilde U_h \bigr) .
        \label{eq: laplacian of density}
    \end{equation}
    Multiplying \eqref{eq: shift of density} by the normalizing constant $\pi\bigl( e^{-(\tilde U_h - U)} \bigr)$ of \eqref{eq: modified target ratio} and inserting \eqref{eq: modified target ratio} and \eqref{eq: laplacian of density} give
    \begin{equation}
        \pi\bigl( e^{-(\tilde U_h - U)} \bigr) \biggl( \tilde\pi_h * \mathcal N\Bigl( 0, \frac h2 I_d \Bigr) \biggr)(x) - \pi(x) = \mathcal T_1 + \mathcal T_2 ,
        \label{eq: density split}
    \end{equation}
    where the two terms $\mathcal T_1$ and $\mathcal T_2$ are 
    \begin{equation*}
        \mathcal T_1 := \pi(x) \biggl( e^{-(\tilde U_h - U)(x)} \Bigl( 1 + \frac h4 \bigl( |\nabla \tilde U_h(x)|^2 - \Delta \tilde U_h(x) \bigr) \Bigr) - 1 \biggr) , \qquad
        \mathcal T_2 := \pi\bigl( e^{-(\tilde U_h - U)} \bigr)\, \mathbb E\bigl[ \varrho_4^{\chi}(x) \bigr] .
    \end{equation*}
	\textbf{Bound of $\mathcal T_1$}. Note that, by \eqref{eq: modified class bounds}, \eqref{eq: modified class} and \eqref{eq: V}, $\bigl| |\nabla \tilde U_h|^2 - \Delta \tilde U_h \bigr| \Le C (1 + |x|)^2$ and
    \begin{equation}
        \bigl| |\nabla \tilde U_h|^2 - \Delta \tilde U_h - V \bigr| \Le |\nabla \tilde U_h - \nabla U|\, |\nabla \tilde U_h + \nabla U| + |\Delta \tilde U_h - \Delta U| \Le C h \bigl( 1 + |x| \bigr)^2 .
        \label{eq: modified V gap}
    \end{equation}
    Taylor's formula for the exponential, with $\tilde U_h - U = \frac h4\, V - \frac h4\, V (1 - \theta_h)$, the bound $|\tilde U_h - U| \Le C h (1 + |x|^2)$, the bound on $|\nabla \tilde U_h|^2 - \Delta \tilde U_h$ and \eqref{eq: modified V gap}, gives
    \begin{equation}
        \biggl| e^{-(\tilde U_h - U)} \Bigl( 1 + \frac h4 \bigl( |\nabla \tilde U_h|^2 - \Delta \tilde U_h \bigr) \Bigr) - 1 \biggr|
        \Le C h^2 \bigl( 1 + |x| \bigr)^4 e^{|\tilde U_h - U|} + \frac h4\, |V| \bigl( 1 - \theta_h \bigr) .
        \label{eq: factor bound}
    \end{equation}
    Since $1 - \theta_h$ vanishes on $\{|x| \Le \delta/h\}$ and $\pi(x) \Le C e^{-\frac{m}{8}|x|^2} \Le C e^{-\frac{m \delta^2}{16 h^2}}\, e^{-\frac{m}{16}|x|^2}$ for $|x| \Ge \delta/h$ by Lemma~\ref{lem: dissipative}, \eqref{eq: V growth} gives
    \begin{equation}
        \pi\bigl( (1+|x|)\, |V|\, (1 - \theta_h) \bigr) \Le C\, \pi\bigl( (1+|x|)^3\, \mathbf 1_{\{|x| \Ge \delta/h\}} \bigr) \Le C e^{-\frac{m \delta^2}{16 h^2}} \Le C h^2 .
        \label{eq: cutoff tail}
    \end{equation}
    Multiplying \eqref{eq: factor bound} by $(1+|x|)\, \pi(x)$ and integrating over $\mathbb R^d$, \eqref{eq: weighted moments} and \eqref{eq: cutoff tail} give
    \begin{equation}
        \int_{\mathbb R^d} \bigl( 1 + |x| \bigr)\, |\mathcal T_1| \,\D x \Le C h^2\, \pi\bigl( (1+|x|)^5\, e^{|\tilde U_h - U|} \bigr) + \frac h4\, \pi\bigl( (1+|x|)\, |V|\, (1 - \theta_h) \bigr) \Le C h^2 .
        \label{eq: factor integral}
    \end{equation}
    \textbf{Bound of $\mathcal T_2$}. Differentiating $\tilde\pi_h \propto e^{-\tilde U_h}$ to the order $4$ and using \eqref{eq: modified class bounds} give
    \begin{equation*}
        \norm{\nabla^4 \tilde\pi_h(y)} \Le C \bigl( 1 + |y| \bigr)^4 \tilde\pi_h(y) , \qquad y \in \mathbb R^d .
    \end{equation*}
    We insert this bound into \eqref{eq: fourth remainder} and substitute $y = x - s\sqrt{h/2}\,\chi$, so that $1 + |x| \Le (1 + |\chi|)(1 + |y|)$. Then
    \begin{equation*}
        \int_{\mathbb R^d} \bigl( 1 + |x| \bigr)\, \mathbb E|\varrho_4^{\chi}(x)| \,\D x
        \Le C h^2\, \mathbb E\bigl[ |\chi|^4 (1+|\chi|) \bigr]\, \tilde\pi_h\bigl( (1+|x|)^5 \bigr) \Le C h^2\, \tilde\pi_h\bigl( (1+|x|)^5 \bigr) .
    \end{equation*}
    Since $\pi\bigl( e^{-(\tilde U_h - U)} \bigr)\, \tilde\pi_h\bigl( (1+|x|)^5 \bigr) = \pi\bigl( (1+|x|)^5\, e^{-(\tilde U_h - U)} \bigr) \Le C$ by \eqref{eq: modified target ratio} and \eqref{eq: weighted moments},
    \begin{equation}
        \int_{\mathbb R^d} \bigl( 1 + |x| \bigr)\, |\mathcal T_2| \,\D x \Le C h^2 .
        \label{eq: remainder integral}
    \end{equation}

    By \eqref{eq: density split}, \eqref{eq: factor integral} and \eqref{eq: remainder integral}, we obtain
    \begin{equation}
        \int_{\mathbb R^d} \bigl( 1 + |x| \bigr)\, \Biggl| \pi\bigl( e^{-(\tilde U_h - U)} \bigr) \biggl( \tilde\pi_h * \mathcal N\Bigl( 0, \frac h2 I_d \Bigr) \biggr)(x) - \pi(x) \Biggr| \,\D x \Le C h^2 .
        \label{eq: unnormalized comparison}
    \end{equation}
    Since $\tilde\pi_h * \mathcal N(0, \frac h2 I_d)$ and $\pi$ are probability densities, \eqref{eq: unnormalized comparison} gives $\bigl| \pi\bigl( e^{-(\tilde U_h - U)} \bigr) - 1 \bigr| \Le C h^2$, so that $\pi\bigl( e^{-(\tilde U_h - U)} \bigr) \Ge \frac12$ after shrinking $h_0$. Dividing by $\pi\bigl( e^{-(\tilde U_h - U)} \bigr)$ in \eqref{eq: unnormalized comparison} then gives \eqref{eq: density comparison}.
\end{proof}

\subsection{Proof of Corollary~\ref{cor: modified poisson}: modified discrete Poisson solution}
\label{sec: modified poisson proof}

\begin{proof}[Proof of Corollary~\ref{cor: modified poisson}]
    By \eqref{eq: heat derivatives}, $\tilde f_h$ has bounded derivatives of orders $1$ to $4$, with $\norm{\nabla \tilde f_h}_\infty \Le \norm{\nabla f}_\infty$. The modified target $\tilde\pi_h$ is invariant under $e^{t\tilde{\mathcal L}_h}$. By \eqref{eq: modified target ratio}, \eqref{eq: weighted moments} and $\pi\bigl( e^{-(\tilde U_h - U)} \bigr) \Ge \frac12$ from the proof of Lemma~\ref{lem: shifted bias},
    \begin{equation}
        \tilde\pi_h\bigl( 1 + |x| \bigr) \Le C .
        \label{eq: modified target moment}
    \end{equation}
    As in the proof of \eqref{eq: relaxation}, the bounds \eqref{eq: modified target moment} and \eqref{eq: smoothing} for the modified dynamics at $j = 1$ give
    \begin{equation}
        \bigl| \bigl( e^{ih\tilde{\mathcal L}_h} \tilde f_h \bigr)(x) - \tilde\pi_h(\tilde f_h) \bigr| \Le C \norm{\nabla f}_\infty (1 + |x|)\, e^{-\lambda i h} , \qquad i \Ge 0 .
        \label{eq: modified relaxation}
    \end{equation}
    Hence the series $\tilde\phi_h = h \sum_{i=0}^{\infty} \bigl( e^{ih\tilde{\mathcal L}_h} \tilde f_h - \tilde\pi_h(\tilde f_h) \bigr)$ converges, satisfies \eqref{eq: modified poisson} by telescoping, and is differentiable term by term. By \eqref{eq: heat derivatives} for the term $i = 0$,
    \begin{equation*}
        \norm{\nabla^j \tilde\phi_h}_\infty \Le C \norm{\nabla f}_\infty h^{\frac{3-j}{2}} + h \sum_{i=1}^{\infty} \norm{\nabla^j ( e^{ih\tilde{\mathcal L}_h} \tilde f_h )}_\infty , \qquad j = 1, 2, 3, 4 ,
    \end{equation*}
    and the bound \eqref{eq: summed smoothing} with $\tilde f_h$ in place of $f$ gives \eqref{eq: modified poisson bound}. Finally, $\tilde\phi_h \in C^4(\mathbb R^d)$, since its terms have bounded derivatives of orders $1$ to $4$ by \eqref{eq: short time} for the modified dynamics.
\end{proof}

\subsection{Proofs of Lemmas~\ref{lem: moments} and \ref{lem: sizes}}
\label{sec: moments proof}

Lemma~\ref{lem: langevin moments} bounds one step of the Langevin dynamics \eqref{eq: langevin} and of the modified dynamics \eqref{eq: modified langevin}, and expands the moments of the step. The proofs of Lemmas~\ref{lem: moments} and \ref{lem: sizes} use the bounds \eqref{eq: langevin sizes} for the modified dynamics, and the expansions of the moments are used in Appendix~\ref{sec: matching proof}.

\begin{lemma}
\label{lem: langevin moments}
    Under Assumption~\ref{ass: U}, for $0 \Le s \Le h$, with $C$ depending only on $M$ and $d$ and $\pnorm{\cdot}_{12}$ taken under $\mathbb E_0$,
    \begin{equation}
        \pnorm{Y_0(s) - Y_0}_{12} \Le C \sqrt h \bigl( 1 + |Y_0| \bigr) , \qquad
        \pnorm{Y_0(h) - Y_0 - \sqrt{2h}\, \chi_0}_{12} \Le C h \bigl( 1 + |Y_0| \bigr) ,
        \label{eq: langevin sizes}
    \end{equation}
    and, with all derivatives evaluated at $Y_0$ and $1 \Le i,j,k \Le d$,
    \begin{align}
        \Bigl| \mathbb E_0\bigl[ Y_0(h) - Y_0 \bigr] + h\, \nabla U - \frac{h^2}{2} \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) \Bigr| &\Le C h^3 \bigl( 1+|Y_0| \bigr)^3 , \label{eq: langevin m1}\\
        \Bigl\| \mathbb E_0\bigl[ (Y_0(h) - Y_0)^{\otimes 2} \bigr] - 2h\, I_d - h^2 \bigl( \nabla U^{\otimes 2} - 2\, \nabla^2 U \bigr) \Bigr\| &\Le C h^3 \bigl( 1+|Y_0| \bigr)^4 , \label{eq: langevin m2}\\
        \Bigl| \bigl( \mathbb E_0\bigl[ (Y_0(h) - Y_0)^{\otimes 3} \bigr] \bigr)_{ijk} + 2h^2 \bigl( \delta_{ij}\, \partial_k U + \delta_{ik}\, \partial_j U + \delta_{jk}\, \partial_i U \bigr) \Bigr| &\Le C h^3 \bigl( 1+|Y_0| \bigr)^{6} . \label{eq: langevin m3}
    \end{align}
    The bounds \eqref{eq: langevin sizes}--\eqref{eq: langevin m3} hold for the modified dynamics \eqref{eq: modified langevin}, with $\tilde U_h$ and $\tilde Y_0(s)$ in place of $U$ and $Y_0(s)$.
\end{lemma}

\begin{proof}
	The displacement $W_s$ of \eqref{eq: displacements} satisfies
    \begin{equation}
        W_s = -\int_0^s \nabla U\bigl( Y_0 + W_r \bigr) \D r + \sqrt2\, B_s , \qquad 0 \Le s \Le h ,
        \label{eq: displacement equation}
    \end{equation}
    with $\sqrt2\, B_h = \sqrt{2h}\, \chi_0$. The linear growth of $\nabla U$ and Gr\"onwall's inequality give
    \begin{equation*}
        \sup_{s \Le h} |W_s| \Le C \Bigl( h \bigl( 1 + |Y_0| \bigr) + \sup_{r \Le h} |B_r| \Bigr) , \qquad
        \bigl| W_h - \sqrt{2h}\, \chi_0 \bigr| \Le Mh \Bigl( 1 + |Y_0| + \sup_{r \Le h} |W_r| \Bigr) .
    \end{equation*}
    Since $\mathbb E \bigl[ \sup_{r \Le h} |B_r|^{12} \bigr] \Le C h^{6}$ by the Burkholder--Davis--Gundy inequality, these two bounds give \eqref{eq: langevin sizes}.

    Next we prove \eqref{eq: langevin m1}--\eqref{eq: langevin m3}. By \eqref{eq: displacement equation}, $W_s$ is a diffusion started at $W_0 = 0$ with the generator
    \begin{equation}
        (\mathcal L_{Y_0} \psi)(w) := -\nabla U(Y_0 + w) \cdot \nabla \psi(w) + \Delta \psi(w) ,
        \label{eq: displacement generator}
    \end{equation}
    and the entries of $\mathbb E_0\bigl[ W_h^{\otimes n} \bigr]$, $n = 1, 2, 3$, are $\mathbb E_0\bigl[ \psi(W_h) \bigr]$ for the monomials $\psi(w) = w_i$, $w_i w_j$, $w_i w_j w_k$. Since $\psi(0) = 0$, It\^o's formula applied to $\psi$ gives the Taylor expansion to the order $2$ with integral remainder,
    \begin{equation}
        \mathbb E_0\bigl[ \psi(W_h) \bigr] = h\, (\mathcal L_{Y_0}\psi)(0) + \frac{h^2}{2}\, (\mathcal L_{Y_0}^2\psi)(0)
        + \frac12 \int_0^h (h-s)^2\, \mathbb E_0\bigl[ (\mathcal L_{Y_0}^3 \psi)(W_s) \bigr] \D s .
        \label{eq: ito taylor}
    \end{equation}
    By \eqref{eq: displacement generator}, with the derivatives of $U$ evaluated at $Y_0 + w$,
    \begin{equation*}
    \begin{aligned}
        \mathcal L_{Y_0} w_i &= -\partial_i U(Y_0 + w) , \\
        \mathcal L_{Y_0}(w_i w_j) &= 2\delta_{ij} - \partial_i U\, w_j - \partial_j U\, w_i , \\
        \mathcal L_{Y_0}\bigl( w_i w_j w_k \bigr) &= 2 \bigl( \delta_{ij}\, w_k + \delta_{ik}\, w_j + \delta_{jk}\, w_i \bigr) - \bigl( \partial_i U\, w_j w_k + \partial_j U\, w_i w_k + \partial_k U\, w_i w_j \bigr) .
    \end{aligned}
    \end{equation*}
    Applying $\mathcal L_{Y_0}$ once more and evaluating at $w = 0$, where the derivatives of $U$ are evaluated at $Y_0$, give
    \begin{equation*}
    \begin{aligned}
        \mathcal L_{Y_0} w_i &= -\partial_i U , & \qquad
        \mathcal L_{Y_0}^2 w_i &= \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr)_i , \\
        \mathcal L_{Y_0}(w_i w_j) &= 2\delta_{ij} , & \qquad
        \mathcal L_{Y_0}^2(w_i w_j) &= 2\, \partial_i U\, \partial_j U - 4\, \partial_i \partial_j U , \\
        \mathcal L_{Y_0}\bigl( w_i w_j w_k \bigr) &= 0 , & \qquad
        \mathcal L_{Y_0}^2\bigl( w_i w_j w_k \bigr) &= -4 \bigl( \delta_{ij}\, \partial_k U + \delta_{ik}\, \partial_j U + \delta_{jk}\, \partial_i U \bigr) .
    \end{aligned}
    \end{equation*}
    Inserting these values into \eqref{eq: ito taylor} gives the explicit terms of \eqref{eq: langevin m1}--\eqref{eq: langevin m3}.

    It remains to bound the integral term of \eqref{eq: ito taylor}. The functions $\mathcal L_{Y_0} \psi$, $\mathcal L_{Y_0}^2 \psi$ and $\mathcal L_{Y_0}^3 \psi$ are sums of products of components of $w$, components of $\nabla U(Y_0 + w)$ and derivatives of $U$ of orders $2$ to $5$, which are bounded by Assumption~\ref{ass: U}. An application of $\mathcal L_{Y_0}$ raises the number of factors $w$ and $\nabla U$ of a product by at most $1$, and only when the term $-\nabla U(Y_0 + w) \cdot \nabla$ of \eqref{eq: displacement generator} differentiates a derivative of $U$ of order at least $2$, which does not occur in $\psi$ or $\mathcal L_{Y_0} \psi$. Hence each product in $\mathcal L_{Y_0}^3 \psi$ has at most one such factor more than $\psi$, and the linear growth of $\nabla U$ gives
    \begin{equation*}
    \begin{aligned}
        \bigl| (\mathcal L_{Y_0}^3 w_i)(w) \bigr| &\Le C \bigl( 1 + |Y_0| + |w| \bigr)^{2} , \\
        \bigl| \bigl( \mathcal L_{Y_0}^3 (w_i w_j) \bigr)(w) \bigr| &\Le C \bigl( 1 + |Y_0| + |w| \bigr)^{3} , \\
        \bigl| \bigl( \mathcal L_{Y_0}^3 (w_i w_j w_k) \bigr)(w) \bigr| &\Le C \bigl( 1 + |Y_0| + |w| \bigr)^{4} .
    \end{aligned}
    \end{equation*}
    By \eqref{eq: langevin sizes}, the integral term of \eqref{eq: ito taylor} is therefore bounded by $C h^3 (1+|Y_0|)^{2}$, $C h^3 (1+|Y_0|)^{3}$ and $C h^3 (1+|Y_0|)^{4}$ for the three monomials, which are at most the right sides of \eqref{eq: langevin m1}--\eqref{eq: langevin m3}.

    Finally, the argument uses only that $U \in C^5(\mathbb R^d)$ satisfies the bounds of Assumption~\ref{ass: U} up to order $5$. By Lemma~\ref{lem: modified class}, the modified potential $\tilde U_h$ satisfies them with $2M$ in place of $M$, so the argument applies to the modified dynamics \eqref{eq: modified langevin}, with $\tilde U_h$, $\tilde Y_0(s)$ and $\tilde W_s$ in place of $U$, $Y_0(s)$ and $W_s$.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem: moments}]
    By \eqref{eq: sglm markov}, the $k$-th step of the chain is
    \begin{equation*}
        Y_{k+1} - Y_k = \sqrt{2h}\, \chi_k - h\, \hat g(Z_k; \xi_k) , \qquad Z_k = Y_k + \sqrt{\frac h2}\, \chi_k .
    \end{equation*}
    We first bound the displacement over one step. By the moment bound of Assumption~\ref{ass: sg},
    \begin{equation*}
        \mathbb E_k |\hat g(Z_k; \xi_k)|^{12} \Le M^{12}\, \mathbb E_k \bigl( 1 + |Z_k| \bigr)^{12} \Le C \bigl( 1 + |Y_k| \bigr)^{12} ,
    \end{equation*}
    and with the moments of $\chi_k$,
    \begin{equation}
        \mathbb E_k |Y_{k+1} - Y_k|^{2} \Le C h + C h^{2} \bigl( 1 + |Y_k| \bigr)^{2} , \qquad
        \mathbb E_k |Y_{k+1} - Y_k|^{12} \Le C h^{6} \bigl( 1 + |Y_k| \bigr)^{12} .
        \label{eq: step displacement}
    \end{equation}
    Since $\tilde Y_k(h) - Y_k$ is driven by $(B_{kh+t} - B_{kh})_{0 \Le t \Le h}$, which is independent of $\mathcal F_k$, the bound on $\tilde Y_0(s) - Y_0$ in \eqref{eq: langevin sizes} for the modified dynamics, at the index $k$, gives
    \begin{equation}
        \mathbb E_k |\tilde Y_k(h) - Y_k|^{12} \Le C h^{6} \bigl( 1 + |Y_k| \bigr)^{12} .
        \label{eq: exact displacement}
    \end{equation}

    Next we derive the drift condition. The splitting $\langle Y_k, \nabla U(Z_k)\rangle = \langle Z_k, \nabla U(Z_k)\rangle - \sqrt{h/2}\, \langle \chi_k, \nabla U(Z_k)\rangle$, the drift bound in \eqref{eq: dissipative bounds} at $Z_k$ and the bound on $|\nabla U(Z_k)|$ give, after taking $\mathbb E_k$ with $\mathbb E_k |Z_k|^2 \Ge |Y_k|^2$,
    \begin{equation}
        \langle Y_k, \mathbb E_k [\nabla U(Z_k)] \rangle \Ge \frac m4 |Y_k|^2 - C - C \sqrt h \bigl( 1 + |Y_k| \bigr) \Ge \frac{m}{8} |Y_k|^2 - C .
        \label{eq: drift condition}
    \end{equation}

    Next we combine \eqref{eq: step displacement} and \eqref{eq: drift condition} into the Lyapunov estimate. Since the estimator is unbiased, $\xi_k$ is independent of $\mathcal F_k$ and $\chi_k$, and $\chi_k$ is centered and independent of $\mathcal F_k$, $\mathbb E_k[Y_{k+1} - Y_k] = -h\, \mathbb E_k[\nabla U(Z_k)]$. For the order $2$, \eqref{eq: drift condition} and \eqref{eq: step displacement} give
    \begin{equation*}
        \mathbb E_k |Y_{k+1}|^2 = |Y_k|^2 - 2h \langle Y_k, \mathbb E_k [\nabla U(Z_k)] \rangle + \mathbb E_k |Y_{k+1} - Y_k|^2 \Le \Bigl( 1 - \frac m4 h \Bigr) |Y_k|^2 + C h + C h^2 \bigl( 1 + |Y_k| \bigr)^2 .
    \end{equation*}
    For the order $12$, Taylor's formula for $y \mapsto |y|^{12}$ gives
    \begin{equation*}
        |Y_{k+1}|^{12} \Le |Y_k|^{12} + 12\, |Y_k|^{10} \langle Y_k, Y_{k+1} - Y_k \rangle + C \bigl( |Y_k|^{10} |Y_{k+1} - Y_k|^2 + |Y_{k+1} - Y_k|^{12} \bigr) ,
    \end{equation*}
    so that \eqref{eq: drift condition} and \eqref{eq: step displacement} give
    \begin{equation*}
        \mathbb E_k |Y_{k+1}|^{12} \Le \Bigl( 1 - \frac{3m}{2} h \Bigr) |Y_k|^{12} + C h\, |Y_k|^{10} + C h^2 \bigl( 1 + |Y_k| \bigr)^{12} .
    \end{equation*}
    For $h \Le h_0$ with $h_0$ small enough, Young's inequality turns the bounds on $\mathbb E_k |Y_{k+1}|^2$ and $\mathbb E_k |Y_{k+1}|^{12}$ into
    \begin{equation}
        \mathbb E_k |Y_{k+1}|^{2} \Le \Bigl( 1 - \frac m8 h \Bigr) |Y_k|^{2} + C h , \qquad
        \mathbb E_k |Y_{k+1}|^{12} \Le \bigl( 1 - m h \bigr) |Y_k|^{12} + C h .
        \label{eq: lyapunov}
    \end{equation}
    Iterating the bound on $\mathbb E_k |Y_{k+1}|^{12}$ in \eqref{eq: lyapunov} over $k$ gives $\mathbb E |Y_k|^{12} \Le |Y_0|^{12} + C$, the bound on $\mathbb E|Y_k|^{12}$ in \eqref{eq: moment bounds}. Averaging \eqref{eq: step displacement} and \eqref{eq: exact displacement} over the law of $Y_k$ then gives the displacement bounds in \eqref{eq: moment bounds}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem: sizes}]
    For the full gradient step $D_0 = \sqrt{2h}\, \chi_0 - h\, \nabla U(Z_0)$ of \eqref{eq: full gradient step}, the moments of $\chi_0$ give
    \begin{equation*}
        \pnorm{\sqrt{2h}\, \chi_0}_4 \Le C\sqrt h , \qquad \pnorm{h\, \nabla U(Z_0)}_4 \Le M h \Bigl( 1 + |Y_0| + \sqrt{\frac h2}\, \pnorm{\chi_0}_4 \Bigr) \Le C h (1+|Y_0|) ,
    \end{equation*}
    and their sum gives the bound on $\pnorm{D_0}_4$ in \eqref{eq: sizes}. The bound on $\pnorm{\tilde Y_0(s) - Y_0}_4$ follows from \eqref{eq: langevin sizes} for the modified dynamics.

    Next we bound the discretization part $Y_0 + D_0 - \tilde Y_0(h) = D_0 - \tilde W_h$ of the local error. By \eqref{eq: modified langevin}, the displacement $\tilde W_h$ of \eqref{eq: displacements} is $\tilde W_h = \sqrt{2h}\, \chi_0 - \int_0^h \nabla \tilde U_h(Y_0 + \tilde W_r)\,\D r$, and subtracting it from $D_0$ gives
    \begin{equation}
        Y_0 + D_0 - \tilde Y_0(h) = \underbrace{\int_0^h \Bigl( \nabla U\bigl( Y_0 + \tilde W_r \bigr) - \nabla U( Z_0 ) \Bigr) \D r}_{=: \mathcal T_1} + \underbrace{\int_0^h \Bigl( \nabla \tilde U_h\bigl( Y_0 + \tilde W_r \bigr) - \nabla U\bigl( Y_0 + \tilde W_r \bigr) \Bigr) \D r}_{=: \mathcal T_2} .
        \label{eq: local error explicit}
    \end{equation}
    By $\norm{\nabla^2 U} \Le M$ with $Z_0 = Y_0 + \sqrt{h/2}\, \chi_0$ and by \eqref{eq: modified class}, the integrands of $\mathcal T_1$ and $\mathcal T_2$ satisfy
    \begin{equation*}
        \bigl| \nabla U( Y_0 + \tilde W_r ) - \nabla U( Z_0 ) \bigr| \Le M \Bigl| \tilde W_r - \sqrt{\frac h2}\, \chi_0 \Bigr| , \qquad
        \bigl| ( \nabla \tilde U_h - \nabla U )( Y_0 + \tilde W_r ) \bigr| \Le C h \bigl( 1 + |Y_0| + |\tilde W_r| \bigr) .
    \end{equation*}
    Minkowski's inequality and the bound $\pnorm{\tilde W_r}_4 \Le C \sqrt h\, (1 + |Y_0|)$ from \eqref{eq: langevin sizes} for the modified dynamics give
    \begin{equation*}
    \begin{aligned}
        \pnorm{\mathcal T_1}_4 & \Le M \int_0^h \Bigl( \pnorm{\tilde W_r}_4 + \sqrt{\frac h2}\, \pnorm{\chi_0}_4 \Bigr) \D r \Le C h^{\frac32} (1+|Y_0|) , \\
        \pnorm{\mathcal T_2}_4 & \Le C h \int_0^h \bigl( 1 + |Y_0| + \pnorm{\tilde W_r}_4 \bigr) \D r \Le C h^2 (1+|Y_0|) .
    \end{aligned}
    \end{equation*}
    Adding the bounds on $\pnorm{\mathcal T_1}_4$ and $\pnorm{\mathcal T_2}_4$ gives the bound on $\pnorm{Y_0 + D_0 - \tilde Y_0(h)}_4$ in \eqref{eq: sizes}.

    Next we prove the expansion \eqref{eq: local error expansion}. Taylor's formula for $\nabla U$ at $Y_0$, with $\norm{\nabla^3 U} \Le M$, gives
    \begin{equation*}
        \nabla U(Y_0 + w) = \nabla U(Y_0) + \nabla^2 U(Y_0)\, w + \varrho_2(w) , \qquad |\varrho_2(w)| \Le \frac M2 |w|^2 .
    \end{equation*}
    Since $\sqrt{h/2}\, \chi_0 = \frac{1}{\sqrt2} B_h$ and $\tilde W_r = \sqrt2\, B_r - \int_0^r \nabla \tilde U_h(Y_0 + \tilde W_q)\, \D q$, the integrand of $\mathcal T_1$ is
    \begin{equation*}
        \nabla U( Y_0 + \tilde W_r ) - \nabla U( Z_0 ) = \sqrt2\, \nabla^2 U(Y_0) \Bigl( B_r - \frac12 B_h \Bigr) - \nabla^2 U(Y_0) \int_0^r \nabla \tilde U_h(Y_0 + \tilde W_q)\, \D q + \varrho_2(\tilde W_r) - \varrho_2\Bigl( \sqrt{\frac h2}\, \chi_0 \Bigr) .
    \end{equation*}
    By the linear growth of $\nabla \tilde U_h$ from \eqref{eq: modified class bounds} and by \eqref{eq: langevin sizes} for the modified dynamics, each of the last three terms of this identity has $\pnorm{\cdot}_4$ at most $C h (1+|Y_0|)^2$, uniformly in $r \in [0,h]$. Hence
    \begin{equation}
        \biggl\| \nabla U( Y_0 + \tilde W_r ) - \nabla U( Z_0 ) - \sqrt2\, \nabla^2 U(Y_0) \Bigl( B_r - \frac12 B_h \Bigr) \biggr\|_4 \Le C h \bigl( 1+|Y_0| \bigr)^2 , \qquad r \in [0,h] .
        \label{eq: integrand expansion}
    \end{equation}
    Integrating \eqref{eq: integrand expansion} over $r \in [0,h]$ and adding the bound on $\pnorm{\mathcal T_2}_4$ give \eqref{eq: local error expansion}. Finally, the pair $\bigl( B_h, \int_0^h ( B_s - \frac12 B_h )\, \D s \bigr)$ is Gaussian and centered, and its two components are uncorrelated, since
    \begin{equation*}
        \mathbb E_0\biggl[ B_h \otimes \int_0^h B_s\, \D s \biggr] = \frac{h^2}{2}\, I_d = \mathbb E_0\Bigl[ B_h \otimes \frac h2 B_h \Bigr] .
    \end{equation*}
    Hence $\int_0^h ( B_s - \frac12 B_h )\, \D s$ is independent of $\chi_0 = B_h / \sqrt h$.
\end{proof}

\subsection{Proof of Lemma~\ref{lem: moment matching}}
\label{sec: matching proof}

The next lemma expands the moments of the full gradient step $D_0$, which the proof of Lemma~\ref{lem: moment matching} compares with the moments of $\tilde W_h$ in Lemma~\ref{lem: langevin moments}.

\begin{lemma}
\label{lem: step moments}
    Under Assumption~\ref{ass: U}, for every $h \in (0,1]$, with all derivatives of $U$ evaluated at $Y_0$ and $C$ depending only on $M$ and $d$, the moments of the full gradient step $D_0$ of \eqref{eq: full gradient step} satisfy, for $1 \Le i,j,k \Le d$,
    \begin{align}
        \Bigl| \mathbb E_0[D_0] + h\, \nabla U + \frac{h^2}{4}\, \nabla \Delta U \Bigr| &\Le C h^3 , \label{eq: step m1}\\
        \Bigl\| \mathbb E_0\bigl[ D_0^{\otimes 2} \bigr] - 2h\, I_d - h^2 \bigl( \nabla U^{\otimes 2} - 2\, \nabla^2 U \bigr) \Bigr\| &\Le C h^3 \bigl( 1+|Y_0| \bigr) , \label{eq: step m2}\\
        \Bigl| \bigl( \mathbb E_0\bigl[ D_0^{\otimes 3} \bigr] \bigr)_{ijk} + 2h^2 \bigl( \delta_{ij}\, \partial_k U + \delta_{ik}\, \partial_j U + \delta_{jk}\, \partial_i U \bigr) \Bigr| &\Le C h^3 \bigl( 1+|Y_0| \bigr)^{3} . \label{eq: step m3}
    \end{align}
\end{lemma}

\begin{proof}
    The full gradient step is $D_0 = \sqrt{2h}\, \chi_0 - h\, \nabla U(Z_0)$ with $Z_0 - Y_0 = \sqrt{h/2}\, \chi_0$. Taylor's formula to the order $3$, with $\norm{\nabla^5 U} \Le M$, gives
    \begin{equation}
        \nabla U(Z_0) = \nabla U + \sqrt{\frac h2}\, \nabla^2 U\, \chi_0 + \frac h4\, \nabla^3 U\bigl[ \chi_0^{\otimes 2} \bigr] + \frac16 \Bigl( \frac h2 \Bigr)^{\frac32} \nabla^4 U\bigl[ \chi_0^{\otimes 3} \bigr] + \varrho_4 , \qquad |\varrho_4| \Le \frac{M}{24} \Bigl( \frac h2 \Bigr)^2 |\chi_0|^4 ,
        \label{eq: gradient taylor}
    \end{equation}
    and Taylor's formula to the orders $0$ and $1$, with $\norm{\nabla^2 U} \Le M$ and $\norm{\nabla^3 U} \Le M$, gives
    \begin{equation}
        \bigl| \nabla U(Z_0) - \nabla U \bigr| \Le M \sqrt{\frac h2}\, |\chi_0| , \qquad
        \Bigl| \nabla U(Z_0) - \nabla U - \sqrt{\frac h2}\, \nabla^2 U\, \chi_0 \Bigr| \Le \frac{M h}{4}\, |\chi_0|^2 .
        \label{eq: gradient taylor crude}
    \end{equation}
    Next we prove \eqref{eq: step m1}--\eqref{eq: step m3} in turn.\\[6pt]
    \textbf{Moment of order $1$}. The terms of degrees $1$ and $3$ in $\chi_0$ of \eqref{eq: gradient taylor} have mean $0$, and $\nabla^3 U[\mathbb E_0 \chi_0^{\otimes 2}] = \nabla^3 U[I_d] = \nabla \Delta U$, so that
    \begin{equation}
        \mathbb E_0\bigl[ \nabla U(Z_0) \bigr] = \nabla U + \frac h4\, \nabla \Delta U + \mathcal O(h^2) .
        \label{eq: gradient mean}
    \end{equation}
    Since $\mathbb E_0[D_0] = -h\, \mathbb E_0[\nabla U(Z_0)]$, \eqref{eq: gradient mean} gives \eqref{eq: step m1}.\\[6pt]
    \textbf{Moment of order $2$}. Since $\mathbb E_0[\chi_0^{\otimes 2}] = I_d$,
    \begin{equation}
        \mathbb E_0\bigl[ D_0^{\otimes 2} \bigr] = 2h\, I_d - 2h\sqrt{2h}\, \mathrm{Sym}\, \mathbb E_0\bigl[ \chi_0 \otimes \nabla U(Z_0) \bigr] + h^2\, \mathbb E_0\bigl[ \nabla U(Z_0)^{\otimes 2} \bigr] .
        \label{eq: step square}
    \end{equation}
    We insert \eqref{eq: gradient taylor} into $\chi_0 \otimes \nabla U(Z_0)$. The terms $\chi_0 \otimes \nabla U$ and $\chi_0 \otimes \nabla^3 U[\chi_0^{\otimes 2}]$ have mean $0$, $\mathbb E_0[\chi_0 \otimes \nabla^2 U\, \chi_0] = \nabla^2 U$, and the terms with $\nabla^4 U[\chi_0^{\otimes 3}]$ and $\varrho_4$ have conditional means of norm at most $C h^{\frac32}$, so that
    \begin{equation*}
        \mathbb E_0\bigl[ \chi_0 \otimes \nabla U(Z_0) \bigr] = \sqrt{\frac h2}\, \nabla^2 U + \mathcal O\bigl( h^{\frac32} \bigr) .
    \end{equation*}
    For the term $h^2\, \mathbb E_0[ \nabla U(Z_0)^{\otimes 2} ]$, the mean \eqref{eq: gradient mean} and the bound on $|\nabla U(Z_0) - \nabla U|$ in \eqref{eq: gradient taylor crude} give
    \begin{equation*}
        \Bigl\| \mathbb E_0\bigl[ \nabla U(Z_0)^{\otimes 2} \bigr] - \nabla U^{\otimes 2} \Bigr\| \Le 2\, |\nabla U|\, \bigl| \mathbb E_0[\nabla U(Z_0)] - \nabla U \bigr| + \mathbb E_0 \bigl| \nabla U(Z_0) - \nabla U \bigr|^2 \Le C h (1+|Y_0|) .
    \end{equation*}
    Inserting these two estimates into \eqref{eq: step square} gives \eqref{eq: step m2}.\\[6pt]
    \textbf{Moment of order $3$}. Since $\mathbb E_0[\chi_0^{\otimes 3}] = 0$,
    \begin{equation}
        \mathbb E_0\bigl[ D_0^{\otimes 3} \bigr] = - 6h^2\, \mathrm{Sym}\, \mathbb E_0\bigl[ \chi_0^{\otimes 2} \otimes \nabla U(Z_0) \bigr] + 3h^2 \sqrt{2h}\, \mathrm{Sym}\, \mathbb E_0\bigl[ \chi_0 \otimes \nabla U(Z_0)^{\otimes 2} \bigr] - h^3\, \mathbb E_0\bigl[ \nabla U(Z_0)^{\otimes 3} \bigr] .
        \label{eq: step cube}
    \end{equation}
    By \eqref{eq: gradient taylor crude}, since $\chi_0^{\otimes 2} \otimes \nabla^2 U\, \chi_0$ has mean $0$,
    \begin{equation*}
        \mathbb E_0\bigl[ \chi_{0,i}\, \chi_{0,j}\, \nabla U(Z_0)_k \bigr] = \delta_{ij}\, \partial_k U + \mathcal O(h) ,
    \end{equation*}
    and the symmetrization of $\delta_{ij}\, \partial_k U$ is $\frac13 ( \delta_{ij}\, \partial_k U + \delta_{ik}\, \partial_j U + \delta_{jk}\, \partial_i U )$. Since $\chi_0 \otimes \nabla U^{\otimes 2}$ has mean $0$, \eqref{eq: gradient taylor crude} and $|\nabla U(Z_0)| \Le M ( 1 + |Y_0| + \sqrt{h/2}\, |\chi_0| )$ give
    \begin{equation*}
        \Bigl\| \mathbb E_0\bigl[ \chi_0 \otimes \nabla U(Z_0)^{\otimes 2} \bigr] \Bigr\| \Le C \sqrt h\, (1+|Y_0|) , \qquad
        \Bigl\| \mathbb E_0\bigl[ \nabla U(Z_0)^{\otimes 3} \bigr] \Bigr\| \Le C (1+|Y_0|)^3 .
    \end{equation*}
    Inserting these three estimates into \eqref{eq: step cube} gives \eqref{eq: step m3}.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem: moment matching}]
    The moments of the displacement $\tilde W_h$ of \eqref{eq: displacements} are expanded in \eqref{eq: langevin m1}--\eqref{eq: langevin m3} for the modified dynamics, with coefficients in $\tilde U_h$, and those of $D_0$ in \eqref{eq: step m1}--\eqref{eq: step m3} with coefficients in $U$. Lemma~\ref{lem: step moments} applies since $h \Le h_0 \Le 1$, and all derivatives are evaluated at $Y_0$. The coefficients of the two expansions differ through $\tilde U_h - U$, and by \eqref{eq: modified class} and the linear growth of $\nabla U$, and of $\nabla \tilde U_h$ from Lemma~\ref{lem: modified class},
    \begin{equation}
        \max_{1 \Le i \Le 3} \norm{\nabla^i \tilde U_h - \nabla^i U} \Le C h \bigl( 1+|Y_0| \bigr) , \qquad \norm{\nabla \tilde U_h^{\otimes 2} - \nabla U^{\otimes 2}} \Le C h \bigl( 1+|Y_0| \bigr)^2 .
        \label{eq: coefficient replacement}
    \end{equation}
    Next we prove \eqref{eq: moment matching}, first for $j = 2$ and $j = 3$ and then for $j = 1$.\\[6pt]
    \textbf{Moments of orders $2$ and $3$}. By \eqref{eq: coefficient replacement}, replacing $\tilde U_h$ by $U$ in the terms of order $h^2$ of \eqref{eq: langevin m2} and \eqref{eq: langevin m3} for the modified dynamics changes them in norm by at most $C h^3 (1+|Y_0|)^2$. Then the approximations of $\mathbb E_0[\tilde W_h^{\otimes j}]$ are those of $\mathbb E_0[D_0^{\otimes j}]$ in \eqref{eq: step m2} and \eqref{eq: step m3}, which proves \eqref{eq: moment matching} for $j = 2$ and $j = 3$.\\[6pt]
    \textbf{Moment of order $1$}. By \eqref{eq: coefficient replacement}, $\norm{\nabla^2 U} \Le M$ and the linear growth of $\nabla \tilde U_h$, replacing $\tilde U_h$ by $U$ in the term of order $h^2$ of \eqref{eq: langevin m1} for the modified dynamics changes it in norm by at most $C h^3 (1+|Y_0|)^2$. Subtracting \eqref{eq: step m1} then gives
    \begin{equation}
        \Bigl| \mathbb E_0[\tilde W_h] - \mathbb E_0[D_0] + h \bigl( \nabla \tilde U_h - \nabla U \bigr) - \frac{h^2}{2} \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) - \frac{h^2}{4}\, \nabla \Delta U \Bigr| \Le C h^3 \bigl( 1+|Y_0| \bigr)^3 .
        \label{eq: m1 difference}
    \end{equation}
    On the ball $\{|Y_0| \Le \delta/h\}$, $\nabla \tilde U_h - \nabla U = \frac h4 \nabla V = \frac h4 ( 2\, \nabla^2 U\, \nabla U - \nabla \Delta U )$ by \eqref{eq: V}, so that
    \begin{equation}
        h \bigl( \nabla \tilde U_h - \nabla U \bigr) - \frac{h^2}{2} \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) - \frac{h^2}{4}\, \nabla \Delta U = 0 .
        \label{eq: m1 cancellation}
    \end{equation}
    Outside that ball, where $1 \Le h|Y_0|/\delta$, \eqref{eq: coefficient replacement} and Assumption~\ref{ass: U} bound the left side of \eqref{eq: m1 cancellation} in norm by
    \begin{equation*}
        C h^2 (1+|Y_0|) \Le C h^2 (1+|Y_0|) \cdot \frac{h |Y_0|}{\delta} \Le C h^3 (1+|Y_0|)^2 .
    \end{equation*}
    On the ball and outside it, \eqref{eq: m1 difference} therefore gives \eqref{eq: moment matching} for $j = 1$.
\end{proof}

\subsection{Proof of Lemma~\ref{lem: sg weak}}
\label{sec: sg weak proof}

We first prove the Lipschitz bound \eqref{eq: covariance lipschitz} on the covariance of the gradient noise. With the gradient noise $\eta$ of \eqref{eq: gradient noise}, the covariance \eqref{eq: covariance} is $\mathcal C(x) = \mathbb E_\xi\bigl[ \eta(x;\xi) \otimes \eta(x;\xi) \bigr]$. By Assumption~\ref{ass: sg}, let
\begin{equation}
    S(x;\xi) := \sup_{|x'-x| \Le 1} \norm{\nabla \hat g(x';\xi)} , \qquad \mathbb E_\xi \big[S(x;\xi)^2 \big] \Le M^2 .
    \label{eq: hessian sup}
\end{equation}
For $|x - x'| \Le 1$, the definition of $S$ and $\norm{\nabla^2 U} \Le M$ give
\begin{equation*}
    |\eta(x;\xi) - \eta(x';\xi)| \Le \bigl( S(x;\xi) + M \bigr) |x - x'| ,
\end{equation*}
and the Cauchy--Schwarz inequality, with the variance bound of Assumption~\ref{ass: sg} and \eqref{eq: hessian sup}, gives
\begin{equation*}
    \norm{\mathcal C(x) - \mathcal C(x')} \Le \mathbb E_\xi\Bigl[ \bigl( |\eta(x;\xi)| + |\eta(x';\xi)| \bigr) |\eta(x;\xi) - \eta(x';\xi)| \Bigr] \Le 2\Sigma^{\frac12} \cdot 2M\, |x - x'| .
\end{equation*}
For $|x - x'| > 1$, since $\mathcal C(x)$ and $\mathcal C(x')$ are positive semidefinite, the bound on the trace in \eqref{eq: covariance} gives
\begin{equation*}
    \norm{\mathcal C(x) - \mathcal C(x')} \Le \mathrm{Tr}\bigl[ \mathcal C(x) \bigr] + \mathrm{Tr}\bigl[ \mathcal C(x') \bigr] \Le 2\Sigma \Le 2\Sigma\, |x - x'| .
\end{equation*}
Since $\Sigma \Le M^2$, both cases give \eqref{eq: covariance lipschitz}.

\begin{proof}[Proof of Lemma~\ref{lem: sg weak}]
    Work at the index $0$. We first prove the bound for the full gradient step,
    \begin{equation}
        \bigl| \mathbb E_0\bigl[ G(Y_0 + D_0) - G(\tilde Y_0(h)) \bigr] \bigr| \Le C \Bigl( h^3 \sum_{j=1}^{3} \norm{\nabla^j G}_\infty + h^{\frac72} \norm{\nabla^4 G}_\infty \Bigr) \bigl( 1+|Y_0| \bigr)^6 .
        \label{eq: weak}
    \end{equation}
    With the interpolated position $Q_\varsigma := \varsigma ( Y_0 + D_0 ) + (1 - \varsigma) \tilde Y_0(h)$, $\varsigma \in [0,1]$,
    \begin{equation}
        G(Y_0 + D_0) - G(\tilde Y_0(h)) = \int_0^1 \bigl\langle \nabla G(Q_\varsigma),\, Y_0 + D_0 - \tilde Y_0(h) \bigr\rangle \,\D\varsigma .
        \label{eq: weak start}
    \end{equation}
    Taylor's formula for $\nabla G$ at $Y_0$ gives
    \begin{equation*}
        \nabla G(Y_0 + v) = \nabla G(Y_0) + \nabla^2 G(Y_0)\, v + \frac12\, \nabla^3 G(Y_0)\bigl[ v^{\otimes2} \bigr] + \varrho_3(v) ,
        \qquad |\varrho_3(v)| \Le \frac16\, \norm{\nabla^4 G}_\infty\, |v|^3 ,
    \end{equation*}
    and inserting this formula with $v = Q_\varsigma - Y_0$ into \eqref{eq: weak start}, where $Q_1 - Y_0 = D_0$ and $Q_0 - Y_0 = \tilde W_h$, gives
    \begin{equation}
        G(Y_0 + D_0) - G(\tilde Y_0(h)) = \sum_{j=1}^{3} \frac{1}{j!}\, \nabla^j G(Y_0) : \Bigl( D_0^{\otimes j} - \tilde W_h^{\otimes j} \Bigr) + \varrho_4 ,
        \label{eq: G expansion full}
    \end{equation}
    with the remainder
    \begin{equation*}
        \varrho_4 := \int_0^1 \bigl\langle \varrho_3(Q_\varsigma - Y_0),\, Y_0 + D_0 - \tilde Y_0(h) \bigr\rangle \,\D\varsigma .
    \end{equation*}
    For the terms of orders $1$ to $3$ in \eqref{eq: G expansion full}, the moment matching \eqref{eq: moment matching} gives
    \begin{equation}
        \biggl| \sum_{j=1}^{3} \frac{1}{j!}\, \nabla^j G(Y_0) : \mathbb E_0\bigl[ D_0^{\otimes j} - \tilde W_h^{\otimes j} \bigr] \biggr|
        \Le C h^3 \sum_{j=1}^{3} \norm{\nabla^j G}_\infty \bigl( 1 + |Y_0| \bigr)^{6} .
        \label{eq: weak matched}
    \end{equation}

    It remains to bound the mean $\mathbb E_0[\varrho_4]$ of the remainder. With $\mathfrak g$ of \eqref{eq: leading term}, the integrand of $\varrho_4$ is
    \begin{equation}
        \bigl\langle \varrho_3(Q_\varsigma - Y_0),\, Y_0 + D_0 - \tilde Y_0(h) \bigr\rangle = \mathcal T_1 + \mathcal T_2 + \mathcal T_3 ,
        \label{eq: remainder split}
    \end{equation}
    where the terms $\mathcal T_1$ to $\mathcal T_3$ are
    \begin{equation*}
    \begin{gathered}
        \mathcal T_1 = \bigl\langle \varrho_3(\sqrt{2h}\, \chi_0),\, \mathfrak g \bigr\rangle , \qquad
        \mathcal T_2 = \bigl\langle \varrho_3(\sqrt{2h}\, \chi_0),\, Y_0 + D_0 - \tilde Y_0(h) - \mathfrak g \bigr\rangle , \\
        \mathcal T_3 = \bigl\langle \varrho_3(Q_\varsigma - Y_0) - \varrho_3(\sqrt{2h}\, \chi_0),\, Y_0 + D_0 - \tilde Y_0(h) \bigr\rangle .
    \end{gathered}
    \end{equation*}
    By Lemma~\ref{lem: sizes}, $\mathfrak g$ is centered and independent of $\chi_0$, so
    \begin{equation*}
        \mathbb E_0[\mathcal T_1] = \Bigl\langle \mathbb E_0\bigl[ \varrho_3(\sqrt{2h}\, \chi_0) \bigr],\, \mathbb E_0[\mathfrak g] \Bigr\rangle = 0 .
    \end{equation*}
    The bound on $\varrho_3$ at $v = \sqrt{2h}\, \chi_0$, H\"older's inequality and the expansion \eqref{eq: local error expansion} give
    \begin{equation*}
        \bigl| \mathbb E_0[\mathcal T_2] \bigr| \Le \frac16\, \norm{\nabla^4 G}_\infty\, (2h)^{\frac32}\, \pnorm{\chi_0}_4^3\, \pnorm{Y_0 + D_0 - \tilde Y_0(h) - \mathfrak g}_4 \Le C \norm{\nabla^4 G}_\infty\, h^{\frac72} \bigl( 1+|Y_0| \bigr)^{2} .
    \end{equation*}
    For $\mathcal T_3$, the derivative of $\varrho_3$ satisfies
    \begin{equation*}
        \nabla \varrho_3(v) = \nabla^2 G(Y_0 + v) - \nabla^2 G(Y_0) - \nabla^3 G(Y_0)[v] , \qquad \norm{\nabla \varrho_3(v)} \Le \frac12\, \norm{\nabla^4 G}_\infty\, |v|^2 ,
    \end{equation*}
    so that
    \begin{equation*}
        \bigl| \varrho_3(v) - \varrho_3(v') \bigr| \Le \frac12\, \norm{\nabla^4 G}_\infty \bigl( |v| + |v'| \bigr)^2 |v - v'| , \qquad v, v' \in \mathbb R^d .
    \end{equation*}
    By the definition of $Q_\varsigma$,
    \begin{equation*}
        Q_\varsigma - Y_0 = \varsigma D_0 + (1-\varsigma) \tilde W_h , \qquad
        Q_\varsigma - Y_0 - \sqrt{2h}\, \chi_0 = -\varsigma h\, \nabla U(Z_0) + (1-\varsigma) \bigl( \tilde W_h - \sqrt{2h}\, \chi_0 \bigr) ,
    \end{equation*}
    so Assumption~\ref{ass: U}, \eqref{eq: sizes} and \eqref{eq: langevin sizes} for the modified dynamics give
    \begin{equation*}
        \pnorm{Q_\varsigma - Y_0}_4 \Le C \sqrt h\, \bigl( 1+|Y_0| \bigr) , \qquad
        \pnorm{Q_\varsigma - Y_0 - \sqrt{2h}\, \chi_0}_4 \Le C h\, \bigl( 1+|Y_0| \bigr) .
    \end{equation*}
    The Lipschitz bound on $\varrho_3$ with $v = Q_\varsigma - Y_0$ and $v' = \sqrt{2h}\, \chi_0$, H\"older's inequality and \eqref{eq: sizes} then give
    \begin{equation*}
    \begin{aligned}
        \bigl| \mathbb E_0[\mathcal T_3] \bigr| & \Le \frac12\, \norm{\nabla^4 G}_\infty \Bigl( \pnorm{Q_\varsigma - Y_0}_4 + \pnorm{\sqrt{2h}\, \chi_0}_4 \Bigr)^2\, \pnorm{Q_\varsigma - Y_0 - \sqrt{2h}\, \chi_0}_4\, \pnorm{Y_0 + D_0 - \tilde Y_0(h)}_4 \\
        & \Le C \norm{\nabla^4 G}_\infty\, h^{\frac72} \bigl( 1+|Y_0| \bigr)^{4} .
    \end{aligned}
    \end{equation*}
    Integrating \eqref{eq: remainder split} over $\varsigma \in [0,1]$, with the bounds on $\mathbb E_0[\mathcal T_1]$, $\mathbb E_0[\mathcal T_2]$ and $\mathbb E_0[\mathcal T_3]$, gives
    \begin{equation}
        \bigl| \mathbb E_0[\varrho_4] \bigr| \Le C \norm{\nabla^4 G}_\infty\, h^{\frac72} \bigl( 1+|Y_0| \bigr)^{4} .
        \label{eq: weak remainder}
    \end{equation}
    By \eqref{eq: G expansion full}, the bounds \eqref{eq: weak matched} and \eqref{eq: weak remainder} prove \eqref{eq: weak}.

    Next we add the gradient noise. The vectors $Y_0$, $D_0$, $Z_0$ and $\tilde Y_0(h)$ are functions of $Y_0$ and $(B_s)_{s \Le h}$, and $\xi_0$ is independent of $\mathcal F_0$ and $(B_s)_{s \Le h}$, so Assumption~\ref{ass: sg} and \eqref{eq: covariance} give
    \begin{equation*}
        \mathbb E\bigl[ \eta_0 \,\big|\, \mathcal F_0, (B_s)_{s \Le h} \bigr] = 0 , \qquad \mathbb E\bigl[ \eta_0 \otimes \eta_0 \,\big|\, \mathcal F_0, (B_s)_{s \Le h} \bigr] = \mathcal C(Z_0) .
    \end{equation*}
    Taylor's formula for $G$ at $Y_0 + D_0$, to the order $2$ in $h\eta_0$, gives
    \begin{equation*}
        G(Y_1) = G(Y_0 + D_0) - h\, \nabla G(Y_0+D_0) \cdot \eta_0 + \frac{h^2}{2}\, \nabla^2 G(Y_0+D_0)[\eta_0, \eta_0] + \varrho_3^{\eta_0} , \qquad |\varrho_3^{\eta_0}| \Le \frac{h^3}{6}\, \norm{\nabla^3 G}_\infty\, |\eta_0|^3 .
    \end{equation*}
    Given $\mathcal F_0$ and $(B_s)_{s \Le h}$, the linear term has mean $0$ and the quadratic term has mean $\frac{h^2}{2}\, \nabla^2 G(Y_0+D_0) : \mathcal C(Z_0)$. The moment bound of Assumption~\ref{ass: sg} and $|\nabla U(x)| \Le M(1+|x|)$ give $\mathbb E_0 |\eta_0|^3 \Le C ( 1 + |Y_0| )^3$. Hence
    \begin{equation}
        \Bigl| \mathbb E_0\bigl[ G(Y_1) - G(Y_0 + D_0) \bigr] - \frac{h^2}{2}\, \mathbb E_0\bigl[ \nabla^2 G(Y_0 + D_0) : \mathcal C(Z_0) \bigr] \Bigr| \Le C h^3 \norm{\nabla^3 G}_\infty \bigl( 1+|Y_0| \bigr)^3 .
        \label{eq: sg noise expansion}
    \end{equation}
    Adding \eqref{eq: weak} and \eqref{eq: sg noise expansion} proves \eqref{eq: sg weak} at the index $0$, and by time homogeneity at every $k$.
\end{proof}

\section{Proof of Theorem~\ref{thm: sglm mse}: MSE of SGLM}
\label{app: mse}

Throughout this appendix, $C$ depends only on $m$, $M$, $R$ and $d$, and $h \Le h_0$ with $h_0 \Le \min\{\lambda^{-1}, \frac12\}$, $\lambda$ the rate of Lemma~\ref{lem: smoothing}, so that Corollary~\ref{cor: modified poisson} applies. By \eqref{eq: modified poisson bound}, $\tilde\phi_h$ is Lipschitz with the constant $C \norm{\nabla f}_\infty$.

\begin{proof}[Proof of Theorem~\ref{thm: sglm mse}]
    Let first $Nh \Ge 1$, so that $N \Ge 2$ since $h \Le \frac12$. The first step of \eqref{eq: sglm} gives
    \begin{equation*}
        Z_k = Y_k + \sqrt{\frac h2}\,\chi_k , \quad k \Ge 1 , \quad Y_1 := Z_0 - h\, \hat g(Z_0;\xi_0) + \sqrt{\frac h2}\,\chi_0 ,
    \end{equation*}
    with $(Y_k)_{k \Ge 1}$ the chain \eqref{eq: sglm markov} started from $Y_1$, where $Y_1$ is independent of $(B_t - B_h)_{t \Ge h}$ and $(\xi_k)_{k \Ge 1}$, and $\mathbb E |Y_1|^{12} \Le C (1 + |Z_0|)^{12}$ by the moment bound of Assumption~\ref{ass: sg}. With the moments of $\pi$ in Lemma~\ref{lem: dissipative}, and $|f(x) - \pi(f)| \Le C \norm{\nabla f}_\infty (1 + |x|)$, \eqref{eq: mse} gives
    \begin{equation*}
        \MSE(N,h) \Le \frac{C \norm{\nabla f}_\infty^2}{Nh} \bigl( 1+|Z_0| \bigr)^2 + 2\, \mathbb E \Biggl[ \biggl| \frac{1}{N-1} \sum_{k=1}^{N-1} f(Z_k) - \pi(f) \biggr|^2 \Biggr] .
    \end{equation*}
    The expectation is the MSE of $N-1$ samples of the chain started from $Y_1$, with $(N-1)h \Ge \frac12$. Conditioning on $Y_1$, it therefore suffices to prove \eqref{eq: sglm mse} for the samples $Z_k = Y_k + \sqrt{h/2}\,\chi_k$, $k \Ge 0$, of the chain \eqref{eq: sglm markov} started from a given $Y_0$, with $Y_0$ in place of $Z_0$ and $Nh \Ge \frac12$. By \eqref{eq: mse split modified} and $\frac{h}{N} \Le \frac{1}{Nh}$, it suffices to prove
    \begin{equation}
        \mathbb E \Biggl[ \biggl| \frac1N \sum_{k=0}^{N-1} \bigl( \tilde f_h(Y_k) - \tilde\pi_h(\tilde f_h) \bigr) \biggr|^2 \Biggr] \Le C \norm{\nabla f}_\infty^2 \biggl[ \bigl( 1 + |Y_0| \bigr)^2 \Bigl( \frac{1}{Nh} + \Sigma^2 h^2 \Bigr) + \bigl( 1 + |Y_0| \bigr)^{12} h^4 \log^2\frac1h \biggr] .
        \label{eq: markov mse}
    \end{equation}
    By \eqref{eq: time average formula} and Cauchy's inequality, the left side of \eqref{eq: markov mse} is at most
    \begin{equation}
        \frac{3}{N^2h^2}\, \mathbb E\bigl| \tilde\phi_h(Y_0) - \tilde\phi_h(Y_N) \bigr|^2
        + \frac{3}{N^2}\, \mathbb E\biggl| \sum_{k=0}^{N-1} \Upsilon_k \biggr|^2
        + \frac{3}{N^2}\, \mathbb E\biggl| \sum_{k=0}^{N-1} \Psi_k \biggr|^2 ,
        \label{eq: three terms}
    \end{equation}
    where $\Upsilon_k$ and $\Psi_k$ are defined in \eqref{eq: Upsilon Psi}. The terms $\Upsilon_k$ and $\Psi_k$ are $\mathcal F_{k+1}$-measurable, and $\mathbb E_k \Psi_k = 0$ by \eqref{eq: modified poisson}. Hence the $\Psi_k$, and the centered terms $\Upsilon_k - \mathbb E_k \Upsilon_k$, are martingale differences with respect to $(\mathcal F_k)_{k \Ge 0}$, and their sums have additive second moments. Lemma~\ref{lem: moments} also gives
    \begin{equation}
        \mathbb E|Y_k|^2 \Le C \bigl( 1+|Y_0| \bigr)^2 , \qquad \mathbb E|Y_{k+1} - Y_k|^2 + \mathbb E|\tilde Y_k(h) - Y_k|^2 \Le C h \bigl( 1+|Y_0| \bigr)^2 .
        \label{eq: markov second moments}
    \end{equation}
    For the boundary term of \eqref{eq: three terms}, the Lipschitz bound on $\tilde\phi_h$ and \eqref{eq: markov second moments} give
    \begin{equation}
        \frac{3}{N^2h^2}\, \mathbb E\bigl| \tilde\phi_h(Y_0) - \tilde\phi_h(Y_N) \bigr|^2 \Le \frac{C \norm{\nabla f}_\infty^2}{N^2h^2}\, \mathbb E|Y_N - Y_0|^2 \Le \frac{C \norm{\nabla f}_\infty^2}{N^2 h^2} \bigl( 1+|Y_0| \bigr)^2 \Le \frac{C \norm{\nabla f}_\infty^2}{Nh} \bigl( 1+|Y_0| \bigr)^2 .
    \label{eq: boundary term bound}
    \end{equation}
    For the discretization term of \eqref{eq: three terms}, the bound \eqref{eq: sg weak} with $G = \tilde\phi_h$, the bounds \eqref{eq: modified poisson bound} and the bound on the trace in \eqref{eq: covariance} bound the conditional mean of $\Upsilon_k$,
    \begin{equation*}
        |\mathbb E_k \Upsilon_k| \Le C \norm{\nabla f}_\infty \Bigl( \Sigma h + h^2 \log\frac1h\, \bigl( 1+|Y_k| \bigr)^6 \Bigr) ,
    \end{equation*}
    since $\frac{h^2}{2} \bigl| \mathbb E_k\bigl[ \nabla^2 \tilde\phi_h(Y_k + D_k) : \mathcal C(Z_k) \bigr] \bigr| \Le \frac{\Sigma h^2}{2} \norm{\nabla^2 \tilde\phi_h}_\infty \Le C \norm{\nabla f}_\infty \Sigma h^2$. For the second moment, the Lipschitz bound on $\tilde\phi_h$, the split \eqref{eq: local error split} at the index $k$, the bound \eqref{eq: sizes} on its discretization part and the variance bound of Assumption~\ref{ass: sg} give
    \begin{equation*}
        \mathbb E_k |\Upsilon_k|^2 \Le \frac{C \norm{\nabla f}_\infty^2}{h^2}\, \mathbb E_k \bigl| Y_{k+1} - \tilde Y_k(h) \bigr|^2 \Le C \norm{\nabla f}_\infty^2 \Bigl( h \bigl( 1+|Y_k| \bigr)^2 + \Sigma \Bigr) \Le C \norm{\nabla f}_\infty^2 \bigl( 1+|Y_k| \bigr)^2 .
    \end{equation*}
    The orthogonality of the martingale differences $\Upsilon_k - \mathbb E_k \Upsilon_k$ and Lemma~\ref{lem: moments} then give
    \begin{equation}
    \begin{aligned}
        \frac{3}{N^2}\, \mathbb E\biggl| \sum_{k=0}^{N-1} \Upsilon_k \biggr|^2 & \Le \frac6N \sum_{k=0}^{N-1} \mathbb E\bigl| \mathbb E_k \Upsilon_k \bigr|^2 + \frac{6}{N^2} \sum_{k=0}^{N-1} \mathbb E|\Upsilon_k|^2 \\
        & \Le C \norm{\nabla f}_\infty^2 \biggl[ \bigl( 1+|Y_0| \bigr)^2 \Bigl( \Sigma^2 h^2 + \frac1N \Bigr) + \bigl( 1+|Y_0| \bigr)^{12} h^4 \log^2\frac1h \biggr] .
    \end{aligned}
        \label{eq: discretization term bound}
    \end{equation}
    For the martingale term of \eqref{eq: three terms}, $h \Psi_k = \tilde\phi_h(\tilde Y_k(h)) - \mathbb E_k[\tilde\phi_h(\tilde Y_k(h))]$ by \eqref{eq: Upsilon Psi} and \eqref{eq: modified poisson}, so the orthogonality of the $\Psi_k$, the Lipschitz bound on $\tilde\phi_h$ and \eqref{eq: markov second moments} give
    \begin{equation}
        \frac{3}{N^2}\, \mathbb E\biggl| \sum_{k=0}^{N-1} \Psi_k \biggr|^2 = \frac{3}{N^2} \sum_{k=0}^{N-1} \mathbb E|\Psi_k|^2 \Le \frac{C \norm{\nabla f}_\infty^2}{N^2 h^2} \sum_{k=0}^{N-1} \mathbb E\bigl| \tilde Y_k(h) - Y_k \bigr|^2 \Le \frac{C \norm{\nabla f}_\infty^2}{Nh} \bigl( 1+|Y_0| \bigr)^2 .
        \label{eq: martingale term bound}
    \end{equation}
    Adding \eqref{eq: boundary term bound}, \eqref{eq: discretization term bound} and \eqref{eq: martingale term bound}, and using $\frac{1}{N} \Le \frac{1}{Nh}$, proves \eqref{eq: markov mse}.

    Let finally $Nh < 1$. Since $\mathbb E|Z_k|^2 \Le C (1 + |Z_0|)^2$ for every $k \Ge 0$ by the moment bound on $Y_1$ and Lemma~\ref{lem: moments}, Cauchy's inequality gives
    \begin{equation*}
        \MSE(N,h) \Le \frac1N \sum_{k=0}^{N-1} \mathbb E\bigl| f(Z_k) - \pi(f) \bigr|^2 \Le C \norm{\nabla f}_\infty^2 \bigl( 1 + |Z_0| \bigr)^2 \Le \frac{C \norm{\nabla f}_\infty^2}{Nh} \bigl( 1 + |Z_0| \bigr)^2 ,
    \end{equation*}
    which is at most the right side of \eqref{eq: sglm mse}.
\end{proof}

\section{Proofs of the numerical bias estimate}
\label{app: bias}

Throughout this appendix, $(Y_k)_{k \Ge 0}$ is the chain \eqref{eq: sglm markov} and $\mathcal P_h$ its transition kernel. The constant $C$ depends only on $m$, $M$, $R$ and $d$, and $h \Le h_0$ with $M h_0 \Le 1$.

\subsection{Proof of Lemma~\ref{lem: invariant}: invariant distribution}
\label{sec: invariant proof}

\begin{proof}[Proof of Lemma~\ref{lem: invariant}]
    Harris's ergodic theorem \citep[Theorem~1.2]{hairer2008yet} provides sufficient conditions to verify the existence of an invariant distribution of a Markov chain with the transition kernel $\mathcal P_h$. It assumes that $W \colon \mathbb R^d \to [0,\infty)$, $\gamma_h, \alpha_h \in (0,1)$, $K_h \Ge 0$, $R_W > 0$ and a probability measure $\nu$ satisfy the drift condition
    \begin{equation}
        (\mathcal P_h W)(y) \Le \gamma_h W(y) + K_h , \qquad y \in \mathbb R^d ,
        \label{eq: harris drift}
    \end{equation}
    and the minorization on the level set $\{W \Le R_W\}$, with the condition on the level,
    \begin{equation}
        \mathcal P_h(y, \cdot) \Ge \alpha_h\, \nu , \quad W(y) \Le R_W , \qquad
        R_W > \frac{2K_h}{1-\gamma_h} .
        \label{eq: harris minorization}
    \end{equation}
    Then it asserts that $\mathcal P_h$ has a unique invariant distribution $\mu_h$, with
    \begin{equation}
        \lim_{k \to \infty} (\mathcal P_h^k \psi)(y) = \mu_h(\psi) , \qquad y \in \mathbb R^d ,
        \label{eq: harris}
    \end{equation}
    for every bounded measurable $\psi$.

    For the drift condition \eqref{eq: harris drift}, take $W(y) := |y|^2$. The bound on $\mathbb E_k |Y_{k+1}|^2$ in \eqref{eq: lyapunov}, from the proof of Lemma~\ref{lem: moments}, is the drift condition \eqref{eq: harris drift}, and the condition on the level in \eqref{eq: harris minorization} holds, with
    \begin{equation*}
        \gamma_h := 1 - \frac{m}{8} h , \qquad K_h := C h , \qquad R_W := \frac{64 C}{m} > \frac{16 C}{m} = \frac{2K_h}{1 - \gamma_h} ,
    \end{equation*}
    where $C$ is the constant of the bound on $\mathbb E_k |Y_{k+1}|^2$ in \eqref{eq: lyapunov} and $R_W$ depends only on $m$, $M$, $R$ and $d$.

    Next we prove the minorization \eqref{eq: harris minorization} on the level set $\{W \Le R_W\}$, with a constant $\alpha_h$ that may depend on $h$. Let $h_0$ also satisfy $4 M h_0 (2 + R_W^{\frac12}) \Le 1$, let $\mathbb B$ be the closed unit ball and $\nu$ the uniform distribution on $\mathbb B$, and fix $y$ with $W(y) \Le R_W$. From $Y_0 = y$, the step \eqref{eq: sg chain step} reads
    \begin{equation*}
        Y_1 = T_{\xi_0}\Bigl( \frac y2 + \sqrt{\frac h2}\,\chi_0 \Bigr) , \qquad T_\xi(v) := 2v - h\, \hat g\Bigl( \frac y2 + v; \xi \Bigr) , \qquad v \in \mathbb R^d .
    \end{equation*}
    By Markov's inequality, the bound \eqref{eq: hessian sup} on $S$ and the moment bound of Assumption~\ref{ass: sg},
    \begin{equation*}
        \mathbb P_\xi\biggl( S\Bigl( \frac y2;\xi \Bigr) \Le 4M , \quad \Bigl| \hat g\Bigl( \frac y2;\xi \Bigr) \Bigr| \Le 4M \Bigl( 1 + \frac{|y|}{2} \Bigr) \biggr) \Ge \frac12 .
    \end{equation*}
    For $\xi$ in this event, the map $v \mapsto h\, \hat g( \frac y2 + v;\xi )$ is Lipschitz with the constant $4 M h \Le \frac12$ on $\mathbb B$, and the condition $4 M h_0 (2 + R_W^{\frac12}) \Le 1$ then gives $h\, | \hat g( \frac y2 + v;\xi ) | \Le 1$ for $v \in \mathbb B$. Hence $T_\xi$ is Lipschitz with the constant $3$ on $\mathbb B$. Moreover, for every $w \in \mathbb B$, the map $v \mapsto \frac12 \bigl( w + h\, \hat g( \frac y2 + v;\xi ) \bigr)$ is a contraction of $\mathbb B$. Its fixed point $v$ satisfies $T_\xi(v) = w$, so that $\mathbb B \subset T_\xi(\mathbb B)$. Hence the Lebesgue measure satisfies $\mathrm{Leb}(A \cap \mathbb B) \Le 3^d\, \mathrm{Leb}(T_\xi^{-1}(A) \cap \mathbb B)$ for every Borel set $A$. The density of $\frac y2 + \sqrt{h/2}\,\chi_0$ on $\mathbb B$ is at least a constant $\beta_h > 0$ that does not depend on $y$. Since $\chi_0$ is independent of $\xi_0$,
    \begin{equation*}
        \mathcal P_h(y, A) \Ge \frac{\beta_h}{2 \cdot 3^d}\, \mathrm{Leb}(A \cap \mathbb B) = \frac{\beta_h\, \mathrm{Leb}(\mathbb B)}{2 \cdot 3^d}\, \nu(A) ,
    \end{equation*}
    which is the minorization \eqref{eq: harris minorization} with $\alpha_h := \beta_h\, \mathrm{Leb}(\mathbb B) / (2 \cdot 3^d) \in (0,1)$. Harris's ergodic theorem therefore gives the unique invariant distribution $\mu_h$ and the limit \eqref{eq: harris}.

    For the moment bound of $\mu_h$, the limit \eqref{eq: harris} with the truncation $\psi(y) = \min\{|y|^{12}, \ell\}$ for $\ell \Ge 1$ and Lemma~\ref{lem: moments} for the chain started from $Y_0 = 0$ give $\mu_h(\psi) = \lim_{k \to \infty} \mathbb E\bigl[ \psi(Y_k) \bigr] \Le C$. The limit $\ell \to \infty$ gives $\mu_h(|y|^{12}) \Le C$.

    Next we prove the limit of time averages for the chain \eqref{eq: sglm markov}. By the drift condition \eqref{eq: harris drift} and the minorization \eqref{eq: harris minorization}, the chain is positive Harris recurrent \citep{meyn1993markov}. Since $\tilde f_h$ is Lipschitz, the moment bound of $\mu_h$ gives $\mu_h(|\tilde f_h|) < \infty$, and the strong law of large numbers for positive Harris recurrent chains \citep[Theorem~17.0.1(i)]{meyn1993markov} gives, for every initial value $Y_0$,
    \begin{equation}
        \frac1N \sum_{k=0}^{N-1} \tilde f_h(Y_k) \to \mu_h(\tilde f_h) \qquad \text{almost surely} .
        \label{eq: chain limit}
    \end{equation}

    Finally, we prove \eqref{eq: ergodic limit} for the samples of \eqref{eq: sglm} started from the given $Z_0$. As in the proof of Theorem~\ref{thm: sglm mse}, $Z_k = Y_k + \sqrt{h/2}\,\chi_k$ for $k \Ge 1$, with $(Y_k)_{k \Ge 1}$ the chain \eqref{eq: sglm markov} started from $Y_1 = Z_0 - h\, \hat g(Z_0;\xi_0) + \sqrt{h/2}\,\chi_0$, where $Y_1$ is independent of $(\chi_k, \xi_k)_{k \Ge 1}$. As in Section~\ref{subsec: modified test function}, the differences $\mathfrak M_k = f(Z_k) - \tilde f_h(Y_k)$, $k \Ge 1$, are martingale differences with respect to $(\mathcal F_k)_{k \Ge 1}$, with $\mathbb E[\mathfrak M_k^2] \Le \frac{dh}{2} \norm{\nabla f}_\infty^2$ by \eqref{eq: martingale variance}. The limit \eqref{eq: chain limit} for the chain $(Y_k)_{k \Ge 1}$ and the strong law of large numbers for martingale differences bounded in $L^2$ give
    \begin{equation*}
        \frac1N \sum_{k=1}^{N-1} \tilde f_h(Y_k) \to \mu_h(\tilde f_h) , \qquad \frac1N \sum_{k=1}^{N-1} \mathfrak M_k \to 0 \qquad \text{almost surely} .
    \end{equation*}
    Adding the two limits gives $\frac1N \sum_{k=0}^{N-1} f(Z_k) \to \mu_h(\tilde f_h)$ almost surely. The definition \eqref{eq: heat shift} of $\tilde f_h$ and $\pi_h = \mu_h * \mathcal N(0, \frac h2 I_d)$ give
    \begin{equation}
        \mu_h(\tilde f_h) = \pi_h(f) ,
        \label{eq: heat shift invariant}
    \end{equation}
    which proves \eqref{eq: ergodic limit}.
\end{proof}

\subsection{Proof of Theorem~\ref{thm: numerical bias}: numerical bias}
\label{sec: numerical bias proof}

Recall that $\mu_h$ is the invariant distribution of \eqref{eq: sglm markov} and $\pi_h = \mu_h * \mathcal N(0, \frac h2 I_d)$.
The step \eqref{eq: sg chain step} is $Y_1 = Y_0 + D_0 - h\, \eta_0$, with the full gradient step $D_0$ of \eqref{eq: full gradient step} and the gradient noise $\eta_0$ of \eqref{eq: gradient noise}.
The initial value $Y_0 \sim \mu_h$ is independent of $(B_t)_{t \Ge 0}$ and $(\xi_k)_{k \Ge 0}$, so that $Y_1 \sim \mu_h$. With $Y_0$ included in $\mathcal F_0$ of \eqref{eq: filtration}, $\mathbb E_0\bigl[ \psi(\tilde Y_0(h)) \bigr] = (e^{h\tilde{\mathcal L}_h} \psi)(Y_0)$ for every Lipschitz function $\psi$. For every Lipschitz tensor-valued function $\psi$, the heat semigroup satisfies
\begin{equation}
    e^{\frac h4 \Delta} \psi - \psi = \int_0^{h/4} \Delta ( e^{s\Delta} \psi )\, \D s ,
    \label{eq: heat shift integral}
\end{equation}
and $e^{\frac h4 \Delta} f = \tilde f_h$ for every Lipschitz function $f$ by \eqref{eq: heat shift}. The next lemma bounds $\mathcal W_1(\mu_h, \pi)$, with which the proof of Theorem~\ref{thm: numerical bias} moves an expectation from $\mu_h$ to $\pi$.

\begin{lemma}
\label{lem: rough}
    Under Assumptions~\ref{ass: U} and \ref{ass: sg}, for every $h \in (0,h_0]$, $\mathcal W_1(\mu_h, \pi) \Le C h$.
\end{lemma}

\begin{proof}
    By the Kantorovich--Rubinstein duality and a mollification, it suffices to prove $|\mu_h(f) - \pi(f)| \Le C h$ for $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$ and $\norm{\nabla f}_\infty \Le 1$. Let
    \begin{equation*}
        \mathfrak w_h := h \sum_{i=1}^{\infty} \bigl( e^{ih\tilde{\mathcal L}_h} f - \tilde\pi_h(f) \bigr) .
    \end{equation*}
    As in the proof of Corollary~\ref{cor: modified poisson}, the series converges and is differentiable term by term. By the bound \eqref{eq: summed smoothing} and $\norm{\nabla f}_\infty \Le 1$, the function $\mathfrak w_h \in C^4(\mathbb R^d)$ satisfies
    \begin{equation}
        \norm{\nabla \mathfrak w_h}_\infty + \norm{\nabla^2 \mathfrak w_h}_\infty \Le C , \qquad \norm{\nabla^3 \mathfrak w_h}_\infty \Le C \log\frac1h , \qquad \norm{\nabla^4 \mathfrak w_h}_\infty \Le C h^{-\frac12} .
        \label{eq: rough poisson bound}
    \end{equation}
    Telescoping the series gives $\mathfrak w_h - e^{h\tilde{\mathcal L}_h} \mathfrak w_h = h \bigl( e^{h\tilde{\mathcal L}_h} f - \tilde\pi_h(f) \bigr)$. Since $Y_0$ and $Y_1$ have the law $\mu_h$, taking expectations in this identity at $Y_0$ gives
    \begin{equation}
        \mu_h(f) - \tilde\pi_h(f) = \mathbb E\bigl[ f(Y_1) - f(\tilde Y_0(h)) \bigr] + \frac1h\, \mathbb E\bigl[ \mathfrak w_h(Y_1) - \mathfrak w_h(\tilde Y_0(h)) \bigr] .
        \label{eq: rough identity}
    \end{equation}
    For the term with $f$ in \eqref{eq: rough identity}, the bound \eqref{eq: sizes} on $Y_0 + D_0 - \tilde Y_0(h)$, the variance bound of Assumption~\ref{ass: sg} with $\Sigma \Le M^2$ and the moment bound $\mu_h(|y|^{12}) \Le C$ of Lemma~\ref{lem: invariant} give
    \begin{equation*}
        \bigl| \mathbb E\bigl[ f(Y_1) - f(\tilde Y_0(h)) \bigr] \bigr| \Le \mathbb E \bigl| Y_0 + D_0 - \tilde Y_0(h) \bigr| + h\, \mathbb E |\eta_0| \Le C h^{\frac32} + \Sigma^{\frac12} h \Le C h .
    \end{equation*}
    For the term with $\mathfrak w_h$ in \eqref{eq: rough identity}, the bound \eqref{eq: sg weak} at $k = 0$ with $G = \mathfrak w_h$, the bound on the trace in \eqref{eq: covariance}, the bounds \eqref{eq: rough poisson bound} and $\mu_h(|y|^{12}) \Le C$ give
    \begin{equation*}
        \frac1h\, \bigl| \mathbb E\bigl[ \mathfrak w_h(Y_1) - \mathfrak w_h(\tilde Y_0(h)) \bigr] \bigr| \Le \frac{\Sigma h}{2}\, \norm{\nabla^2 \mathfrak w_h}_\infty + C \Bigl( h^2 \sum_{j=1}^{3} \norm{\nabla^j \mathfrak w_h}_\infty + h^{\frac52} \norm{\nabla^4 \mathfrak w_h}_\infty \Bigr) \Le C h .
    \end{equation*}
    Hence
    \begin{equation}
        |\mu_h(f) - \tilde\pi_h(f)| \Le C h .
        \label{eq: rough modified}
    \end{equation}

    It remains to bound $\tilde\pi_h( \tilde f_h - f )$. By \eqref{eq: heat shift integral} with $\psi = f$ and an integration by parts against $\tilde\pi_h \propto e^{-\tilde U_h}$, the linear growth of $\nabla \tilde U_h$ from \eqref{eq: modified class bounds} and the moment bound \eqref{eq: modified target moment} give
    \begin{equation}
        \bigl| \tilde\pi_h( \tilde f_h - f ) \bigr| = \biggl| \int_0^{h/4} \tilde\pi_h\bigl( \nabla \tilde U_h \cdot \nabla ( e^{s\Delta} f ) \bigr)\, \D s \biggr| \Le \frac h4\, \tilde\pi_h\bigl( |\nabla \tilde U_h| \bigr) \Le C h .
        \label{eq: rough shift}
    \end{equation}
    The bounds \eqref{eq: rough modified} and \eqref{eq: rough shift}, with the bound \eqref{eq: shifted bias} on $\tilde\pi_h(\tilde f_h) - \pi(f)$, give $|\mu_h(f) - \pi(f)| \Le C h$.
\end{proof}

The proofs of Theorems~\ref{thm: numerical bias} and \ref{thm: sgld bias} both end with the identification of the coefficient through a Riesz representor $\psi_{\mathcal A}$ of Section~\ref{subsec: riesz}. The next lemma is the common step of the two identifications: it compares $\frac1h\, \pi\bigl( \psi_{\mathcal A}\, ( \omega - e^{h\tilde{\mathcal L}_h} \omega ) \bigr)$ with $\pi( \nabla^2 \omega : \mathcal A )$, and the two proofs apply it to $\omega = \tilde\phi_h$ and to $\omega = e^{h\mathcal L} \phi_h$.

\begin{lemma}
\label{lem: representor semigroup}
    Under Assumptions~\ref{ass: U} and \ref{ass: sg}, let $\mathcal A = \mathcal C$ or $\mathcal A = \nabla^2 U + \mathcal C$. Then $\psi_{\mathcal A}$ satisfies
    \begin{equation}
        \pi\Bigl( \bigl( |\psi_{\mathcal A}| + |\nabla \psi_{\mathcal A}| \bigr) \bigl( 1 + |x| \bigr) \Bigr) \Le C ,
        \label{eq: psi moments}
    \end{equation}
    and, for every $h \in (0,h_0]$ and $\omega \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j \omega}_\infty < \infty$,
    \begin{equation}
        \Bigl| \frac1h\, \pi\bigl( \psi_{\mathcal A}\, ( \omega - e^{h\tilde{\mathcal L}_h} \omega ) \bigr) - \pi\bigl( \nabla^2 \omega : \mathcal A \bigr) \Bigr| \Le C h \sum_{j=1}^{3} \norm{\nabla^j \omega}_\infty .
        \label{eq: representor semigroup}
    \end{equation}
    The bound \eqref{eq: representor semigroup} also holds with $\mathcal L$ in place of $\tilde{\mathcal L}_h$.
\end{lemma}

\begin{proof}
    Since $\nabla^2 U$ is bounded and Lipschitz by Assumption~\ref{ass: U}, and $\mathcal C$ is bounded by \eqref{eq: covariance} and Lipschitz by \eqref{eq: covariance lipschitz}, Lemma~\ref{lem: representor gradient} applies to $\mathcal A$, with $\psi_{\nabla^2 U + \mathcal C} = \psi_{\nabla^2 U} + \psi_{\mathcal C}$ by Theorem~\ref{thm: riesz}. The bounds \eqref{eq: riesz bound} on $\pi( \psi_{\mathcal A}^2 )$ and \eqref{eq: representor gradient} on $\pi( |\nabla \psi_{\mathcal A}|^2 )$, the Cauchy--Schwarz inequality and the moments of $\pi$ in Lemma~\ref{lem: dissipative} then give \eqref{eq: psi moments}.

    Next we prove \eqref{eq: representor semigroup}. By \eqref{eq: backward equation} for $\tilde{\mathcal L}_h$ applied to $\omega$,
    \begin{equation}
        \omega - e^{h\tilde{\mathcal L}_h} \omega = -\int_0^h \tilde{\mathcal L}_h \bigl( e^{s\tilde{\mathcal L}_h} \omega \bigr)\, \D s .
        \label{eq: modified dynkin}
    \end{equation}
    By \eqref{eq: generator} and \eqref{eq: modified generator}, $-\tilde{\mathcal L}_h = -\mathcal L + ( \nabla \tilde U_h - \nabla U ) \cdot \nabla$. Integrating \eqref{eq: modified dynkin} against $\psi_{\mathcal A}\, \pi$, with the identity in \eqref{eq: representor gradient} applied to $e^{s\tilde{\mathcal L}_h} \omega$, gives
    \begin{equation}
        \pi\bigl( \psi_{\mathcal A}\, ( \omega - e^{h\tilde{\mathcal L}_h} \omega ) \bigr) = \int_0^h \Bigl( \underbrace{\pi\bigl( \nabla \psi_{\mathcal A} \cdot \nabla ( e^{s\tilde{\mathcal L}_h} \omega ) \bigr)}_{\mathcal I_1} + \underbrace{\pi\bigl( \psi_{\mathcal A}\, ( \nabla \tilde U_h - \nabla U ) \cdot \nabla ( e^{s\tilde{\mathcal L}_h} \omega ) \bigr)}_{\mathcal I_2} \Bigr) \D s .
        \label{eq: dynkin against psi}
    \end{equation}
    By \eqref{eq: regular decay} for the modified dynamics, for $0 \Le s \Le h$,
    \begin{equation}
        \norm{\nabla ( e^{s\tilde{\mathcal L}_h} \omega )}_\infty \Le C \norm{\nabla \omega}_\infty , \qquad \norm{\nabla^2 ( e^{s\tilde{\mathcal L}_h} \omega )}_\infty + \norm{\nabla^3 ( e^{s\tilde{\mathcal L}_h} \omega )}_\infty \Le C \sum_{j=1}^{3} \norm{\nabla^j \omega}_\infty .
        \label{eq: modified poisson flow}
    \end{equation}
    For $\mathcal I_2$ of \eqref{eq: dynkin against psi}, the bound on $\nabla ( e^{s\tilde{\mathcal L}_h} \omega )$ in \eqref{eq: modified poisson flow}, the bound \eqref{eq: modified class} on $\nabla \tilde U_h - \nabla U$ and \eqref{eq: psi moments} give
    \begin{equation*}
        | \mathcal I_2 | \Le C h \norm{\nabla \omega}_\infty , \qquad 0 \Le s \Le h .
    \end{equation*}
    For the term $\mathcal I_1$ of \eqref{eq: dynkin against psi}, we compare $\nabla ( e^{s\tilde{\mathcal L}_h} \omega )$ with $\nabla \omega$. The identity \eqref{eq: backward equation} for $\tilde{\mathcal L}_h$ applied to $\omega$ gives
    \begin{equation*}
        e^{s\tilde{\mathcal L}_h} \omega - \omega = \int_0^s \tilde{\mathcal L}_h \bigl( e^{r\tilde{\mathcal L}_h} \omega \bigr)\, \D r .
    \end{equation*}
    For $0 \Le r \Le s$, the gradient of the integrand is $\nabla \Delta ( e^{r\tilde{\mathcal L}_h} \omega ) - \nabla^2 ( e^{r\tilde{\mathcal L}_h} \omega )\, \nabla \tilde U_h - \nabla^2 \tilde U_h\, \nabla ( e^{r\tilde{\mathcal L}_h} \omega )$. The bounds \eqref{eq: modified poisson flow}, the linear growth of $\nabla \tilde U_h$ and the bound on $\nabla^2 \tilde U_h$ from \eqref{eq: modified class bounds} then give
    \begin{equation*}
        \bigl| \nabla \bigl( e^{s\tilde{\mathcal L}_h} \omega - \omega \bigr)(x) \bigr| \Le C s \sum_{j=1}^{3} \norm{\nabla^j \omega}_\infty \bigl( 1 + |x| \bigr) , \qquad 0 \Le s \Le h , \quad x \in \mathbb R^d .
    \end{equation*}
    With \eqref{eq: psi moments}, this bound gives
    \begin{equation*}
        \bigl| \mathcal I_1 - \pi\bigl( \nabla \psi_{\mathcal A} \cdot \nabla \omega \bigr) \bigr| \Le C h \sum_{j=1}^{3} \norm{\nabla^j \omega}_\infty , \qquad 0 \Le s \Le h .
    \end{equation*}
    Dividing \eqref{eq: dynkin against psi} by $h$ and inserting the bounds on $\mathcal I_1$ and $\mathcal I_2$, with $\pi( \nabla \psi_{\mathcal A} \cdot \nabla \omega ) = \pi( \nabla^2 \omega : \mathcal A )$ by the identity in \eqref{eq: representor gradient}, prove \eqref{eq: representor semigroup}. With $\mathcal L$ in place of $\tilde{\mathcal L}_h$, the term $\mathcal I_2$ vanishes, and the argument for $\mathcal I_1$ applies with $U$ in place of $\tilde U_h$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm: numerical bias}]
    The proof is accomplished in several steps.\\[6pt]
    \textbf{The coefficient under $\mu_h$.} Since $Y_0$ and $Y_1$ have the law $\mu_h$ and $\mu_h(\tilde f_h) = \pi_h(f)$ by \eqref{eq: heat shift invariant}, taking expectations in the modified discrete Poisson equation \eqref{eq: modified poisson} at $Y_0$ gives
    \begin{equation}
        \pi_h(f) - \pi(f) = \tilde\pi_h(\tilde f_h) - \pi(f) + \frac1h\, \mathbb E\bigl[ \tilde\phi_h(Y_1) - \tilde\phi_h(\tilde Y_0(h)) \bigr] .
        \label{eq: bias identity}
    \end{equation}
    By \eqref{eq: bias identity}, the bound \eqref{eq: shifted bias} on $\tilde\pi_h(\tilde f_h) - \pi(f)$, the bound \eqref{eq: sg weak} at $k = 0$ with $G = \tilde\phi_h$, the bounds \eqref{eq: modified poisson bound} and the moment bound $\mu_h(|y|^{12}) \Le C$ of Lemma~\ref{lem: invariant},
    \begin{equation}
        \Bigl| \pi_h(f) - \pi(f) - \frac h2\, \mathbb E\bigl[ \nabla^2 \tilde\phi_h(Y_0 + D_0) : \mathcal C(Z_0) \bigr] \Bigr| \Le C \norm{\nabla f}_\infty h^2 \log\frac1h ,
        \label{eq: discrete numerical bias}
    \end{equation}
    where the expectation is over $Y_0 \sim \mu_h$.\\[6pt]
    \textbf{From $\mu_h$ to $\pi$.} For $y \in \mathbb R^d$ and $\chi \sim \mathcal N(0, I_d)$, let
    \begin{equation}
        \mathfrak K_h(y) := \mathbb E\biggl[ \nabla^2 \tilde\phi_h\Bigl( y + \sqrt{2h}\, \chi - h\, \nabla U\Bigl( y + \sqrt{\frac h2}\, \chi \Bigr) \Bigr) : \mathcal C\Bigl( y + \sqrt{\frac h2}\, \chi \Bigr) \biggr] ,
        \label{eq: coefficient integrand}
    \end{equation}
    so that the expectation in \eqref{eq: discrete numerical bias} is $\mu_h(\mathfrak K_h)$. The map $y \mapsto y + \sqrt{2h}\, \chi - h\, \nabla U( y + \sqrt{h/2}\, \chi )$ is Lipschitz with the constant $1 + M h \Le 2$, so the bound on the trace in \eqref{eq: covariance}, the Lipschitz bound \eqref{eq: covariance lipschitz} and the bounds \eqref{eq: modified poisson bound} give, for $y, y' \in \mathbb R^d$,
    \begin{equation*}
        |\mathfrak K_h(y) - \mathfrak K_h(y')| \Le \bigl( 2 \Sigma\, \norm{\nabla^3 \tilde\phi_h}_\infty + 4 d M^2 \norm{\nabla^2 \tilde\phi_h}_\infty \bigr) |y - y'| \Le C \norm{\nabla f}_\infty \log\frac1h\, |y - y'| .
    \end{equation*}
    The Kantorovich--Rubinstein duality and Lemma~\ref{lem: rough} then give
    \begin{equation}
        \bigl| \mu_h(\mathfrak K_h) - \pi(\mathfrak K_h) \bigr| \Le C \norm{\nabla f}_\infty h \log\frac1h .
        \label{eq: measure replacement}
    \end{equation}
    Next we prove
    \begin{equation}
        \bigl| \pi(\mathfrak K_h) - \pi\bigl( \nabla^2 \tilde\phi_h : \mathcal C \bigr) \bigr| \Le C \norm{\nabla f}_\infty h \log\frac1h .
        \label{eq: shift removal}
    \end{equation}
    In the proof of \eqref{eq: shift removal}, the last inequality of each bound follows from the bound on the trace in \eqref{eq: covariance}, the Lipschitz bound \eqref{eq: covariance lipschitz}, the bounds \eqref{eq: modified poisson bound} and the moments of $\pi$ in Lemma~\ref{lem: dissipative}. Removing $-h\, \nabla U( y + \sqrt{h/2}\, \chi )$ from the argument of $\nabla^2 \tilde\phi_h$ in \eqref{eq: coefficient integrand} and substituting $x = y + \sqrt{h/2}\, \chi$ give
    \begin{equation}
    \begin{aligned}
        & \biggl| \pi(\mathfrak K_h) - \mathbb E \int_{\mathbb R^d} \pi\Bigl( x - \sqrt{\frac h2}\, \chi \Bigr)\, \nabla^2 \tilde\phi_h\Bigl( x + \sqrt{\frac h2}\, \chi \Bigr) : \mathcal C(x)\, \D x \biggr| \\
        & \qquad \Le \Sigma\, \norm{\nabla^3 \tilde\phi_h}_\infty\, h M \bigl( 1 + \pi(|y|) + \mathbb E |\chi| \bigr) \Le C \norm{\nabla f}_\infty h \log\frac1h .
    \end{aligned}
        \label{eq: shifted integral}
    \end{equation}
    We write the factor of $\mathcal C(x)$ in the integrand of \eqref{eq: shifted integral} as
    \begin{equation*}
        \pi\Bigl( x - \sqrt{\frac h2}\, \chi \Bigr)\, \nabla^2 \tilde\phi_h\Bigl( x + \sqrt{\frac h2}\, \chi \Bigr) = \mathcal T_1 + \mathcal T_2 + \mathcal T_3 ,
    \end{equation*}
    where the terms $\mathcal T_1$ to $\mathcal T_3$ are
    \begin{equation*}
    \begin{gathered}
        \mathcal T_1 = \pi(x)\, \nabla^2 \tilde\phi_h\Bigl( x + \sqrt{\frac h2}\, \chi \Bigr) , \qquad
        \mathcal T_2 = \Bigl( \pi\Bigl( x - \sqrt{\frac h2}\, \chi \Bigr) - \pi(x) \Bigr) \nabla^2 \tilde\phi_h(x) , \\
        \mathcal T_3 = \Bigl( \pi\Bigl( x - \sqrt{\frac h2}\, \chi \Bigr) - \pi(x) \Bigr) \Bigl( \nabla^2 \tilde\phi_h\Bigl( x + \sqrt{\frac h2}\, \chi \Bigr) - \nabla^2 \tilde\phi_h(x) \Bigr) .
    \end{gathered}
    \end{equation*}
    For $\mathcal T_1$, the mean over $\chi$ of $\nabla^2 \tilde\phi_h( x + \sqrt{h/2}\, \chi )$ is $( e^{\frac h4 \Delta} \nabla^2 \tilde\phi_h )(x)$. By \eqref{eq: heat shift integral} with $\psi = \nabla^2 \tilde\phi_h$ and the identity $\nabla ( \mathcal C_{ij}\, \pi ) = ( \nabla \mathcal C_{ij} - \mathcal C_{ij}\, \nabla U )\, \pi$, an integration by parts against $\mathcal C_{ij}\, \pi$ gives
    \begin{equation*}
        \biggl| \mathbb E \int_{\mathbb R^d} \mathcal T_1 : \mathcal C(x)\, \D x - \pi\bigl( \nabla^2 \tilde\phi_h : \mathcal C \bigr) \biggr| \Le \frac h4\, \norm{\nabla^3 \tilde\phi_h}_\infty \sum_{i,j=1}^{d} \pi\bigl( |\nabla \mathcal C_{ij}| + |\mathcal C_{ij}|\, |\nabla U| \bigr) \Le C \norm{\nabla f}_\infty h \log\frac1h .
    \end{equation*}
    For $\mathcal T_2$, the mean over $\chi$ of $\pi( x - \sqrt{h/2}\, \chi ) - \pi(x)$ is $( e^{\frac h4 \Delta} \pi - \pi )(x)$. By \eqref{eq: heat shift integral} with $\psi = \pi$ and the identity $\Delta ( e^{s\Delta} \pi ) = e^{s\Delta} ( V \pi )$ for $V$ of \eqref{eq: V},
    \begin{equation*}
        \biggl| \mathbb E \int_{\mathbb R^d} \mathcal T_2 : \mathcal C(x)\, \D x \biggr| \Le \Sigma\, \norm{\nabla^2 \tilde\phi_h}_\infty\, \frac h4\, \pi\bigl( |V| \bigr) \Le C \norm{\nabla f}_\infty h .
    \end{equation*}
    For $\mathcal T_3$, the identity $\nabla \pi = -\pi\, \nabla U$ and the Lipschitz constant $\norm{\nabla^3 \tilde\phi_h}_\infty$ of $\nabla^2 \tilde\phi_h$ give
    \begin{equation*}
        \biggl| \mathbb E \int_{\mathbb R^d} \mathcal T_3 : \mathcal C(x)\, \D x \biggr| \Le \Sigma\, \norm{\nabla^3 \tilde\phi_h}_\infty\, \frac h2\, \mathbb E |\chi|^2\, \pi\bigl( |\nabla U| \bigr) \Le C \norm{\nabla f}_\infty h \log\frac1h .
    \end{equation*}
    With \eqref{eq: shifted integral}, the bounds on the integrals of $\mathcal T_1$, $\mathcal T_2$ and $\mathcal T_3$ prove \eqref{eq: shift removal}. By \eqref{eq: discrete numerical bias}, \eqref{eq: measure replacement} and \eqref{eq: shift removal},
    \begin{equation}
        \Bigl| \pi_h(f) - \pi(f) - \frac h2\, \pi\bigl( \nabla^2 \tilde\phi_h : \mathcal C \bigr) \Bigr| \Le C \norm{\nabla f}_\infty h^2 \log\frac1h .
        \label{eq: bias under pi}
    \end{equation}
    \textbf{The identification of the coefficient.} By the modified discrete Poisson equation \eqref{eq: modified poisson} and $\pi(\psi_{\mathcal C}) = 0$,
    \begin{equation*}
        \frac1h\, \pi\bigl( \psi_{\mathcal C}\, ( \tilde\phi_h - e^{h\tilde{\mathcal L}_h} \tilde\phi_h ) \bigr) = \pi\bigl( \psi_{\mathcal C}\, \tilde f_h \bigr) .
    \end{equation*}
    The bound \eqref{eq: representor semigroup} of Lemma~\ref{lem: representor semigroup} with $\mathcal A = \mathcal C$ and $\omega = \tilde\phi_h$ and the bounds \eqref{eq: modified poisson bound} then give
    \begin{equation}
        \bigl| \pi\bigl( \nabla^2 \tilde\phi_h : \mathcal C \bigr) - \pi\bigl( \psi_{\mathcal C}\, \tilde f_h \bigr) \bigr| \Le C \norm{\nabla f}_\infty h \log\frac1h .
        \label{eq: coefficient identification}
    \end{equation}

    It remains to bound $\pi( \psi_{\mathcal C}\, ( \tilde f_h - f ) )$. By \eqref{eq: heat shift integral} with $\psi = f$, the identity $\Delta \omega = \mathcal L \omega + \nabla U \cdot \nabla \omega$ and \eqref{eq: representor gradient} with $\omega = e^{s\Delta} f$, the bound $\norm{\nabla ( e^{s\Delta} f )}_\infty \Le \norm{\nabla f}_\infty$ and \eqref{eq: psi moments} give
    \begin{equation*}
        \bigl| \pi\bigl( \psi_{\mathcal C}\, ( \tilde f_h - f ) \bigr) \bigr| = \biggl| \int_0^{h/4} \pi\Bigl( \bigl( \psi_{\mathcal C}\, \nabla U - \nabla \psi_{\mathcal C} \bigr) \cdot \nabla ( e^{s\Delta} f ) \Bigr) \D s \biggr| \Le C \norm{\nabla f}_\infty h .
    \end{equation*}
    With \eqref{eq: coefficient identification} and the identity \eqref{eq: representor poisson} for $\mathcal A = \mathcal C$, the bound on $\pi( \psi_{\mathcal C}\, ( \tilde f_h - f ) )$ gives
    \begin{equation*}
        \bigl| \pi\bigl( \nabla^2 \tilde\phi_h : \mathcal C \bigr) - \pi\bigl( \nabla^2 \phi_0 : \mathcal C \bigr) \bigr| \Le C \norm{\nabla f}_\infty h \log\frac1h ,
    \end{equation*}
    which with \eqref{eq: bias under pi} proves \eqref{eq: numerical bias}. Finally, the bound on the trace in \eqref{eq: covariance} and \eqref{eq: continuous poisson bound} give
    \begin{equation*}
        \frac h2\, \bigl| \pi\bigl( \nabla^2 \phi_0 : \mathcal C \bigr) \bigr| \Le \frac{\Sigma h}{2}\, \norm{\nabla^2 \phi_0}_\infty \Le C \norm{\nabla f}_\infty \Sigma h ,
    \end{equation*}
    which with \eqref{eq: numerical bias} gives $|\pi_h(f) - \pi(f)| \Le C \norm{\nabla f}_\infty ( \Sigma h + h^2 \log\frac1h )$.
\end{proof}

\section{Proofs for the numerical bias comparison with SGLD}
\label{app: sgld}

Throughout this appendix, $(Z_k)_{k \Ge 0}$ is the chain \eqref{eq: sgld} and $\mathcal P_h$ its transition kernel. The constant $C$ depends only on $m$, $M$, $R$ and $d$, and $h \Le h_0$. Since $(\chi_k)_{k \Ge 1}$ is independent of $(\xi_k)_{k \Ge 0}$, the law of the chain is unchanged if its Gaussian increments are taken as $\sqrt{2h}\, \chi_{k+1} = \sqrt2\, (B_{(k+1)h} - B_{kh})$, with $(B_t)_{t \Ge 0}$ a Brownian motion independent of $(\xi_k)_{k \Ge 0}$. For a given $Z_0$, the state $Z_k$ is then $\mathcal F_k$-measurable, with $\mathcal F_k$ and $\mathbb E_k$ of \eqref{eq: filtration}. Statements about a single step are made at the index $0$, with the solution $X_s$ of \eqref{eq: langevin} started from $X_0 = Z_0$ and driven by $(B_t)_{t \Ge 0}$. The step of \eqref{eq: sgld} is then
\begin{equation}
    Z_1 = Z_0 + D_0 - h\, \eta_0 , \qquad D_0 := \sqrt2\, B_h - h\, \nabla U(Z_0) , \qquad \eta_0 := \eta(Z_0;\xi_0) = \hat g(Z_0;\xi_0) - \nabla U(Z_0) ,
    \label{eq: sgld step}
\end{equation}
where $Z_0 + D_0$ is the state after one step of EM from $Z_0$, and $\eta$ is the gradient noise of \eqref{eq: gradient noise}.

\subsection{Invariant distribution of SGLD}
\label{sec: sgld invariant proof}

\begin{lemma}
\label{lem: sgld invariant}
    Under Assumptions~\ref{ass: U} and \ref{ass: sg}, for every $h \in (0,h_0]$, \eqref{eq: sgld} has a unique invariant distribution $\bar\pi_h$, with $\bar\pi_h(|x|^{12}) \Le C$. For every Lipschitz function $f$ and given $Z_0 \in \mathbb R^d$, the chain \eqref{eq: sgld} satisfies
    \begin{equation*}
        \lim_{N \to \infty} \frac1N \sum_{k=0}^{N-1} f(Z_k) = \bar\pi_h(f) \qquad \text{almost surely} .
    \end{equation*}
\end{lemma}

\begin{proof}
    As in the proof of Lemma~\ref{lem: invariant}, we verify the drift condition \eqref{eq: harris drift} and the minorization \eqref{eq: harris minorization} of Harris's ergodic theorem, with $W(x) = |x|^2$. The step of \eqref{eq: sgld} from $Z_k$ is $Z_{k+1} - Z_k = \sqrt{2h}\, \chi_{k+1} - h\, \hat g(Z_k;\xi_k)$. The proof of Lemma~\ref{lem: moments} for \eqref{eq: sglm markov} therefore applies with $Z_k$, $\chi_{k+1}$ and $0$ in place of $Y_k$, $\chi_k$ and the Gaussian shift $\sqrt{h/2}\, \chi_k$. It gives the bound on $\mathbb E_k |Y_{k+1}|^2$ in \eqref{eq: lyapunov} and the bound on $\mathbb E |Y_k|^{12}$ in \eqref{eq: moment bounds} for $Z_k$,
    \begin{equation*}
        \mathbb E_k |Z_{k+1}|^{2} \Le \Bigl( 1 - \frac{m}{8} h \Bigr) |Z_k|^{2} + C h , \qquad \mathbb E |Z_k|^{12} \Le C \bigl( 1 + |Z_0| \bigr)^{12} .
    \end{equation*}
    The bound on $\mathbb E_k |Z_{k+1}|^2$ is the drift condition \eqref{eq: harris drift}, and the condition on the level in \eqref{eq: harris minorization} holds, with $\gamma_h$, $K_h$ and $R_W$ as in Appendix~\ref{sec: invariant proof}, where $C$ is the constant of the bound on $\mathbb E_k |Z_{k+1}|^2$.

    Next we prove the minorization \eqref{eq: harris minorization} on the level set $\{W \Le R_W\}$, with a constant $\alpha_h$ that may depend on $h$. Let $h_0$ also satisfy $4 M h_0 \Le 1$, let $\mathbb B$ be the closed unit ball and $\nu$ the uniform distribution on $\mathbb B$, and fix $Z_0$ with $W(Z_0) \Le R_W$. Given $\xi_0$, the state $Z_1$ has the law $\mathcal N\bigl( Z_0 - h\, \hat g(Z_0;\xi_0) ,\, 2h I_d \bigr)$. By Markov's inequality and the moment bound of Assumption~\ref{ass: sg},
    \begin{equation*}
        \mathbb P_\xi\Bigl( |\hat g(Z_0;\xi)| \Le 4M \bigl( 1 + |Z_0| \bigr) \Bigr) \Ge \frac12 .
    \end{equation*}
    For $\xi_0$ in this event, $4 M h_0 \Le 1$ gives $| x - Z_0 + h\, \hat g(Z_0;\xi_0) | \Le 2 ( 1 + R_W^{\frac12} )$ for $x \in \mathbb B$, so the density of $Z_1$ given $\xi_0$ on $\mathbb B$ is at least a constant $\beta_h > 0$ that does not depend on $Z_0$. Hence, for every Borel set $A$,
    \begin{equation*}
        \mathcal P_h(Z_0, A) \Ge \frac{\beta_h}{2}\, \mathrm{Leb}(A \cap \mathbb B) = \frac{\beta_h\, \mathrm{Leb}(\mathbb B)}{2}\, \nu(A) ,
    \end{equation*}
    which is the minorization \eqref{eq: harris minorization} with $\alpha_h := \beta_h\, \mathrm{Leb}(\mathbb B) / 2 \in (0,1)$. Harris's ergodic theorem therefore gives the unique invariant distribution $\bar\pi_h$ and the limit \eqref{eq: harris} with $\bar\pi_h$ in place of $\mu_h$.

    For the moment bound of $\bar\pi_h$, as in the proof of Lemma~\ref{lem: invariant}, the limit \eqref{eq: harris} with the truncation $\min\{ |x|^{12}, \ell \}$ for $\ell \Ge 1$ and the bound on $\mathbb E |Z_k|^{12}$ for the chain started from $Z_0 = 0$ give $\bar\pi_h(|x|^{12}) \Le C$. Finally, by the drift condition \eqref{eq: harris drift} and the minorization \eqref{eq: harris minorization}, the chain is positive Harris recurrent, and the strong law of large numbers for positive Harris recurrent chains \citep[Theorem~17.0.1(i)]{meyn1993markov} gives the limit of time averages of $f$ for every given $Z_0$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm: sgld bias}: numerical bias of SGLD}
\label{sec: sgld bias proof}

The proofs of Lemmas~\ref{lem: sgld weak} and \ref{lem: sgld lipschitz weak} use the bound \eqref{eq: sgld noise} on the gradient noise, the sizes \eqref{eq: sgld sizes} of the step and the expansion \eqref{eq: em local error expansion} of the local error of EM, which we prove first.

Let $\psi$ be Lipschitz. Since $\sqrt2\, B_h \sim \mathcal N(0, 2h I_d)$ is independent of $\mathcal F_0$ and $\xi_0$, the step \eqref{eq: sgld step} gives
\begin{equation*}
\begin{aligned}
    \mathbb E\bigl[ \psi(Z_1) \,\big|\, \mathcal F_0, \xi_0 \bigr] & = ( e^{h\Delta} \psi )\bigl( Z_0 - h\, \nabla U(Z_0) - h\, \eta_0 \bigr) , \\
    \mathbb E_0\bigl[ \psi(Z_0 + D_0) \bigr] & = ( e^{h\Delta} \psi )\bigl( Z_0 - h\, \nabla U(Z_0) \bigr) .
\end{aligned}
\end{equation*}
Since $\xi_0$ is independent of $\mathcal F_0$, Assumption~\ref{ass: sg}, the definition \eqref{eq: covariance} of $\mathcal C$ and the linear growth of $\nabla U$ give
\begin{equation*}
    \mathbb E_0[\eta_0] = 0 , \qquad \mathbb E_0[\eta_0 \otimes \eta_0] = \mathcal C(Z_0) , \qquad \mathbb E_0 |\eta_0|^3 \Le C \bigl( 1 + |Z_0| \bigr)^3 .
\end{equation*}
Taylor's formula for $e^{h\Delta} \psi$ at $Z_0 - h\, \nabla U(Z_0)$ to the order $2$ in $h\eta_0$ then gives
\begin{equation*}
    \Bigl| \mathbb E_0\bigl[ \psi(Z_1) - \psi(Z_0 + D_0) \bigr] - \frac{h^2}{2}\, \nabla^2 ( e^{h\Delta} \psi )\bigl( Z_0 - h\, \nabla U(Z_0) \bigr) : \mathcal C(Z_0) \Bigr| \Le C h^3 \norm{\nabla^3 ( e^{h\Delta} \psi )}_\infty \bigl( 1 + |Z_0| \bigr)^3 .
\end{equation*}
Since $\nabla^2 ( e^{h\Delta} \psi )$ is Lipschitz with the constant $\norm{\nabla^3 ( e^{h\Delta} \psi )}_\infty$, the bound on the trace in \eqref{eq: covariance} and the linear growth of $\nabla U$ then give
\begin{equation}
    \Bigl| \mathbb E_0\bigl[ \psi(Z_1) - \psi(Z_0 + D_0) \bigr] - \frac{h^2}{2}\, \nabla^2 ( e^{h\Delta} \psi )(Z_0) : \mathcal C(Z_0) \Bigr| \Le C h^3 \norm{\nabla^3 ( e^{h\Delta} \psi )}_\infty \bigl( 1 + |Z_0| \bigr)^3 .
    \label{eq: sgld noise}
\end{equation}

By \eqref{eq: langevin} and \eqref{eq: sgld step}, the local error of EM is
\begin{equation*}
    Z_0 + D_0 - X_h = \int_0^h \bigl( \nabla U(X_s) - \nabla U(Z_0) \bigr) \D s .
\end{equation*}
Lemma~\ref{lem: langevin moments} holds with $Z_0$, $X_s$ and $\sqrt2\, B_h$ in place of $Y_0$, $Y_0(s)$ and $\sqrt{2h}\, \chi_0$. Assumption~\ref{ass: U} and the bounds \eqref{eq: langevin sizes} then give, with $\pnorm{\cdot}_4$ taken under $\mathbb E_0$,
\begin{equation}
    \pnorm{Z_0 + D_0 - X_h}_4 \Le C h^{\frac32} \bigl( 1 + |Z_0| \bigr) , \qquad \pnorm{D_0}_4 + \pnorm{X_h - Z_0}_4 \Le C \sqrt h \bigl( 1 + |Z_0| \bigr) .
    \label{eq: sgld sizes}
\end{equation}
As in \eqref{eq: leading term}, with $Z_0$ in place of $Y_0$, define
\begin{equation}
    \mathfrak g := \sqrt2\, \nabla^2 U(Z_0) \int_0^h \Bigl( B_s - \frac12 B_h \Bigr) \D s ,
    \label{eq: em local error leading}
\end{equation}
which is centered and independent of $B_h$ by Lemma~\ref{lem: sizes}. As in the proof of the expansion \eqref{eq: local error expansion}, Taylor's formula for $\nabla U$ at $Z_0$, with $X_s - Z_0 = \sqrt2\, B_s - \int_0^s \nabla U(X_q)\, \D q$ and the bounds \eqref{eq: langevin sizes}, gives
\begin{equation}
    \Bigl\| Z_0 + D_0 - X_h - \frac{h}{\sqrt2}\, \nabla^2 U(Z_0)\, B_h - \mathfrak g \Bigr\|_4 \Le C h^2 \bigl( 1 + |Z_0| \bigr)^2 .
    \label{eq: em local error expansion}
\end{equation}

The next two lemmas give the local weak error of \eqref{eq: sgld}, the first for a function $G$ with bounded derivatives of orders $1$ to $4$ and the second for a Lipschitz function $f$. The proof of Theorem~\ref{thm: sgld bias} applies the first lemma to $e^{h\mathcal L} \phi_h$ and the second to $f$. The first lemma is the counterpart for \eqref{eq: sgld} of Lemma~\ref{lem: sg weak}, with \eqref{eq: langevin} in place of the modified dynamics \eqref{eq: modified langevin}, and gives the terms of order $h^2$ of the local weak error. These terms contain the derivatives of $e^{h\Delta} G$, with $e^{h\Delta}$ the heat semigroup of the Gaussian increment $\sqrt2\, B_h$.

\begin{lemma}
\label{lem: sgld weak}
    Under Assumptions~\ref{ass: U} and \ref{ass: sg}, for every $h \in (0,h_0]$, $G \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j G}_\infty < \infty$ and $k \Ge 0$, the chain \eqref{eq: sgld} satisfies
    \begin{align}
        & \biggl| \mathbb E_k\bigl[ G(Z_{k+1}) \bigr] - (e^{h\mathcal L} G)(Z_k) - \frac{h^2}{2}\, \nabla^2 ( e^{h\Delta} G )(Z_k) : \bigl( 2\nabla^2 U(Z_k) + \mathcal C(Z_k) \bigr) \\
        \label{eq: sgld weak}
        & + \frac{h^2}{2}\, \nabla ( e^{h\Delta} G )(Z_k) \cdot \bigl( \nabla^2 U(Z_k)\, \nabla U(Z_k) - \nabla \Delta U(Z_k) \bigr) \biggr| \Le C \Bigl( h^3 \sum_{j=1}^{3} \norm{\nabla^j G}_\infty + h^{\frac72} \norm{\nabla^4 G}_\infty \Bigr) \bigl( 1 + |Z_k| \bigr)^{6} . \notag
    \end{align}
\end{lemma}

\begin{proof}
    Work at the index $0$. We first prove, for the function $G$ of \eqref{eq: sgld weak},
    \begin{align}
        & \Bigl| \mathbb E_0\bigl[ G(Z_0 + D_0) - G(X_h) \bigr] - h^2\, \nabla^2 ( e^{h\Delta} G )(Z_0) : \nabla^2 U(Z_0) + \frac{h^2}{2}\, \nabla G(Z_0) \cdot \bigl( \nabla^2 U(Z_0)\, \nabla U(Z_0) - \nabla \Delta U(Z_0) \bigr) \Bigr| \notag \\
        & \quad \Le C \Bigl( h^3 \sum_{j=1}^{3} \norm{\nabla^j G}_\infty + h^{\frac72} \norm{\nabla^4 G}_\infty \Bigr) \bigl( 1 + |Z_0| \bigr)^6 .
        \label{eq: sgld em weak}
    \end{align}
    As in the proof of Lemma~\ref{lem: sg weak}, the expansion \eqref{eq: G expansion full} holds with $Z_0$, the step $D_0$ of \eqref{eq: sgld step}, $X_h$ and $X_h - Z_0$ in place of $Y_0$, the step $D_0$ of \eqref{eq: full gradient step}, $\tilde Y_0(h)$ and $\tilde W_h$, where
    \begin{equation*}
        Q_\varsigma = \varsigma (Z_0 + D_0) + (1 - \varsigma) X_h , \qquad \varrho_4 = \int_0^1 \bigl\langle \varrho_3(Q_\varsigma - Z_0),\, Z_0 + D_0 - X_h \bigr\rangle \,\D\varsigma .
    \end{equation*}
    With $\sqrt2\, B_h$ in place of $\sqrt{2h}\, \chi_0$ and $\mathfrak g$ of \eqref{eq: em local error leading}, the terms $\mathcal T_1$ to $\mathcal T_3$ of \eqref{eq: remainder split} are
    \begin{equation*}
    \begin{gathered}
        \mathcal T_1 = \bigl\langle \varrho_3(\sqrt2\, B_h),\, \mathfrak g \bigr\rangle , \qquad
        \mathcal T_2 = \bigl\langle \varrho_3(\sqrt2\, B_h),\, Z_0 + D_0 - X_h - \mathfrak g \bigr\rangle , \\
        \mathcal T_3 = \bigl\langle \varrho_3(Q_\varsigma - Z_0) - \varrho_3(\sqrt2\, B_h),\, Z_0 + D_0 - X_h \bigr\rangle .
    \end{gathered}
    \end{equation*}
    As in the proof of Lemma~\ref{lem: sg weak}, $\mathbb E_0[\mathcal T_1] = 0$, and the bounds \eqref{eq: sgld sizes} and \eqref{eq: langevin sizes} give $\bigl| \mathbb E_0[\mathcal T_3] \bigr| \Le C \norm{\nabla^4 G}_\infty\, h^{\frac72} ( 1 + |Z_0| )^4$. For $\mathcal T_2$, the bound on $\varrho_3$ at $v = \sqrt2\, B_h$, H\"older's inequality and the expansion \eqref{eq: em local error expansion} give
    \begin{equation*}
        \Bigl| \mathbb E_0[\mathcal T_2] - \frac{h}{\sqrt2}\, \mathbb E_0 \bigl\langle \varrho_3(\sqrt2\, B_h),\, \nabla^2 U(Z_0)\, B_h \bigr\rangle \Bigr| \Le C \norm{\nabla^4 G}_\infty\, h^{\frac72} \bigl( 1 + |Z_0| \bigr)^2 .
    \end{equation*}
    Integrating \eqref{eq: remainder split} over $\varsigma \in [0,1]$ then gives
    \begin{equation}
        \Bigl| \mathbb E_0[\varrho_4] - \frac{h}{\sqrt2}\, \mathbb E_0 \bigl\langle \varrho_3(\sqrt2\, B_h),\, \nabla^2 U(Z_0)\, B_h \bigr\rangle \Bigr| \Le C \norm{\nabla^4 G}_\infty\, h^{\frac72} \bigl( 1 + |Z_0| \bigr)^4 .
        \label{eq: em remainder mean}
    \end{equation}
    By the definition of $\varrho_3$,
    \begin{equation*}
        \varrho_3(\sqrt2\, B_h) = \nabla G(Z_0 + \sqrt2\, B_h) - \nabla G(Z_0) - \sqrt2\, \nabla^2 G(Z_0)\, B_h - \nabla^3 G(Z_0)\bigl[ B_h^{\otimes 2} \bigr] ,
    \end{equation*}
    and the Gaussian integration by parts in $B_h$ gives
    \begin{equation*}
        \mathbb E_0\bigl[ \nabla G(Z_0 + \sqrt2\, B_h) \otimes B_h \bigr] = \sqrt2\, h\, \mathbb E_0\bigl[ \nabla^2 G(Z_0 + \sqrt2\, B_h) \bigr] = \sqrt2\, h\, \nabla^2 ( e^{h\Delta} G )(Z_0) .
    \end{equation*}
    Since $\mathbb E_0[ B_h^{\otimes 2} ] = h I_d$ and the terms of odd degree in $B_h$ of $\bigl\langle \varrho_3(\sqrt2\, B_h),\, \nabla^2 U(Z_0)\, B_h \bigr\rangle$ have mean $0$,
    \begin{equation}
        \frac{h}{\sqrt2}\, \mathbb E_0 \bigl\langle \varrho_3(\sqrt2\, B_h),\, \nabla^2 U(Z_0)\, B_h \bigr\rangle = h^2 \Bigl( \nabla^2 ( e^{h\Delta} G )(Z_0) - \nabla^2 G(Z_0) \Bigr) : \nabla^2 U(Z_0) .
        \label{eq: em gaussian ibp}
    \end{equation}
    Taking $\mathbb E_0$ in the expansion \eqref{eq: G expansion full} and inserting the bound \eqref{eq: em remainder mean} and the identity \eqref{eq: em gaussian ibp} give
    \begin{equation}
    \begin{aligned}
        & \biggl| \mathbb E_0\bigl[ G(Z_0 + D_0) - G(X_h) \bigr] - \sum_{j=1}^{3} \frac{1}{j!}\, \nabla^j G(Z_0) : \mathbb E_0\Bigl[ D_0^{\otimes j} - (X_h - Z_0)^{\otimes j} \Bigr] \\
        & \quad - h^2 \Bigl( \nabla^2 ( e^{h\Delta} G )(Z_0) - \nabla^2 G(Z_0) \Bigr) : \nabla^2 U(Z_0) \biggr| \Le C \norm{\nabla^4 G}_\infty\, h^{\frac72} \bigl( 1 + |Z_0| \bigr)^4 .
    \end{aligned}
        \label{eq: sgld em expansion}
    \end{equation}
    With all derivatives of $U$ evaluated at $Z_0$ and $1 \Le i,j,l \Le d$, the moments of $D_0$ in \eqref{eq: sgld em expansion} are
    \begin{equation*}
    \begin{gathered}
        \mathbb E_0[D_0] = -h\, \nabla U , \qquad \mathbb E_0\bigl[ D_0^{\otimes 2} \bigr] = 2h\, I_d + h^2\, \nabla U^{\otimes 2} , \\
        \Bigl( \mathbb E_0\bigl[ D_0^{\otimes 3} \bigr] \Bigr)_{ijl} = -2h^2 \bigl( \delta_{ij}\, \partial_l U + \delta_{il}\, \partial_j U + \delta_{jl}\, \partial_i U \bigr) - h^3\, \partial_i U\, \partial_j U\, \partial_l U .
    \end{gathered}
    \end{equation*}
    The moments of $X_h - Z_0$ are expanded in \eqref{eq: langevin m1}--\eqref{eq: langevin m3}.\\[6pt]
    \textbf{Moment of order $1$}. The terms $-h\, \nabla U$ cancel, and the term of order $h^2$ in \eqref{eq: langevin m1} remains with the opposite sign, so that
    \begin{equation}
        \Bigl| \mathbb E_0\bigl[ D_0 - (X_h - Z_0) \bigr] + \frac{h^2}{2} \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) \Bigr| \Le C h^3 \bigl( 1 + |Z_0| \bigr)^3 .
        \label{eq: sgld m1}
    \end{equation}
    \textbf{Moment of order $2$}. The terms $2h\, I_d$ and $h^2\, \nabla U^{\otimes 2}$ cancel, and the term $-2h^2\, \nabla^2 U$ in \eqref{eq: langevin m2} remains with the opposite sign, so that
    \begin{equation}
        \Bigl\| \mathbb E_0\bigl[ D_0^{\otimes 2} - (X_h - Z_0)^{\otimes 2} \bigr] - 2h^2\, \nabla^2 U \Bigr\| \Le C h^3 \bigl( 1 + |Z_0| \bigr)^4 .
        \label{eq: sgld m2}
    \end{equation}
    \textbf{Moment of order $3$}. The terms of order $h^2$ cancel, and the term $-h^3\, \nabla U^{\otimes 3}$ of $\mathbb E_0\bigl[ D_0^{\otimes 3} \bigr]$ satisfies $\norm{h^3\, \nabla U^{\otimes 3}} \Le C h^3 \bigl( 1 + |Z_0| \bigr)^3$ by the linear growth of $\nabla U$, so that
    \begin{equation}
        \Bigl\| \mathbb E_0\bigl[ D_0^{\otimes 3} - (X_h - Z_0)^{\otimes 3} \bigr] \Bigr\| \Le C h^3 \bigl( 1 + |Z_0| \bigr)^6 .
        \label{eq: sgld m3}
    \end{equation}
    Inserting \eqref{eq: sgld m1}--\eqref{eq: sgld m3} into \eqref{eq: sgld em expansion}, where the terms with $\nabla^2 G(Z_0) : \nabla^2 U(Z_0)$ cancel, proves \eqref{eq: sgld em weak}. The identity \eqref{eq: heat shift integral}, with $h$ in place of $\frac h4$ and $\psi = \nabla G$, gives $\norm{\nabla ( e^{h\Delta} G ) - \nabla G}_\infty \Le d\, h \norm{\nabla^3 G}_\infty$. With the linear growth of $\nabla^2 U\, \nabla U - \nabla \Delta U$, the bound \eqref{eq: sgld em weak} therefore holds with $\nabla ( e^{h\Delta} G )(Z_0)$ in place of $\nabla G(Z_0)$. Since $\mathbb E_0[G(X_h)] = (e^{h\mathcal L} G)(Z_0)$ and $\norm{\nabla^3 ( e^{h\Delta} G )}_\infty \Le \norm{\nabla^3 G}_\infty$, adding this form of \eqref{eq: sgld em weak} and \eqref{eq: sgld noise} for $\psi = G$ proves \eqref{eq: sgld weak} at the index $0$, and by time homogeneity at every $k$.
\end{proof}

The next lemma is the counterpart of Lemma~\ref{lem: sgld weak} for a Lipschitz function $f$ in place of $G$. The Hessian $\nabla^2 ( e^{h\Delta} f )$ is defined for such $f$, and the term with the Hessian in \eqref{eq: sgld weak} remains, with an error of order $h^2$.

\begin{lemma}
\label{lem: sgld lipschitz weak}
    Under Assumptions~\ref{ass: U} and \ref{ass: sg}, for every $h \in (0,h_0]$, Lipschitz function $f$ and $k \Ge 0$, the chain \eqref{eq: sgld} satisfies
    \begin{equation}
        \biggl| \mathbb E_k\bigl[ f(Z_{k+1}) \bigr] - (e^{h\mathcal L} f)(Z_k) - \frac{h^2}{2}\, \nabla^2 ( e^{h\Delta} f )(Z_k) : \bigl( 2\nabla^2 U(Z_k) + \mathcal C(Z_k) \bigr) \biggr| \Le C h^2 \norm{\nabla f}_\infty \bigl( 1 + |Z_k| \bigr)^{3} .
        \label{eq: sgld lipschitz weak}
    \end{equation}
\end{lemma}

\begin{proof}
    Work at the index $0$. Since $e^{h\Delta} f = e^{\frac h4 \Delta} ( e^{\frac{3h}{4} \Delta} f )$ and $e^{\frac{3h}{4} \Delta} f$ is Lipschitz with the constant $\norm{\nabla f}_\infty$, the bounds \eqref{eq: heat derivatives} give
    \begin{equation}
        \norm{\nabla^j ( e^{h\Delta} f )}_\infty \Le C \norm{\nabla f}_\infty h^{-\frac{j-1}{2}} , \qquad j = 1, 2, 3 .
        \label{eq: gaussian derivatives}
    \end{equation}
    The bound \eqref{eq: sgld noise} for $\psi = f$, with the bound \eqref{eq: gaussian derivatives} at $j = 3$, gives
    \begin{equation}
        \Bigl| \mathbb E_0\bigl[ f(Z_1) - f(Z_0 + D_0) \bigr] - \frac{h^2}{2}\, \nabla^2 ( e^{h\Delta} f )(Z_0) : \mathcal C(Z_0) \Bigr| \Le C h^2 \norm{\nabla f}_\infty \bigl( 1 + |Z_0| \bigr)^3 .
        \label{eq: sgld lipschitz noise}
    \end{equation}
    To compare $Z_0 + D_0$ with $X_h$, with $\mathfrak g$ of \eqref{eq: em local error leading}, let
    \begin{equation*}
        \Lambda_0 := Z_0 + D_0 - \frac{h}{\sqrt2}\, \nabla^2 U(Z_0)\, B_h - \mathfrak g = Z_0 - h\, \nabla U(Z_0) + \sqrt2\, B_h - \sqrt2\, \nabla^2 U(Z_0) \int_0^h B_s\, \D s .
    \end{equation*}
    By the expansion \eqref{eq: em local error expansion},
    \begin{equation}
        \bigl| \mathbb E_0[ f(X_h) - f(\Lambda_0) ] \bigr| \Le C h^2 \norm{\nabla f}_\infty \bigl( 1 + |Z_0| \bigr)^2 .
        \label{eq: sgld lambda replacement}
    \end{equation}
    Given $\mathcal F_0$, the vectors $Z_0 + D_0$ and $\Lambda_0$ are Gaussian with the mean $Z_0 - h\, \nabla U(Z_0)$. The covariance of $Z_0 + D_0$ is $2h I_d$, and the covariance of $\Lambda_0$ is
    \begin{equation*}
        \Gamma := 2h I_d - 2h^2\, \nabla^2 U(Z_0) + \frac{2h^3}{3} \bigl( \nabla^2 U(Z_0) \bigr)^2 .
    \end{equation*}
    Let $h_0$ also satisfy $3 M h_0 \Le 1$, so that $\norm{\Gamma - 2h I_d} \Le 3 M h^2 \Le h$. For $\theta \in [0,1]$, let
    \begin{equation*}
        \Gamma_\theta := 2h I_d + \theta ( \Gamma - 2h I_d ) \succeq h I_d , \qquad F_\theta := f * \mathcal N(0, \Gamma_\theta) ,
    \end{equation*}
    so that $F_0 = e^{h\Delta} f$ and
    \begin{equation*}
        \mathbb E_0\bigl[ f(\Lambda_0) - f(Z_0 + D_0) \bigr] = \bigl( F_1 - e^{h\Delta} f \bigr)\bigl( Z_0 - h\, \nabla U(Z_0) \bigr) .
    \end{equation*}
    Since $F_\theta = e^{\frac h4 \Delta} \bigl( f * \mathcal N(0, \Gamma_\theta - \frac h2 I_d) \bigr)$, the bound \eqref{eq: heat derivatives} at $j = 4$ gives $\norm{\nabla^4 F_\theta}_\infty \Le C h^{-\frac32} \norm{\nabla f}_\infty$. The derivatives of $F_\theta$ in $\theta$ are
    \begin{equation*}
        \partial_\theta F_\theta = \frac12\, \nabla^2 F_\theta : ( \Gamma - 2h I_d ) , \qquad \partial_\theta^2 F_\theta = \frac14\, \nabla^4 F_\theta : ( \Gamma - 2h I_d )^{\otimes 2} ,
    \end{equation*}
    so Taylor's formula in $\theta$ gives
    \begin{equation}
        \Bigl\| F_1 - e^{h\Delta} f - \frac12\, \nabla^2 ( e^{h\Delta} f ) : ( \Gamma - 2h I_d ) \Bigr\|_\infty \Le C h^{-\frac32} \norm{\nabla f}_\infty\, \norm{\Gamma - 2h I_d}^2 \Le C h^{\frac52} \norm{\nabla f}_\infty .
        \label{eq: sgld covariance interpolation}
    \end{equation}
    By the definition of $\Gamma$, the bounds \eqref{eq: gaussian derivatives} at $j = 2$ and $j = 3$, the linear growth of $\nabla U$ and the bound on $\nabla^2 U$ from Assumption~\ref{ass: U},
    \begin{equation}
        \Bigl| \frac12\, \nabla^2 ( e^{h\Delta} f )\bigl( Z_0 - h\, \nabla U(Z_0) \bigr) : ( \Gamma - 2h I_d ) + h^2\, \nabla^2 ( e^{h\Delta} f )(Z_0) : \nabla^2 U(Z_0) \Bigr| \Le C h^2 \norm{\nabla f}_\infty \bigl( 1 + |Z_0| \bigr) .
        \label{eq: sgld covariance leading}
    \end{equation}
    With the identity for $\mathbb E_0[ f(\Lambda_0) - f(Z_0 + D_0) ]$, the bounds \eqref{eq: sgld lambda replacement}--\eqref{eq: sgld covariance leading} give
    \begin{equation}
        \Bigl| \mathbb E_0\bigl[ f(Z_0 + D_0) - f(X_h) \bigr] - h^2\, \nabla^2 ( e^{h\Delta} f )(Z_0) : \nabla^2 U(Z_0) \Bigr| \Le C h^2 \norm{\nabla f}_\infty \bigl( 1 + |Z_0| \bigr)^2 .
        \label{eq: sgld lipschitz em}
    \end{equation}
    Since $\mathbb E_0[f(X_h)] = (e^{h\mathcal L} f)(Z_0)$, adding \eqref{eq: sgld lipschitz noise} and \eqref{eq: sgld lipschitz em} proves \eqref{eq: sgld lipschitz weak} at the index $0$, and by time homogeneity at every $k$.
\end{proof}

The factor $\nabla^2 U\, \nabla U - \nabla \Delta U$ of \eqref{eq: sgld weak} grows linearly. For a function with such a factor, the next lemma bounds the difference of its expectations under two measures, and with this bound the proof of Theorem~\ref{thm: sgld bias} moves an expectation from $\bar\pi_h$ to $\pi$.

\begin{lemma}
\label{lem: local lipschitz}
    Let $\mu$ and $\nu$ be probability measures on $\mathbb R^d$ with $\mu(|x|^{12}) + \nu(|x|^{12}) < \infty$ and $\mathcal W_1(\mu, \nu) \Le 1$, and let $F : \mathbb R^d \to \mathbb R$ satisfy, for some $K \Ge 0$,
    \begin{equation}
        |F(x) - F(x')| \Le K \bigl( 1 + |x| + |x'| \bigr) |x - x'| , \qquad x, x' \in \mathbb R^d .
        \label{eq: local lipschitz}
    \end{equation}
    Then
    \begin{equation*}
        |\mu(F) - \nu(F)| \Le 3 K \bigl( 1 + \mu(|x|^{12}) + \nu(|x|^{12}) \bigr)\, \mathcal W_1(\mu, \nu)^{\frac{10}{11}} .
    \end{equation*}
\end{lemma}

\begin{proof}
    If $\mathcal W_1(\mu,\nu) = 0$, then $\mu = \nu$. Otherwise, let $\rho := \mathcal W_1(\mu,\nu)^{-\frac1{11}} \Ge 1$, and let $T_\rho$ be the retraction onto the closed ball of radius $\rho$, with $T_\rho(x) := x$ for $|x| \Le \rho$ and $T_\rho(x) := \rho x / |x|$ for $|x| > \rho$. The map $T_\rho$ is Lipschitz with the constant $1$. By the condition \eqref{eq: local lipschitz}, $F \circ T_\rho$ is Lipschitz with the constant $K(1 + 2\rho) \Le 3K\rho$, so the Kantorovich--Rubinstein duality gives
    \begin{equation*}
        \bigl| \mu(F \circ T_\rho) - \nu(F \circ T_\rho) \bigr| \Le 3 K \rho\, \mathcal W_1(\mu,\nu) .
    \end{equation*}
    For $|x| > \rho \Ge 1$, the condition \eqref{eq: local lipschitz} with $x' = T_\rho(x)$ gives
    \begin{equation*}
        |F(x) - F(T_\rho(x))| \Le K \bigl( 1 + 2|x| \bigr) |x| \Le 3 K |x|^2 \Le 3 K \rho^{-10} |x|^{12} ,
    \end{equation*}
    while $F = F \circ T_\rho$ on $\{|x| \Le \rho\}$. Hence
    \begin{equation*}
        |\mu(F) - \nu(F)| \Le 3 K \rho\, \mathcal W_1(\mu,\nu) + 3 K \rho^{-10} \bigl( \mu(|x|^{12}) + \nu(|x|^{12}) \bigr) = 3 K \bigl( 1 + \mu(|x|^{12}) + \nu(|x|^{12}) \bigr)\, \mathcal W_1(\mu,\nu)^{\frac{10}{11}} .
    \end{equation*}
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm: sgld bias}]
    The proof is accomplished in several steps.\\[6pt]
    \textbf{The discrete form \eqref{eq: sgld bias}.} We first take $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$. Then $\phi_h$ and $e^{h\mathcal L} \phi_h$ are in $C^4(\mathbb R^d)$, with bounded derivatives of orders $1$ to $4$, by \eqref{eq: regular decay}. Let $Z_0 \sim \bar\pi_h$ be independent of $(B_t)_{t \Ge 0}$ and $(\xi_k)_{k \Ge 0}$, so that $Z_1 \sim \bar\pi_h$. With $Z_0$ included in $\mathcal F_0$ of \eqref{eq: filtration}, $\mathbb E_0[\phi_h(X_h)] = (e^{h\mathcal L} \phi_h)(Z_0)$, and taking expectations in the discrete Poisson equation \eqref{eq: discrete poisson} at $Z_0$ gives
    \begin{equation}
        \bar\pi_h(f) - \pi(f) = \frac1h\, \mathbb E\bigl[ \phi_h(Z_1) - \phi_h(X_h) \bigr] .
        \label{eq: sgld bias identity}
    \end{equation}
    We split $\phi_h = h\, ( f - \pi(f) ) + e^{h\mathcal L} \phi_h$ and apply the bound \eqref{eq: sgld lipschitz weak} at $k = 0$ to $f$ and the bound \eqref{eq: sgld weak} at $k = 0$ with $G = e^{h\mathcal L} \phi_h$. With $\norm{\nabla ( e^{h\Delta} f )}_\infty \Le \norm{\nabla f}_\infty$ and the linear growth of $\nabla^2 U\, \nabla U - \nabla \Delta U$ for the term with $h\, \nabla ( e^{h\Delta} f )$ in $\nabla ( e^{h\Delta} \phi_h )$, the bounds \eqref{eq: poisson bound} and the moment bound $\bar\pi_h(|x|^{12}) \Le C$ of Lemma~\ref{lem: sgld invariant} then give
    \begin{equation}
    \begin{aligned}
        & \biggl| \mathbb E\bigl[ \phi_h(Z_1) - \phi_h(X_h) \bigr] - \frac{h^2}{2}\, \bar\pi_h\Bigl( \nabla^2 ( e^{h\Delta} \phi_h ) : \bigl( 2\nabla^2 U + \mathcal C \bigr) \Bigr) \\
        & \quad + \frac{h^2}{2}\, \bar\pi_h\Bigl( \nabla ( e^{h\Delta} \phi_h ) \cdot \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) \Bigr) \biggr| \Le C \norm{\nabla f}_\infty h^3 \log\frac1h .
    \end{aligned}
        \label{eq: sgld discrete step}
    \end{equation}
    Dividing \eqref{eq: sgld discrete step} by $h$ and inserting \eqref{eq: sgld bias identity} prove \eqref{eq: sgld bias}.\\[6pt]
    \textbf{From $\bar\pi_h$ to $\pi$.} By the bounds \eqref{eq: gaussian derivatives} for $h f$ and \eqref{eq: poisson bound} for $e^{h\mathcal L} \phi_h$,
    \begin{equation}
        \norm{\nabla ( e^{h\Delta} \phi_h )}_\infty + \norm{\nabla^2 ( e^{h\Delta} \phi_h )}_\infty \Le C \norm{\nabla f}_\infty , \qquad \norm{\nabla^3 ( e^{h\Delta} \phi_h )}_\infty \Le C \norm{\nabla f}_\infty \log\frac1h .
        \label{eq: smoothed poisson bound}
    \end{equation}
    Next we bound $\mathcal W_1(\bar\pi_h, \pi)$. The bounds \eqref{eq: smoothed poisson bound}, the bound on the trace in \eqref{eq: covariance}, the linear growth of $\nabla^2 U\, \nabla U - \nabla \Delta U$ and the moment bound $\bar\pi_h(|x|^{12}) \Le C$ of Lemma~\ref{lem: sgld invariant} give
    \begin{equation*}
        \Bigl| \bar\pi_h\Bigl( \nabla^2 ( e^{h\Delta} \phi_h ) : \bigl( 2\nabla^2 U + \mathcal C \bigr) \Bigr) \Bigr| + \Bigl| \bar\pi_h\Bigl( \nabla ( e^{h\Delta} \phi_h ) \cdot \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) \Bigr) \Bigr| \Le C \norm{\nabla f}_\infty ,
    \end{equation*}
    and the discrete form \eqref{eq: sgld bias} gives $|\bar\pi_h(f) - \pi(f)| \Le C \norm{\nabla f}_\infty h$. The Kantorovich--Rubinstein duality and a mollification then give
    \begin{equation}
        \mathcal W_1(\bar\pi_h, \pi) \Le C h .
        \label{eq: w1 rough sgld}
    \end{equation}
    By the bounds \eqref{eq: smoothed poisson bound}, the bounds on $\nabla^2 U$ and $\nabla^3 U$ from Assumption~\ref{ass: U}, the bound on the trace in \eqref{eq: covariance} and the Lipschitz bound \eqref{eq: covariance lipschitz}, the function $\nabla^2 ( e^{h\Delta} \phi_h ) : ( 2\nabla^2 U + \mathcal C )$ is Lipschitz with the constant $C \norm{\nabla f}_\infty \log\frac1h$. The Kantorovich--Rubinstein duality and the bound \eqref{eq: w1 rough sgld} then give
    \begin{equation}
        \Bigl| ( \bar\pi_h - \pi )\Bigl( \nabla^2 ( e^{h\Delta} \phi_h ) : \bigl( 2\nabla^2 U + \mathcal C \bigr) \Bigr) \Bigr| \Le C \norm{\nabla f}_\infty h \log\frac1h .
        \label{eq: sgld hessian transfer}
    \end{equation}
    Assumption~\ref{ass: U} gives
    \begin{equation}
        \bigl| \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr)(x) - \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr)(x') \bigr| \Le C \bigl( 1 + |x| + |x'| \bigr) |x - x'| , \qquad x, x' \in \mathbb R^d .
        \label{eq: drift correction lipschitz}
    \end{equation}
    By the bounds \eqref{eq: smoothed poisson bound}, the bound \eqref{eq: drift correction lipschitz} and the linear growth of $\nabla^2 U\, \nabla U - \nabla \Delta U$, the function $\nabla ( e^{h\Delta} \phi_h ) \cdot ( \nabla^2 U\, \nabla U - \nabla \Delta U )$ satisfies the condition \eqref{eq: local lipschitz} with $K = C \norm{\nabla f}_\infty$. Let $h_0$ also satisfy $C h_0 \Le 1$ for the constant $C$ of \eqref{eq: w1 rough sgld}, so that $\mathcal W_1(\bar\pi_h, \pi) \Le 1$. Lemma~\ref{lem: local lipschitz} with $\mu = \bar\pi_h$ and $\nu = \pi$, the bound \eqref{eq: w1 rough sgld}, the moment bound $\bar\pi_h(|x|^{12}) \Le C$ of Lemma~\ref{lem: sgld invariant} and the moments of $\pi$ in Lemma~\ref{lem: dissipative} then give
    \begin{equation}
        \Bigl| ( \bar\pi_h - \pi )\Bigl( \nabla ( e^{h\Delta} \phi_h ) \cdot \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) \Bigr) \Bigr| \Le C \norm{\nabla f}_\infty h^{\frac{10}{11}} .
        \label{eq: sgld gradient transfer}
    \end{equation}
    Since $h^{\frac{1}{11}} \log\frac1h \Le C$, the discrete form \eqref{eq: sgld bias} and the bounds \eqref{eq: sgld hessian transfer} and \eqref{eq: sgld gradient transfer} give
    \begin{equation}
        \biggl| \bar\pi_h(f) - \pi(f) - \frac h2\, \pi\Bigl( \nabla^2 ( e^{h\Delta} \phi_h ) : \bigl( 2\nabla^2 U + \mathcal C \bigr) - \nabla ( e^{h\Delta} \phi_h ) \cdot \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) \Bigr) \biggr| \Le C \norm{\nabla f}_\infty h^{\frac{21}{11}} .
        \label{eq: sgld under pi}
    \end{equation}
    \textbf{The identification of the coefficient.} By the identity \eqref{eq: sgld integration by parts} with $\psi = e^{h\Delta} \phi_h$, which has bounded derivatives of orders $1$ to $3$ by \eqref{eq: smoothed poisson bound},
    \begin{equation}
        \pi\Bigl( \nabla^2 ( e^{h\Delta} \phi_h ) : \bigl( 2\nabla^2 U + \mathcal C \bigr) - \nabla ( e^{h\Delta} \phi_h ) \cdot \bigl( \nabla^2 U\, \nabla U - \nabla \Delta U \bigr) \Bigr) = \pi\Bigl( \nabla^2 ( e^{h\Delta} \phi_h ) : \bigl( \nabla^2 U + \mathcal C \bigr) \Bigr) .
        \label{eq: sgld coefficient riesz}
    \end{equation}
    Next we prove, with Lemma~\ref{lem: representor semigroup} for $\mathcal A = \nabla^2 U + \mathcal C$ and its Riesz representor $\psi_{\nabla^2 U} + \psi_{\mathcal C}$,
    \begin{equation}
        \Bigl| \pi\bigl( ( \psi_{\nabla^2 U} + \psi_{\mathcal C} )\, f \bigr) - \pi\Bigl( \nabla^2 ( e^{h\Delta} \phi_h ) : \bigl( \nabla^2 U + \mathcal C \bigr) \Bigr) \Bigr| \Le C \norm{\nabla f}_\infty h \log\frac1h .
        \label{eq: sgld coefficient identification}
    \end{equation}
    By \eqref{eq: discrete poisson}, $e^{h\mathcal L} \phi_h - e^{h\mathcal L} ( e^{h\mathcal L} \phi_h ) = h\, \bigl( e^{h\mathcal L} f - \pi(f) \bigr)$. The bound \eqref{eq: representor semigroup} for $\mathcal L$ with $\omega = e^{h\mathcal L} \phi_h$, $\pi( \psi_{\nabla^2 U} + \psi_{\mathcal C} ) = 0$ and the bounds \eqref{eq: poisson bound} then give
    \begin{equation*}
        \Bigl| \pi\bigl( ( \psi_{\nabla^2 U} + \psi_{\mathcal C} )\, e^{h\mathcal L} f \bigr) - \pi\Bigl( \nabla^2 ( e^{h\mathcal L} \phi_h ) : \bigl( \nabla^2 U + \mathcal C \bigr) \Bigr) \Bigr| \Le C \norm{\nabla f}_\infty h \log\frac1h .
    \end{equation*}
    To replace $e^{h\mathcal L} f$ by $f$, the identity \eqref{eq: backward equation} applied to $f$, the identity in \eqref{eq: representor gradient} with $\omega = e^{s\mathcal L} f$, the decay \eqref{eq: grad decay} and the bound \eqref{eq: psi moments} give
    \begin{equation*}
        \Bigl| \pi\bigl( ( \psi_{\nabla^2 U} + \psi_{\mathcal C} )\, ( e^{h\mathcal L} f - f ) \bigr) \Bigr| = \biggl| \int_0^h \pi\Bigl( \nabla ( \psi_{\nabla^2 U} + \psi_{\mathcal C} ) \cdot \nabla ( e^{s\mathcal L} f ) \Bigr) \D s \biggr| \Le C \norm{\nabla f}_\infty h .
    \end{equation*}
    To replace $e^{h\mathcal L} \phi_h$ by $e^{h\Delta} \phi_h$, we write $e^{h\Delta} \phi_h - e^{h\mathcal L} \phi_h = h\, e^{h\Delta} ( f - \pi(f) ) + ( e^{h\Delta} - \mathcal I ) e^{h\mathcal L} \phi_h$ by \eqref{eq: discrete poisson}. The bound \eqref{eq: gaussian derivatives} at $j = 1$ gives $\norm{\nabla ( h\, e^{h\Delta} f )}_\infty \Le C \norm{\nabla f}_\infty h$, and the identity \eqref{eq: heat shift integral}, with $h$ in place of $\frac h4$ and $\psi = \nabla ( e^{h\mathcal L} \phi_h )$, and the bounds \eqref{eq: poisson bound} give $\norm{\nabla ( ( e^{h\Delta} - \mathcal I ) e^{h\mathcal L} \phi_h )}_\infty \Le d\, h\, \norm{\nabla^3 ( e^{h\mathcal L} \phi_h )}_\infty \Le C \norm{\nabla f}_\infty h \log\frac1h$. The identity in \eqref{eq: representor gradient} with $\omega = e^{h\Delta} \phi_h - e^{h\mathcal L} \phi_h$ and the bound \eqref{eq: psi moments} then give
    \begin{equation*}
        \Bigl| \pi\Bigl( \nabla^2 ( e^{h\Delta} \phi_h - e^{h\mathcal L} \phi_h ) : \bigl( \nabla^2 U + \mathcal C \bigr) \Bigr) \Bigr| \Le C \norm{\nabla f}_\infty h \log\frac1h .
    \end{equation*}
    The bound from \eqref{eq: representor semigroup} and the two replacements prove \eqref{eq: sgld coefficient identification}.

    The bound \eqref{eq: sgld under pi}, the identity \eqref{eq: sgld coefficient riesz}, the bound \eqref{eq: sgld coefficient identification} and the identity \eqref{eq: representor poisson} for $\mathcal A = \nabla^2 U + \mathcal C$ prove \eqref{eq: sgld bias continuous} for $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$.\\[6pt]
    \textbf{A Lipschitz $f$.} A mollification of $f$ gives \eqref{eq: sgld bias} and \eqref{eq: sgld bias continuous} for every Lipschitz function $f$. Finally, the Kantorovich--Rubinstein duality and the bound \eqref{eq: w1 rough sgld} give $|\bar\pi_h(f) - \pi(f)| \Le C \norm{\nabla f}_\infty h$.
\end{proof}



\section{Proofs for the explicit coefficient in Wasserstein-1 error}
\label{app: w1}

Throughout this appendix, the constant $C$ depends only on $m$, $M$, $R$ and $d$.



\subsection{Proof of Corollary~\ref{cor: w1 limits}: limits of the Wasserstein-1 errors}
\label{sec: w1 limits proof}

\begin{proof}[Proof of Corollary~\ref{cor: w1 limits}]
    Dividing \eqref{eq: w1 limits} by $h$ and letting $h \to 0$ gives both limits of Corollary~\ref{cor: w1 limits}.

    For the positivity, a function $\psi \in L^2_0(\pi)$ with $\norm{\psi\, \pi}_{\mathrm{KR}} = 0$ has $\pi(f \psi) = 0$ for every $f \in C_c^\infty(\mathbb R^d)$, hence $\psi = 0$ $\pi$-almost everywhere. By \eqref{eq: psi hessian}, $\psi_{\nabla^2 U}$ is continuous and unbounded, since $|\Delta U| \Le d M$ while $|\nabla U(x)| \Ge \langle x, \nabla U(x) \rangle / |x| \to \infty$ as $|x| \to \infty$ by \eqref{eq: dissipative bounds}. Hence $\psi_{\nabla^2 U}$ is nonzero on a nonempty open set, which has positive $\pi$-measure since $\pi(x) > 0$ for every $x \in \mathbb R^d$, and $\norm{\psi_{\nabla^2 U}\, \pi}_{\mathrm{KR}} > 0$. For $\psi_{\mathcal C}$, let $r > 0$ and
    \begin{equation}
        \varphi_r(x) := \bigl( r^2 + |x|^2 \bigr)^{\frac12} , \qquad x \in \mathbb R^d ,
        \label{eq: phi r}
    \end{equation}
    so that $\max_{1 \Le j \Le 4} \norm{\nabla^j \varphi_r}_\infty < \infty$ and
    \begin{equation}
    \begin{gathered}
        |\nabla \varphi_r| \Le 1 , \qquad \nabla^2 \varphi_r = \frac{\varphi_r^2 I_d - x \otimes x}{\varphi_r^3} \succeq \frac{r^2 I_d}{\varphi_r^3} , \qquad \norm{\nabla^2 \varphi_r} \Le \frac{1}{\varphi_r} , \\
        \Delta \varphi_r = \frac{d-1}{\varphi_r} + \frac{r^2}{\varphi_r^3} , \qquad |\nabla \Delta \varphi_r| = \frac{(d-1)\, |x|}{\varphi_r^3} + \frac{3 r^2 |x|}{\varphi_r^5} \Le \frac{d+1}{2 r^2} .
    \end{gathered}
        \label{eq: phi r bounds}
    \end{equation}
    If $\norm{\psi_{\mathcal C}\, \pi}_{\mathrm{KR}} = 0$, then $\psi_{\mathcal C} = 0$ $\pi$-almost everywhere, and \eqref{eq: riesz} with $\omega = \varphi_1$, the bound on $\nabla^2 \varphi_1$ in \eqref{eq: phi r bounds} and $\mathcal C(x)$ positive semidefinite give
    \begin{equation*}
        0 = \pi\bigl( \psi_{\mathcal C}\, (-\mathcal L \varphi_1) \bigr) = \pi\bigl( \nabla^2 \varphi_1 : \mathcal C \bigr) \Ge \pi\biggl( \frac{\mathrm{Tr}\bigl[ \mathcal C(x) \bigr]}{(1 + |x|^2)^{\frac32}} \biggr) .
    \end{equation*}
    The trace $\mathrm{Tr}\bigl[ \mathcal C(x) \bigr]$ is nonnegative and continuous by \eqref{eq: covariance lipschitz}, so it vanishes identically, and so does $\mathcal C(x)$.
\end{proof}


\subsection{Proof of Lemma~\ref{lem: w1 counterexample}: failure of the comparison}
\label{sec: w1 counterexample proof}

\begin{proof}[Proof of Lemma~\ref{lem: w1 counterexample}]
    The proof checks the assumptions, writes the Kantorovich--Rubinstein norm in dimension $1$ as the integral \eqref{eq: kr one dimension}, computes $\psi_{\mathcal C}$ explicitly, and evaluates both norms of Lemma~\ref{lem: w1 counterexample} by quadrature. Assumption~\ref{ass: U} holds with $m = 1$, $R = 0$ and every $M \Ge 1$. The estimator is unbiased, $|\hat g(x;\xi)| \Le |x| + 10$, its derivative $1 + 10\, \xi\, a'(x)$ is bounded by $1 + 10 \norm{a'}_\infty$, and its variance is $100\, a(x)^2 \Le 100$, so Assumption~\ref{ass: sg} holds for $M$ large enough.

    Here $\pi$ is the standard Gaussian distribution, with $\pi'(x) = -x\, \pi(x)$. For $\psi \in L^2_0(\pi)$ and $s \in \mathbb R$, the Cauchy--Schwarz inequality on $(-\infty, s]$ for $s \Le 0$, and on $[s, \infty)$ for $s > 0$ since $\pi(\psi) = 0$, gives
    \begin{equation*}
        \biggl| \int_{-\infty}^s \psi(y)\, \pi(y)\, \D y \biggr| \Le \pi(\psi^2)^{\frac12} e^{-\frac14 s^2} ,
    \end{equation*}
    so for $f$ Lipschitz with $|f'| \Le 1$ an integration by parts gives
    \begin{equation*}
        \pi(f \psi) = -\int_{\mathbb R} f'(s) \int_{-\infty}^s \psi(y)\, \pi(y)\, \D y \,\D s \Le \int_{\mathbb R} \biggl| \int_{-\infty}^s \psi(y)\, \pi(y)\, \D y \biggr| \,\D s ,
    \end{equation*}
    with equality for $f(s) = -\int_0^s \mathrm{sign}\bigl( \int_{-\infty}^{s'} \psi(y)\, \pi(y)\, \D y \bigr) \D s'$. Hence
    \begin{equation}
        \norm{\psi\, \pi}_{\mathrm{KR}} = \int_{\mathbb R} \biggl| \int_{-\infty}^s \psi(y)\, \pi(y)\, \D y \biggr| \,\D s .
        \label{eq: kr one dimension}
    \end{equation}

    By \eqref{eq: psi hessian}, $\psi_{\nabla^2 U}(x) = \frac12 (x^2 - 1)$, and
    \begin{equation}
        \int_{-\infty}^s \psi_{\nabla^2 U}(y)\, \pi(y)\, \D y = -\frac12\, s\, \pi(s) , \qquad s \in \mathbb R .
        \label{eq: G hessian}
    \end{equation}
    For $\mathcal A = a^2$, let $J(x) := \int_{-\infty}^x y\, a(y)^2\, \D y$ and $\zeta := \pi(a^2) - \pi(J)$. The function $J - a^2 + \zeta$ is smooth and bounded, satisfies $\pi(J - a^2 + \zeta) = 0$, and has the derivative $x\, a^2 - (a^2)'$, so for $\omega \in C^4(\mathbb R)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j \omega}_\infty < \infty$, \eqref{eq: generator ibp} gives
    \begin{equation*}
        \pi\bigl( (J - a^2 + \zeta) (-\mathcal L \omega) \bigr) = \pi\bigl( (x\, a^2 - (a^2)')\, \omega' \bigr) = -\int_{\mathbb R} \omega'(x)\, \bigl( a(x)^2 \pi(x) \bigr)'\, \D x = \pi\bigl( \omega''\, a^2 \bigr) ,
    \end{equation*}
    which is \eqref{eq: riesz} for $\mathcal A = a^2$. By the uniqueness in Theorem~\ref{thm: riesz},
    \begin{equation}
        \psi_{a^2} = J - a^2 + \zeta .
        \label{eq: psi shells}
    \end{equation}
    Since $\mathcal C = 100\, a^2$, the linearity in Theorem~\ref{thm: riesz} gives $\psi_{\mathcal C} = 100\, \psi_{a^2}$. Quadrature in \eqref{eq: kr one dimension}, with \eqref{eq: G hessian} and \eqref{eq: psi shells}, then gives the values in Lemma~\ref{lem: w1 counterexample}, so \eqref{eq: w1 comparison} fails.
\end{proof}


\subsection{Proof of Theorem~\ref{thm: batch threshold}: threshold on the batch size}
\label{sec: batch threshold proof}

\begin{proof}[Proof of Theorem~\ref{thm: batch threshold}]
    The proof bounds $\norm{( \psi_{\nabla^2 U} + \psi_{\mathcal C} )\, \pi}_{\mathrm{KR}}$ from below by testing \eqref{eq: riesz} with a function of the distance to the minimizer of $U$, bounds $\norm{\psi_{\mathcal C}\, \pi}_{\mathrm{KR}}$ from above through \eqref{eq: bochner}, and compares the $2$ bounds. Let $x_*$ be the minimizer of $U$, and let $\varphi_r$ be the function of \eqref{eq: phi r} with $r^2 = d/m$ and $x - x_*$ in place of $x$. The gradient of $\mathcal L \varphi_r = \Delta \varphi_r - \nabla U \cdot \nabla \varphi_r$ satisfies
    \begin{equation}
        |\nabla (\mathcal L \varphi_r)| \Le 2\, \norm{\nabla^2 U}_\infty + \frac{d+1}{2r^2} \Le 3\, \norm{\nabla^2 U}_\infty ,
        \label{eq: lipschitz phi R}
    \end{equation}
    since \eqref{eq: phi r bounds} bounds $|\nabla \Delta \varphi_r|$ by $\frac{d+1}{2r^2} = \frac{(d+1)\, m}{2d} \Le m \Le \norm{\nabla^2 U}_\infty$, and, by $|\nabla \varphi_r| \Le 1$, $\norm{\nabla^2 \varphi_r} \Le \varphi_r^{-1}$ and $|\nabla U(x)| = |\nabla U(x) - \nabla U(x_*)| \Le \norm{\nabla^2 U}_\infty\, \varphi_r(x)$,
    \begin{equation*}
        |\nabla(\nabla U \cdot \nabla \varphi_r)| \Le \norm{\nabla^2 U}\, |\nabla \varphi_r| + \norm{\nabla^2 \varphi_r}\, |\nabla U| \Le 2 \norm{\nabla^2 U}_\infty .
    \end{equation*}

    By \eqref{eq: lipschitz phi R}, the function $-\mathcal L \varphi_r / (3\, \norm{\nabla^2 U}_\infty)$ is $1$-Lipschitz, and \eqref{eq: riesz} with $\omega = \varphi_r$ gives $\pi((-\mathcal L \varphi_r)\, \psi_{\mathcal A}) = \pi(\nabla^2 \varphi_r : \mathcal A)$ for $\mathcal A = \nabla^2 U$ and $\mathcal A = \mathcal C$. As $\nabla^2 \varphi_r$ and $\mathcal C(x)$ are positive semidefinite, $\pi(\nabla^2 \varphi_r : \mathcal C) \Ge 0$, and as $\nabla^2 U \succeq m I_d$, $\nabla^2 \varphi_r : \nabla^2 U \Ge m\, \Delta \varphi_r$. Hence
    \begin{equation}
        \norm{( \psi_{\nabla^2 U} + \psi_{\mathcal C} )\, \pi}_{\mathrm{KR}} \Ge \frac{\pi\bigl( (-\mathcal L \varphi_r)\, ( \psi_{\nabla^2 U} + \psi_{\mathcal C} ) \bigr)}{3\, \norm{\nabla^2 U}_\infty} \Ge \frac{m\, \pi( \Delta \varphi_r )}{3\, \norm{\nabla^2 U}_\infty} .
        \label{eq: threshold lower}
    \end{equation}
    The identity \eqref{eq: generator ibp} with $f_1 = 1$ and $f_2 = \frac12 |x - x_*|^2$ gives $\pi(\nabla U \cdot (x - x_*)) = d$, and $\nabla U(x) \cdot (x - x_*) \Ge m |x - x_*|^2$, so $\pi(|x - x_*|^2) \Le d/m$. As functions of $|x - x_*|^2$, $(r^2 + |x - x_*|^2)^{-\frac12}$ and $(r^2 + |x - x_*|^2)^{-\frac32}$ are convex and decreasing, so Jensen's inequality and $r^2 = d/m$ give
    \begin{equation*}
        \pi( \Delta \varphi_r ) \Ge \frac{d-1}{(2d/m)^{\frac12}} + \frac{d/m}{(2d/m)^{\frac32}} = \Bigl( \frac{m}{2d} \Bigr)^{\frac12} \Bigl( d - \frac12 \Bigr) \Ge \Bigl( \frac{m d}{8} \Bigr)^{\frac12} ,
    \end{equation*}
    and \eqref{eq: threshold lower} gives
    \begin{equation}
        \norm{( \psi_{\nabla^2 U} + \psi_{\mathcal C} )\, \pi}_{\mathrm{KR}} \Ge \frac{m^{\frac32} \sqrt d}{6 \sqrt2\, \norm{\nabla^2 U}_\infty} \Ge \frac{m^{\frac32} \sqrt d}{9\, \norm{\nabla^2 U}_\infty} .
        \label{eq: threshold lower explicit}
    \end{equation}

    For $\norm{\psi_{\mathcal C}\, \pi}_{\mathrm{KR}}$ in \eqref{eq: w1 comparison}, let $f \in C^4(\mathbb R^d)$ with $\max_{1 \Le j \Le 4} \norm{\nabla^j f}_\infty < \infty$ and $\norm{\nabla f}_\infty \Le 1$, and let $\phi_0$ be given by \eqref{eq: continuous poisson solution}, so that $\phi_0$ has bounded derivatives of orders $1$ to $4$ by \eqref{eq: regular decay}. By \eqref{eq: bochner} with $\omega = \phi_0$ and $\nabla^2 U \succeq m I_d$, and \eqref{eq: generator ibp},
    \begin{equation}
        \pi\bigl( \nabla^2 \phi_0 : \nabla^2 \phi_0 \bigr) + m\, \pi\bigl( |\nabla \phi_0|^2 \bigr) \Le \pi\bigl( (f - \pi(f))^2 \bigr) = \pi\bigl( \nabla f \cdot \nabla \phi_0 \bigr) \Le \pi\bigl( |\nabla \phi_0|^2 \bigr)^{\frac12} .
        \label{eq: bochner phi0}
    \end{equation}
    By \eqref{eq: bochner phi0}, $\pi(|\nabla \phi_0|^2)^{\frac12} \Le m^{-1}$ and then $\pi(\nabla^2 \phi_0 : \nabla^2 \phi_0) \Le m^{-1}$. The Cauchy--Schwarz inequality, as in the proof of \eqref{eq: riesz functional bound}, and \eqref{eq: riesz} with $\omega = \phi_0$ give $|\pi(f \psi_{\mathcal C})| = |\pi(\nabla^2 \phi_0 : \mathcal C)| \Le \sqrt d\, \norm{\mathcal C}_\infty\, m^{-\frac12}$, and the supremum over $f$ gives
    \begin{equation}
        \norm{\psi_{\mathcal C}\, \pi}_{\mathrm{KR}} \Le \frac{\sqrt d\, \norm{\mathcal C}_\infty}{m^{\frac12}} .
        \label{eq: threshold upper}
    \end{equation}
    Hence, whenever $\norm{\mathcal C}_\infty \Le m^2 / (9\, \norm{\nabla^2 U}_\infty)$, \eqref{eq: threshold upper} and \eqref{eq: threshold lower explicit} give
    \begin{equation*}
        \norm{\psi_{\mathcal C}\, \pi}_{\mathrm{KR}} \Le \frac{\sqrt d\, \norm{\mathcal C}_\infty}{m^{\frac12}} \Le \frac{m^{\frac32} \sqrt d}{9\, \norm{\nabla^2 U}_\infty} \Le \norm{( \psi_{\nabla^2 U} + \psi_{\mathcal C} )\, \pi}_{\mathrm{KR}} ,
    \end{equation*}
    which is \eqref{eq: w1 comparison}. For SGLM and SGLD with a batch of $P$ independent stochastic gradients, the averaged estimator satisfies Assumption~\ref{ass: sg} with $\Sigma/P$ in place of $\Sigma$ (Section~\ref{subsec: mse analysis}), so Corollary~\ref{cor: w1 limits} applies with its covariance $\mathcal C(x)/P$, and $\norm{\mathcal C}_\infty / P \Le m^2 / (9\, \norm{\nabla^2 U}_\infty)$ exactly when $P$ satisfies \eqref{eq: batch threshold}. The representor of $\mathcal C(x)/P$ is $\psi_{\mathcal C}/P$ by the linearity in Theorem~\ref{thm: riesz}, and \eqref{eq: w1 comparison} with $\psi_{\mathcal C}/P$ in place of $\psi_{\mathcal C}$ is \eqref{eq: w1 comparison batch}.
\end{proof}
